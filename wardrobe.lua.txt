-- 1. FULL SPEED & NORMAL MOVEMENT ENFORCER
-- ==========================================================
local function ensureNormalMovement()
	local char = lp.Character
	if not char then return end

	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum then
		if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
		if hum.JumpPower < 50 then hum.JumpPower = 50 end
		if hum.PlatformStand then hum.PlatformStand = false end
	end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	if hrp and hrp.Anchored then hrp.Anchored = false end

	local torso = char:FindFirstChild("LowerTorso") or char:FindFirstChild("Torso")
	if torso and torso.Anchored then torso.Anchored = false end

	local head = char:FindFirstChild("Head")
	if head then
		for _, c in ipairs(head:GetChildren()) do
			if c:IsA("BillboardGui") then c.Enabled = false end
		end
	end
	for _, c in ipairs(char:GetChildren()) do
		if c:IsA("BillboardGui") then c.Enabled = false end
	end

	for _, g in ipairs(pg:GetDescendants()) do
		if g:IsA("TextButton") or g:IsA("TextLabel") then
			local t = g.Text:lower()
			if t:find("exit try") or t:find("stop try") then
				g.Visible = false
				local p = g.Parent
				if p and p:IsA("Frame") and #p:GetChildren() <= 3 then
					p.Visible = false
				end
			end
		end
	end

	pcall(function()
		ContextActionService:UnbindAction("FreezeMovement")
		ContextActionService:UnbindAction("ShopFreeze")
		ContextActionService:UnbindAction("TryOnFreeze")
	end)
end

RunService.RenderStepped:Connect(function() pcall(ensureNormalMovement) end)
RunService.Heartbeat:Connect(function() pcall(ensureNormalMovement) end)

local savedItems = {}

	["Heels"] = "HeelMorphs",
	["Skirts"] = "Skirts",

local function equipItemDirectly(itemName, cardRef)
	pcall(function()
		local pe = pg.CharacterCreation.Inventory:FindFirstChild("PromptEquip")
		if pe and pe:IsA("BindableEvent") then
			pe:Fire(itemName)
		end
	end)

	if cardRef then
		pcall(function()
			local btn = cardRef:FindFirstChild("Button") or cardRef:FindFirstChild("Tryon")
			if btn and btn:IsA("TextButton") then
				for _, con in pairs(getconnections(btn.MouseButton1Click)) do
					con:Fire()
				end
			end
		end)
	end

	task.spawn(function()
		for i = 1, 15 do
			task.wait(0.05)
			ensureNormalMovement()
		end
	end)
end

local function performCardInjection(data)
	local cc = pg:FindFirstChild("CharacterCreation")
	if not cc then return end
	local inv = cc:FindFirstChild("Inventory")
	if not inv then return end
	local inner = inv:FindFirstChild("Inner")
	if not inner or not inner:IsA("ScrollingFrame") then return end

	local itemName = data.name
	local itemImage = data.image
	local gameCatName = categoryMapping[data.category] or data.category or "Accessories"

	local existing = inner:FindFirstChild(itemName)
	if existing then
		local cVal = existing:FindFirstChild("Category")
		if cVal and cVal:IsA("StringValue") then
			cVal.Value = gameCatName
		end
		return existing
	end

	local template = inv:FindFirstChild("ItemTemplate")
	local newCard
	if template and template:IsA("TextButton") then
		newCard = template:Clone()
	else
		newCard = Instance.new("TextButton")
		newCard.Size = UDim2.new(0, 72, 0, 72)
		newCard.BackgroundColor3 = Color3.fromRGB(24, 24, 24)
		newCard.BorderSizePixel = 0
	end

	newCard.Name = itemName
	newCard.Visible = true

	local tip = newCard:FindFirstChild("Tooltip")
	if tip and tip:IsA("TextLabel") then tip.Text = itemName end

	local imgBtn = newCard:FindFirstChild("Image")
	if imgBtn and (imgBtn:IsA("ImageLabel") or imgBtn:IsA("ImageButton")) then
		if itemImage and itemImage ~= "" then imgBtn.Image = itemImage end
	end

	local q = newCard:FindFirstChild("QuantityLabel")
	if q and q:IsA("TextLabel") then q.Text = "1" end

	local catVal = newCard:FindFirstChild("Category")
	if not catVal then
		catVal = Instance.new("StringValue")
		catVal.Name = "Category"
		catVal.Parent = newCard
	end
	catVal.Value = gameCatName

	local idVal = newCard:FindFirstChild("Id")
	if not idVal then
		idVal = Instance.new("StringValue")
		idVal.Name = "Id"
		idVal.Parent = newCard
	end
	idVal.Value = itemName

	newCard.MouseButton1Click:Connect(function()
		equipItemDirectly(itemName, data.cardRef)
	end)

	newCard.Parent = inner
	return newCard
end

local function addToDressUp(itemName, itemImage, uiCatName, cardRef)
	pcall(function()
		savedItems[itemName] = {
			name = itemName,
			image = itemImage,
			category = uiCatName,
			cardRef = cardRef,
		}
		performCardInjection(savedItems[itemName])
	end)
end

task.spawn(function()
	while true do
		task.wait(0.8)
		pcall(function()
			local cc = pg:FindFirstChild("CharacterCreation")
			local inv = cc and cc:FindFirstChild("Inventory")
			local inner = inv and inv:FindFirstChild("Inner")
			if inner then
				for name, data in pairs(savedItems) do
					if not inner:FindFirstChild(name) then
						performCardInjection(data)
					end
				end
			end
		end)
	end
end)

local function findItemImage(card)
	local cur = card:FindFirstChild("CurrentImage", true)
	if cur and cur:IsA("ImageLabel") and cur.Image ~= "" and not ignoredImages[cur.Image] then
		return cur.Image, cur.ImageRectOffset, cur.ImageRectSize
	end

	local img = card:FindFirstChild("Image")
	if img and img:IsA("ImageLabel") and img.Image ~= "" and not ignoredImages[img.Image] then
		return img.Image, img.ImageRectOffset, img.ImageRectSize
	end

	local prev = card:FindFirstChild("Preview")
	if prev and prev:IsA("ImageLabel") and prev.Image ~= "" and not ignoredImages[prev.Image] then
		return prev.Image, prev.ImageRectOffset, prev.ImageRectSize
	end

	for _, child in ipairs(card:GetDescendants()) do
		if (child:IsA("ImageLabel") or child:IsA("ImageButton")) and child.Visible then
			local id = tostring(child.Image)
			if id ~= "" and not ignoredImages[id] 
				and child.Name ~= "ButtonBackground" 
				and child.Name ~= "Imagenon" 
				and child.Name ~= "BackArrow" 
				and child.Name ~= "NextArrow" 
				and child.Name ~= "Left" 
				and child.Name ~= "Right" then
				return id, child.ImageRectOffset, child.ImageRectSize
			end
		end
	end

	return "", Vector2.new(0, 0), Vector2.new(0, 0)
end

local function processLiveCard(card, defaultCat)
	if not card:IsA("Frame") or card.Name == "UIListLayout" or card.Name == "UIGridLayout" or card.Name == "UIPadding" then return end

	local nameObj = card:FindFirstChild("name")
	local itemName = (nameObj and nameObj:IsA("TextLabel") and nameObj.Text ~= "") and nameObj.Text or card.Name
	if itemName == "" or itemName == " " then return end

	local priceObj = card:FindFirstChild("price")
	local itemPrice = (priceObj and priceObj:IsA("TextLabel")) and priceObj.Text or ""
	itemPrice = itemPrice:gsub("ðŸ’Ž", "Diamonds: "):gsub("💎", "Diamonds: ")
	itemPrice = itemPrice:gsub("ðŸ ¬", "Candy: "):gsub("🍬", "Candy: ")
	itemPrice = itemPrice:gsub("ðŸ’µ", "Cash: "):gsub("💵", "Cash: ")

	local descObj = card:FindFirstChild("smalldescription")
	local itemDesc = ""
	if descObj and descObj.Text ~= "" and descObj.Text ~= " " then
		itemDesc = descObj.Text
	else
		local noteObj = card:FindFirstChild("note")
		if noteObj and noteObj.Text ~= "" and noteObj.Text ~= "Sparkles!" then
			itemDesc = noteObj.Text
		end
	end

	local hasToggles = card:FindFirstChild("sliderwrapper") ~= nil or card:FindFirstChild("CurrentImage") ~= nil
	local imgId, rOff, rSz = findItemImage(card)

	local targetCat = defaultCat
	local lowerName = itemName:lower()
	if defaultCat == "Accessories" or defaultCat == "Seasonal" then
		if lowerName:find("heel") or lowerName:find("boot") or lowerName:find("shoe") or lowerName:find("loafer") 
			or lowerName:find("platform") or lowerName:find("footwear") or lowerName:find("sandal") or lowerName:find("sneaker") then
			targetCat = "Heels"
		elseif lowerName:find("skirt") or lowerName:find("dress") or lowerName:find("tut") then
			targetCat = "Skirts"
		elseif lowerName:find("wing") then
			targetCat = "Wings"
		else
			targetCat = "Accessories"
		end
	end

	addItem(targetCat, {
		name = itemName,
		price = itemPrice,
		desc = itemDesc,
		image = imgId,
		rectOff = rOff,
		rectSize = rSz,
		hasToggles = hasToggles,
		cardRef = card,
	})
end

local function scrapeScrollFrame(sf, defaultCat)
	if not sf then return end
	for _, child in ipairs(sf:GetChildren()) do
		processLiveCard(child, defaultCat)
	end
end

local function mk(cls, props, parent)
	local o = Instance.new(cls)
	for k, v in pairs(props) do o[k] = v end
	if parent then o.Parent = parent end
	return o
end

local function corner(r, parent)
	return mk("UICorner", {CornerRadius = UDim.new(0, r)}, parent)
end

local function stroke(t, c, parent)
	return mk("UIStroke", {Thickness = t, Color = c, ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, parent)
end

local function formatCommas(str)
	local numStr = str:gsub("[^%d]", "")
	if numStr == "" then return "" end
	local k
	while true do
		numStr, k = string.gsub(numStr, "^(-?%d+)(%d%d%d)", '%1,%2')
		if k == 0 then break end
	end
	return numStr
end

local function createStatRow(labelText, defaultVal, callback)
	local row = mk("Frame", {
		Size = UDim2.new(1, 0, 0, 48),
		BackgroundColor3 = C.bg,
		BorderSizePixel = 0,
		ZIndex = 13,
	}, StatsScroll)

		BackgroundTransparency = 1,
		Text = labelText:upper(),
		TextColor3 = C.sub,
		Font = Enum.Font.GothamBold,
		TextSize = 8.5,
		TextXAlignment = Enum.TextXAlignment.Left,
		ZIndex = 14,
	}, row)

	local box = mk("TextBox", {
		Size = UDim2.new(1, -12, 0, 22),
		Position = UDim2.new(0, 6, 0, 22),
		BackgroundColor3 = C.card,
		Text = defaultVal,
		PlaceholderText = "Enter amount...",
		TextColor3 = C.text,
		PlaceholderColor3 = C.muted,
		Font = Enum.Font.GothamBold,
		TextSize = 10,
		BorderSizePixel = 0,
		ClearTextOnFocus = false,
		ZIndex = 14,
	}, row)
	corner(3, box)
	stroke(1, C.border, box)

	box:GetPropertyChangedSignal("Text"):Connect(function()
		local raw = box.Text
		local formatted = formatCommas(raw)
		if raw ~= formatted then box.Text = formatted end
		activeSpoofs[labelText] = formatted
		callback(formatted)
	end)
end

local function updateDiamonds(val)
	if val == "" then return end
	pcall(function() pg.Shop.Header.diamondamount.Text = val end)
	pcall(function() pg.HUD.Frame.Middle.DiamondsFrame.DiamondAmount.Text = val end)
	pcall(function() pg.MermaidShopDiamondCount.Frame.DiamondAmount.Text = val end)
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("TextLabel") and (d.Name:lower():find("diamondamount") or d.Name:lower():find("diamondcount")) then
			pcall(function() d.Text = val end)
		end
	end
end

local function updateLevel(val)
	if val == "" then return end
	pcall(function() pg.TopBar.Topbar.ButtonList["NOT IN USE"].LevelNumber.Text = val end)
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("TextLabel") and (d.Name:lower():find("levelnumber") or (d.Name:lower() == "level" and tonumber(d.Text:gsub("[^%d]", "")))) then
			pcall(function() d.Text = val end)
		end
	end
end

local function updateCandy(val)
	if val == "" then return end
	pcall(function() pg.Shop.Header.candyamount.Text = val end)
	pcall(function() pg.HUD.Frame.HalloweenCandy.CandyAmount.Text = val end)
end

local function updateCash(val)
	if val == "" then return end
	pcall(function() pg.HUD.Frame.Middle.CloudCashStuff.CloudCash.Text = val end)
end

local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = Main.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UIS.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)
