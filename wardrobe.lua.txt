local Players = game:GetService("Players")
local StarterGui = game:GetService("StarterGui")
local lp = Players.LocalPlayer
local pg = lp:WaitForChild("PlayerGui")

local function notify(title, msg)
	pcall(function()
		StarterGui:SetCore("SendNotification", { Title = title, Text = msg, Duration = 5 })
	end)
end

task.spawn(function()
	local dir = pg:FindFirstChild("Shop") and pg.Shop:FindFirstChild("Directory")
	if not dir then
		notify("ERROR", "Shop Directory not found. Open Shop first!")
		return
	end

	local lines = {}
	table.insert(lines, "=== GETASSETLIST DATA DUMP ===")

	local cats = {"Skirts", "Shoes", "Accessories", "Wings"}
	for _, catName in ipairs(cats) do
		local catFolder = dir:FindFirstChild(catName)
		local getList = catFolder and catFolder:FindFirstChild("GetAssetList")
		if getList and getList:IsA("BindableFunction") then
			local s, res = pcall(function() return getList:Invoke() end)
			if s and type(res) == "table" then
				local count = 0
				for _ in pairs(res) do count = count + 1 end
				table.insert(lines, catName .. " Asset Count: " .. tostring(count))
				local sample = 0
				for k, v in pairs(res) do
					sample = sample + 1
					if sample <= 2 then
						table.insert(lines, "  Sample: Key=" .. tostring(k) .. ", Val=" .. tostring(v))
					end
				end
			else
				table.insert(lines, catName .. ": Invoke failed or returned non-table")
			end
		else
			table.insert(lines, catName .. ": GetAssetList BindableFunction not found")
		end
	end

	table.insert(lines, "==============================")
	local report = table.concat(lines, "\n")
	if setclipboard then setclipboard(report) elseif toclipboard then toclipboard(report) end
	notify("DONE!", "Data dump copied to clipboard!")
	print(report)
end)
