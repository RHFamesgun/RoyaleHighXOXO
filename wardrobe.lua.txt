-- ==========================================================
-- DRESS-UP ARGUMENT & THUMBNAIL PROBE
-- ==========================================================
local rep = game:GetService("ReplicatedStorage")
local pg = game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")

task.spawn(function()
    local out = {}
    local function log(s) table.insert(out, s) print(s) end
    log("=== DRESS-UP ARGUMENT & BUTTON PROBE ===")

    -- 1. Test GetButtonCopy on an item
    local shoesDir = pg:FindFirstChild("Shop") and pg.Shop.Directory:FindFirstChild("Shoes")
    if shoesDir and shoesDir:FindFirstChild("GetButtonCopy") then
        local s, res = pcall(function()
            return shoesDir.GetButtonCopy:Invoke("Bat-Mo-Heels")
        end)
        log("Shoes GetButtonCopy('Bat-Mo-Heels') success: " .. tostring(s))
        if s and res and typeof(res) == "Instance" then
            log("  -> Returned Instance: " .. res.ClassName .. " | Name: " .. res.Name)
            for _, c in ipairs(res:GetDescendants()) do
                if c:IsA("ImageLabel") or c:IsA("ImageButton") then
                    log("  -> Image child: " .. c.Name .. " = " .. tostring(c.Image) .. " (RectSize: " .. tostring(c.ImageRectSize) .. ")")
                end
            end
        end
    end

    -- 2. Test GetInventoryCategoryItems
    local invRF = rep:FindFirstChild("Inventory") and rep.Inventory:FindFirstChild("GetInventoryCategoryItems")
    if invRF then
        for _, cat in ipairs({"Shoes", "Skirts", "Accessories", "Wings", "All"}) do
            local s, res = pcall(function() return invRF:InvokeServer(cat) end)
            log("GetInventoryCategoryItems('" .. cat .. "') -> s=" .. tostring(s) .. " type=" .. typeof(res))
            if s and type(res) == "table" then
                local count = 0
                for k, v in pairs(res) do
                    count = count + 1
                    if count <= 2 then
                        log("    Sample item [" .. tostring(k) .. "]: type=" .. typeof(v))
                        if type(v) == "table" then
                            for subK, subV in pairs(v) do
                                log("       " .. tostring(subK) .. " = " .. tostring(subV))
                            end
                        end
                    end
                end
                log("    Total items returned for " .. cat .. ": " .. count)
            end
        end
    end

    -- 3. Test PromptEquip structure in CharacterCreation
    local cc = pg:FindFirstChild("CharacterCreation")
    local promptEquip = cc and cc.Inventory:FindFirstChild("PromptEquip")
    log("PromptEquip BindableEvent found: " .. tostring(promptEquip ~= nil))

    log("========================================")
    local report = table.concat(out, "\n")
    if setclipboard then
        setclipboard(report)
        log("[!] Copied report to clipboard!")
    end
end)
