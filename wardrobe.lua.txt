local function unfreezeControls()
	pcall(function()
		local PlayerModule = lp:FindFirstChild("PlayerScripts") and lp.PlayerScripts:FindFirstChild("PlayerModule")
		if PlayerModule then
			local Controls = require(PlayerModule)
			if Controls and Controls.GetControls then
				Controls:GetControls():Enable()
			end
		end
	end)
	pcall(function()
		ContextActionService:UnbindAction("FreezeMovement")
		ContextActionService:UnbindAction("ShopFreeze")
		ContextActionService:UnbindAction("TryOnFreeze")
		ContextActionService:UnbindAction("DisableMovement")
	end)
end

local function deepClone(inst)
	if not inst then return nil end
	pcall(function()
		inst.Archivable = true
		for _, d in ipairs(inst:GetDescendants()) do
			d.Archivable = true
		end
	end)
	return inst:Clone()
end

local function reattachModelJoints(mod, char, joints)
	for _, wd in ipairs(joints) do
		local p0 = mod:FindFirstChild(wd.part0Name, true)
		local p1 = char:FindFirstChild(wd.part1Name)
		if p0 and p1 then
			if wd.className == "Motor6D" then
				local m = Instance.new("Motor6D")
				m.Part0 = p0
				m.Part1 = p1
				if wd.c0 then m.C0 = wd.c0 end
				if wd.c1 then m.C1 = wd.c1 end
				m.Parent = p0
			elseif wd.className == "WeldConstraint" then
				local wc = Instance.new("WeldConstraint")
				wc.Part0 = p0
				wc.Part1 = p1
				wc.Parent = p0
			else
				local w = Instance.new("Weld")
				w.Part0 = p0
				w.Part1 = p1
				if wd.c0 then w.C0 = wd.c0 end
				if wd.c1 then w.C1 = wd.c1 end
				w.Parent = p0
			end
		end
	end
end

local function restoreAllSavedLayers()
	local char = lp.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")

	for rawName, data in pairs(savedWornLayers) do
		local layerName = rawName .. "_Layer"
		local existing = char:FindFirstChild(layerName) or char:FindFirstChild(rawName)
		if not existing and data.backup then
			pcall(function()
				local copy = deepClone(data.backup)
				copy.Name = layerName
				copy:SetAttribute("CustomSavedLayer", true)

				if data.isAccessory and hum then
					hum:AddAccessory(copy)
				elseif data.isModel then
					reattachModelJoints(copy, char, data.joints or {})
					copy.Parent = char
				else
					copy.Parent = char
				end
			end)
		end
	end
end

local function clearAllWornLayers()
	local char = lp.Character
	for rawName, data in pairs(savedWornLayers) do
		if char then
			local layerName = rawName .. "_Layer"
			local i1 = char:FindFirstChild(layerName)
			local i2 = char:FindFirstChild(rawName)
			if i1 then
				i1:SetAttribute("CustomSavedLayer", nil)
				pcall(function() i1:Destroy() end)
			end
			if i2 then
				i2:SetAttribute("CustomSavedLayer", nil)
				pcall(function() i2:Destroy() end)
			end
		end
	end
	savedWornLayers = {}

	if char then
		for _, child in ipairs(char:GetChildren()) do
			if child:GetAttribute("CustomSavedLayer") or child.Name:find("_Layer") then
				child:SetAttribute("CustomSavedLayer", nil)
				pcall(function() child:Destroy() end)
			end
		end
	end
end

			if char and next(savedWornLayers) ~= nil then

		end)
	end
end)

local seenItems = {}

local function addItem(cat, item)
	if not item.name or item.name == "" or item.name == " " then return end
	if seenItems[item.name] then return end
	seenItems[item.name] = true

	item.hasRealImage = (item.image ~= nil and item.image ~= "" and item.image:find("rbxassetid://") ~= nil)

	if not categories[cat] then categories[cat] = {} end
	table.insert(categories[cat], item)
end

local function classifyCategory(itemName, defaultCat)
	local lower = itemName:lower()
	if lower:find("halo") then
		return "Halos"
	elseif lower:find("heel") or lower:find("boot") or lower:find("shoe") or lower:find("loafer") 
		or lower:find("platform") or lower:find("footwear") or lower:find("sandal") or lower:find("sneaker") then
		return "Heels"
	elseif lower:find("skirt") or lower:find("dress") or lower:find("tut") then
		return "Skirts"
	elseif lower:find("bodice") or lower:find("corset") or lower:find("bustier") or lower:find("torso") then
		return "Bodices"
	elseif lower:find("wing") then
		return "Wings"
	else
		return defaultCat or "Accessories"
	end
end

local function ingestFromGetAssetList()
	local shop = pg:FindFirstChild("Shop")
	local dir = shop and shop:FindFirstChild("Directory")
	if not dir then return end

	local mapping = {
		["Skirts"] = { folder = dir:FindFirstChild("Skirts"), cat = "Skirts" },
		["Shoes"] = { folder = dir:FindFirstChild("Shoes"), cat = "Heels" },
		["Accessories"] = { folder = dir:FindFirstChild("Accessories"), cat = "Accessories" },
		["Wings"] = { folder = dir:FindFirstChild("Wings"), cat = "Wings" },
	}

	for key, info in pairs(mapping) do
		if info.folder then
			local getList = info.folder:FindFirstChild("GetAssetList")
			if getList and getList:IsA("BindableFunction") then
				pcall(function()
					local items = getList:Invoke()
					if type(items) == "table" then
						for k, v in pairs(items) do
							if type(v) == "table" then
								local name = v.Name or v.name or v.ItemName or v.itemName or v.Title or (type(k) == "string" and k)
								if name and type(name) == "string" and name ~= "" then
									local price = v.Price or v.price or v.Cost or v.cost or ""
									local desc = v.Description or v.description or v.Desc or v.desc or (name .. " (" .. info.cat .. ")")
									local img = v.Image or v.image or v.Icon or v.icon or v.Thumbnail or v.thumbnail or ""
									if type(img) == "number" then img = "rbxassetid://" .. tostring(img) end
									local cat = classifyCategory(name, info.cat)
									addItem(cat, {
										name = name,
										price = tostring(price),
										desc = tostring(desc),
										image = tostring(img),
										rectOff = v.ImageRectOffset or Vector2.new(0, 0),
										rectSize = v.ImageRectSize or Vector2.new(0, 0),
										hasToggles = (v.Toggles ~= nil or v.hasToggles ~= nil),
										cardRef = nil,
									})
								end
							elseif type(v) == "string" and v ~= "" then
								local cat = classifyCategory(v, info.cat)
								addItem(cat, {
									name = v,
									price = "Special",
									desc = v .. " (" .. cat .. ")",
									image = "",
									rectOff = Vector2.new(0, 0),
									rectSize = Vector2.new(0, 0),
									hasToggles = false,
									cardRef = nil,
								})
							end
						end
					end
				end)
			end
		end
	end
end

local function virtualClick(btn)
	if not btn then return false end
	local pos = btn.AbsolutePosition
	local sz = btn.AbsoluteSize
	local cx = pos.X + sz.X / 2
	local cy = pos.Y + sz.Y / 2

	if VIM then
		pcall(function()
			VIM:SendMouseButtonEvent(cx, cy, 0, true, game, 1)
			task.wait(0.04)
			VIM:SendMouseButtonEvent(cx, cy, 0, false, game, 1)
		end)
		return true
	end

	pcall(function()
		if firesignal then firesignal(btn.MouseButton1Click) end
		if getconnections then
			for _, c in pairs(getconnections(btn.MouseButton1Click)) do c:Fire() end
			for _, c in pairs(getconnections(btn.Activated)) do c:Fire() end
		end
	end)
	return true
end

local function searchShopForItem(itemName)
	local shop = pg:FindFirstChild("Shop")
	if not shop then return end

	for _, d in ipairs(shop:GetDescendants()) do
		if d:IsA("TextBox") and (d.Name:lower():find("search") or (d.PlaceholderText and d.PlaceholderText:lower():find("search"))) then
			pcall(function()
				d.Text = itemName
				if getconnections then
					for _, c in pairs(getconnections(d.FocusLost)) do c:Fire(true) end
				end
			end)
		end
	end

	for _, d in ipairs(shop:GetDescendants()) do
		if (d:IsA("TextButton") or d:IsA("ImageButton")) and (d.Name:lower():find("offsale") or (d:IsA("TextButton") and d.Text:lower():find("offsale"))) then
			pcall(function()
				virtualClick(d)
			end)
		end
	end
end

local function findLiveShopCard(itemName)
	local shop = pg:FindFirstChild("Shop")
	if not shop then return nil end

	local lower = itemName:lower()
	for _, d in ipairs(shop:GetDescendants()) do
		if d:IsA("Frame") and d:FindFirstChild("Tryon") then
			local n = d:FindFirstChild("name")
			if (n and n:IsA("TextLabel") and n.Text:lower() == lower) or d.Name:lower() == lower then
				return d
			end
		end
	end

	searchShopForItem(itemName)
	task.wait(0.12)

	for _, d in ipairs(shop:GetDescendants()) do
		if d:IsA("Frame") and d:FindFirstChild("Tryon") then
			local n = d:FindFirstChild("name")
			if (n and n:IsA("TextLabel") and n.Text:lower() == lower) or d.Name:lower() == lower then
				return d
			end
		end
	end

	return nil
end

local function triggerShopTryon(card)
	if not card then return false end
	local btn = card:FindFirstChild("Tryon") or card:FindFirstChild("Button") or card:FindFirstChildWhichIsA("TextButton")
	if not btn then return false end

	return virtualClick(btn)
end

local function equipItemWithLayering(itemName, itemCat, cardRef)
	local char = lp.Character
	if not char then return end

	if itemCat == "Skirts" or itemCat == "Heels" then
		for name, data in pairs(savedWornLayers) do
			if data.category == itemCat then
				local lyr = char:FindFirstChild(name .. "_Layer") or char:FindFirstChild(name)
				if lyr then
					lyr:SetAttribute("CustomSavedLayer", nil)
					pcall(function() lyr:Destroy() end)
				end
				savedWornLayers[name] = nil
			end
		end
	end

	local preExisting = {}
	for _, child in ipairs(char:GetChildren()) do
		if child:IsA("Accessory") or child:IsA("Model") then
			preExisting[child] = true
		end
	end

	local card = cardRef or findLiveShopCard(itemName)
	triggerShopTryon(card)

	task.spawn(function()
		local newItem = nil
		for step = 1, 16 do
			task.wait(0.06)
			for _, child in ipairs(char:GetChildren()) do
				if (child:IsA("Accessory") or child:IsA("Model")) 
				   and not standardBodyParts[child.Name] 
				   and not preExisting[child] 
				   and not child:GetAttribute("CustomSavedLayer") then
					newItem = child
					break
				end
			end
			if newItem then break end
		end

		if newItem then
			pcall(function()
				newItem:SetAttribute("CustomSavedLayer", true)
				local jointsList = {}
				for _, w in ipairs(newItem:GetDescendants()) do
					if (w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D")) and w.Part1 and w.Part1:IsDescendantOf(char) then
						table.insert(jointsList, {
							part0Name = w.Part0 and w.Part0.Name or "",
							part1Name = w.Part1.Name,
							c0 = (w:IsA("Motor6D") or w:IsA("Weld")) and w.C0 or nil,
							c1 = (w:IsA("Motor6D") or w:IsA("Weld")) and w.C1 or nil,
							className = w.ClassName,
						})
					end
				end

				local backupCopy = deepClone(newItem)
				for _, d in ipairs(backupCopy:GetDescendants()) do
					if d.Name:lower():find("tryon") then d:Destroy() end
				end
				backupCopy:SetAttribute("IsTryOn", nil)
				backupCopy:SetAttribute("TryOn", nil)

				savedWornLayers[itemName] = {
					backup = backupCopy,
					joints = jointsList,
					isAccessory = newItem:IsA("Accessory"),
					isModel = newItem:IsA("Model"),
					category = itemCat,
				}
			end)
		end

		restoreAllSavedLayers()
		unfreezeControls()
	end)
end

local function formatCommas(str)
	local numStr = str:gsub("[^%d]", "")
	if numStr == "" then return "" end
	local k
	while true do
		numStr, k = string.gsub(numStr, "^(-?%d+)(%d%d%d)", '%1,%2')
		if k == 0 then break end
	end
	return numStr
end
