local function disableFreezeOnlyConnections()
    if getconnections and previewItemRemote then
        pcall(function()
            for _, conn in ipairs(getconnections(previewItemRemote.OnClientEvent)) do
                pcall(function()
                    local fn = conn.Function
                    local isFreeze = false
                    if getconstants and fn then
                        for _, c in ipairs(getconstants(fn)) do
                            if c == "ExitTryOnNonWings" or c == "PrimaryPart" or c == "Anchored" then
                                isFreeze = true
                                break
                            end
                        end
                    end
                    if isFreeze then
                        conn:Disable()
                    end
                end)
            end
        end)
    end
end

local function suppressGuiObj(desc)
    if not desc then return end
    if desc:IsA("BillboardGui") or desc:IsA("SurfaceGui") or desc:IsA("ScreenGui") then
        desc.Enabled = false
        if not desc:GetAttribute("RH_Suppressed") then
            desc:SetAttribute("RH_Suppressed", true)
            pcall(function()
                desc:GetPropertyChangedSignal("Enabled"):Connect(function()
                    if desc.Enabled then desc.Enabled = false end
                end)
            end)
        end
    elseif desc:IsA("GuiObject") then
        desc.Visible = false
        if not desc:GetAttribute("RH_Suppressed") then
            desc:SetAttribute("RH_Suppressed", true)
            pcall(function()
                desc:GetPropertyChangedSignal("Visible"):Connect(function()
                    if desc.Visible then desc.Visible = false end
                end)
            end)
        end
    end
end

local function isOverheadTryOnBillboard(inst)
    if not inst then return false end
    local bb = (inst:IsA("BillboardGui") or inst:IsA("SurfaceGui")) and inst or inst:FindFirstAncestorWhichIsA("BillboardGui") or inst:FindFirstAncestorWhichIsA("SurfaceGui")
    if not bb then return false end
    local nl = inst.Name:lower()
    if nl:find("tryon") or nl:find("try_on") or nl:find("try on") then return true end
    if (inst:IsA("TextLabel") or inst:IsA("TextButton")) then
        local tl = (inst.Text or ""):lower()
        if tl:find("try on") or tl:find("trying on") or tl:find("tryon") then return true end
    end
    return false
end

local function hideTryOnOverlays()
    pcall(function()
        -- Only hide ExitTryOn buttons in HUD/Shop (never touch Shop's "Try On" buttons!)
        local hud = pg:FindFirstChild("HUD")
        if hud then
            for _, ch in ipairs(hud:GetDescendants()) do
                if ch:IsA("GuiObject") then
                    local nl = ch.Name:lower()
                    local isTryBanner = nl:find("exittryon") or nl:find("tryonmode")
                    if not isTryBanner and (ch:IsA("TextLabel") or ch:IsA("TextButton")) then
                        local tl = (ch.Text or ""):lower()
                        if tl:find("try on mode") or tl:find("exit try on") then
                            isTryBanner = true
                        end
                    end
                    if isTryBanner then
                        suppressGuiObj(ch)
                    end
                end
            end
        end
        local shop = pg:FindFirstChild("Shop")
        if shop then
            local ex = shop:FindFirstChild("ExitTryOnNonWings", true)
            if ex and ex:IsA("GuiObject") then suppressGuiObj(ex) end
        end
        -- Hide overhead TryOn BillboardGuis on Character (Head/Torso)
        local char = lp.Character
        if char then
            for _, d in ipairs(char:GetDescendants()) do
                if isOverheadTryOnBillboard(d) then
                    local bb = (d:IsA("BillboardGui") or d:IsA("SurfaceGui")) and d or d:FindFirstAncestorWhichIsA("BillboardGui") or d:FindFirstAncestorWhichIsA("SurfaceGui")
                    if bb then suppressGuiObj(bb) end
                end
            end
        end
        for _, gui in ipairs(pg:GetChildren()) do
            if gui.Name ~= "RH_Wardrobe_v49" and gui.Name ~= "Shop" then
                for _, bb in ipairs(gui:GetDescendants()) do
                    if bb:IsA("BillboardGui") then
                        for _, d in ipairs(bb:GetDescendants()) do
                            if isOverheadTryOnBillboard(d) then suppressGuiObj(bb) break end
                        end
                    end
                end
            end
        end
    end)
end

local function releasePlayerMovement()
    local char = lp.Character
    if not char then return end

    -- 1. Unanchor all character parts and primary part
    local pp = char.PrimaryPart or char:FindFirstChild("HumanoidRootPart")
    if pp and pp.Anchored then pp.Anchored = false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if hrp and hrp.Anchored then hrp.Anchored = false end

    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Anchored then
            part.Anchored = false
        end
    end

    -- 2. Humanoid walkspeed, jumping, sitting, platformstand
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        if hum.PlatformStand then hum.PlatformStand = false end
        if hum.Sit then hum.Sit = false end
        if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
        if hum.JumpPower and hum.JumpPower < 50 then hum.JumpPower = 50 end
        if hum.JumpHeight and hum.JumpHeight < 7.2 then hum.JumpHeight = 7.2 end

        -- Stop only forced freeze animations (preserve item idle/hold/pose animations!)
        for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
            local tn = track.Animation and track.Animation.Name:lower() or ""
            if tn:find("freeze") then
                pcall(function() track:Stop(0.05) end)
            end
        end
    end

    -- 3. Unbind all 10 known ContextActionService freeze actions in Royale High
    local freezeActions = {
        "ControlsDisabled", "DisableControlsCutscene", "IntroScriptDisableControls",
        "RH3CutsceneControlsDisabled", "FreezeMovement", "ShopFreeze", "TryOnFreeze",
        "DisableMovement", "CharacterCreationFreeze", "TryOnControl"
    }
    for _, act in ipairs(freezeActions) do
        pcall(function() ContextActionService:UnbindAction(act) end)
    end

    -- 4. Enable Roblox PlayerModule controls
    pcall(function()
        local pm = lp.PlayerScripts:FindFirstChild("PlayerModule")
        if pm then
            local mod = require(pm)
            if mod and mod.GetControls then
                local ctrl = mod:GetControls()
                if ctrl and ctrl.Enable then
                    ctrl:Enable()
                end
            end
        end
    end)

    -- 5. Only trigger ExitPreview.OnClientEvent connections that handle anchoring (NEVER close PreviewTogglesFrame or ColorsGui!)
    if exitPreviewRemote and getconnections then
        pcall(function()
            for _, conn in ipairs(getconnections(exitPreviewRemote.OnClientEvent)) do
                local fn = conn.Function
                local isAnchorReset = false
                if getconstants and fn then
                    for _, c in ipairs(getconstants(fn)) do
                        if c == "PrimaryPart" or c == "Anchored" or c == "ExitTryOnNonWings" then
                            isAnchorReset = true
                            break
                        end
                    end
                end
                if isAnchorReset then
                    pcall(function() conn:Fire() end)
                end
            end
        end)
    end

    -- 6. Hide all TryOn billboard HUD elements
    pcall(hideTryOnOverlays)
end

local activeCustomFlyTrack = nil
local activeCustomFlyAnimId = nil

local function stopCustomFlyTrack()
    if activeCustomFlyTrack then
        pcall(function() activeCustomFlyTrack:Stop(0.15) end)
        activeCustomFlyTrack = nil
        activeCustomFlyAnimId = nil
    end
end

local function updateCustomFlyAnimations()
    local char = lp.Character
    if not char then stopCustomFlyTrack() return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
    if not hum or not hrp or not bakedFolder then
        stopCustomFlyTrack()
        return
    end

    -- Detect Royale High flight state (PlatformStand / Physics state / flight BodyMovers)
    local isFlying = hum.PlatformStand
                  or hum:GetState() == Enum.HumanoidStateType.Physics
                  or hrp:FindFirstChildWhichIsA("BodyVelocity") ~= nil
                  or hrp:FindFirstChildWhichIsA("LinearVelocity") ~= nil
                  or hrp:FindFirstChildWhichIsA("BodyPosition") ~= nil
                  or hrp:FindFirstChildWhichIsA("AlignPosition") ~= nil

    if not isFlying then
        stopCustomFlyTrack()
        return
    end

    local targetAnim = nil
    for _, baked in ipairs(bakedFolder:GetChildren()) do
        local cf = baked:FindFirstChild("CustomFly", true)
        if cf then
            local animFolder = cf:FindFirstChild("Animations") or cf
            for _, a in ipairs(animFolder:GetChildren()) do
                if a:IsA("Animation") and a.AnimationId ~= "" then
                    targetAnim = a
                    break
                end
            end
            if targetAnim then break end
        end
    end

    if not targetAnim then
        stopCustomFlyTrack()
        return
    end

    if activeCustomFlyAnimId ~= targetAnim.AnimationId or (activeCustomFlyTrack and not activeCustomFlyTrack.IsPlaying) then
        stopCustomFlyTrack()
        local animator = hum:FindFirstChildOfClass("Animator") or hum
        local ok, tr = pcall(function() return animator:LoadAnimation(targetAnim) end)
        if ok and tr then
            tr.Looped = true
            tr.Priority = Enum.AnimationPriority.Action
            tr:Play(0.15)
            activeCustomFlyTrack = tr
            activeCustomFlyAnimId = targetAnim.AnimationId
        end
    end
end

local hiddenBodyPartsBySlot = {}

enforceHiddenBodyParts = function()
    local char = lp.Character
    if not char then return end
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
    if not bakedFolder then return end

    for slotKey, partsMap in pairs(hiddenBodyPartsBySlot) do
        if bakedFolder:FindFirstChild(slotKey) then
            for partName, _ in pairs(partsMap) do
                if partName ~= "HumanoidRootPart" and partName ~= "Head" then
                    local limb = char:FindFirstChild(partName)
                    if limb and limb:IsA("BasePart") and limb.Transparency < 0.99 then
                        limb.Transparency = 1
                    end
                end
            end
        end
    end
end

local function restoreUnhiddenBodyParts(removedPartsMap)
    local char = lp.Character
    if not char or not removedPartsMap then return end
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")

    for partName, _ in pairs(removedPartsMap) do
        if partName ~= "HumanoidRootPart" and partName ~= "Head" then
            local stillHidden = false
            if bakedFolder then
                for slotKey, partsMap in pairs(hiddenBodyPartsBySlot) do
                    if bakedFolder:FindFirstChild(slotKey) and partsMap[partName] then
                        stillHidden = true
                        break
                    end
                end
            end
            if not stillHidden then
                local limb = char:FindFirstChild(partName)
                if limb and limb:IsA("BasePart") then
                    limb.Transparency = 0
                end
            end
        end
    end
end

-- Strictly identify base avatar body parts and hair to prevent hair duplication
local function isBaseAvatarObject(inst)
    if not inst then return true end
    if defaultAvatarInstances[inst] then return true end
    if inst:GetAttribute("RH_BakedLayer") then return true end
    if inst.Name == "Humanoid" or inst.Name == "Animate" or inst.Name == "Head" then return true end
    if standardBodyParts[inst.Name] then return true end

    -- Check if it's avatar hair (Accessory, Model, Folder, or BasePart with HairAttachment or hair keywords)
    local char = lp.Character
    local inEquippedStorage = workspace:FindFirstChild("EquippedStorage") and inst:IsDescendantOf(workspace.EquippedStorage)
    if not inEquippedStorage then
        if inst:IsA("Accessory") or (char and inst.Parent == char and (inst:IsA("Model") or inst:IsA("Folder"))) then
            if inst:FindFirstChild("HairAttachment", true) or inst:FindFirstChild("Hair", true) then
                return true
            end
            local ln = inst.Name:lower()
            if ln:find("hair") or ln:find("wig") or ln:find("bangs") or ln:find("curl")
               or ln:find("ponytail") or ln:find("pigtail") or ln:find("afro") or ln:find("braid")
               or ln:find("locs") or ln:find("bun") or ln:find("fade") or ln:find("extension") then
                return true
            end
        end
    end

    if inst.Name == "Handle" and inst.Parent and inst.Parent:IsA("Accessory") then
        if inst:FindFirstChild("HairAttachment") then return true end
    end

    return false
end

-- Find all actual Royale High clothing pieces attached to avatar
local function findRoyaleHighItemsOnAvatar()
    local char = lp.Character
    if not char then return {} end
    local itemsFound = {}
    local seen = {}

    local function addItem(inst)
        if not inst or seen[inst] then return end
        if isBaseAvatarObject(inst) then return end
        seen[inst] = true
        table.insert(itemsFound, inst)
    end

    -- 1. Scan EquippedStorage & WornItems folders
    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            addItem(child)
        end
    end

    -- 2. Scan Character hierarchy for custom models/folders/items
    for _, d in ipairs(char:GetChildren()) do
        if not isBaseAvatarObject(d) and d.Name ~= "RH_BakedWardrobe" then
            if d:IsA("Model") or d:IsA("Folder") or d:IsA("Accessory") or d:IsA("MeshPart") or d:IsA("Tool") then
                addItem(d)
            end
        end
    end

    -- 3. Scan welds connected to character rig limbs
    local charLimbs = {}
    for _, limb in ipairs(char:GetChildren()) do
        if limb:IsA("BasePart") and standardBodyParts[limb.Name] then
            charLimbs[limb] = true
        end
    end

    local searchRoots = { char }
    local es = workspace:FindFirstChild("EquippedStorage")
    if es then table.insert(searchRoots, es) end

    for _, root in ipairs(searchRoots) do
        for _, w in ipairs(root:GetDescendants()) do
            if (w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D")) then
                local otherPart = nil
                if charLimbs[w.Part0] and w.Part1 and not charLimbs[w.Part1] then
                    otherPart = w.Part1
                elseif charLimbs[w.Part1] and w.Part0 and not charLimbs[w.Part0] then
                    otherPart = w.Part0
                end

                if otherPart and not isBaseAvatarObject(otherPart) then
                    local container = otherPart
                    while container.Parent
                          and container.Parent ~= char
                          and container.Parent ~= workspace
                          and container.Parent ~= es
                          and not container.Parent:IsA("Workspace") do
                        if container.Parent.Name == lp.Name or container.Parent.Name == tostring(lp.UserId) then
                            break
                        end
                        container = container.Parent
                    end
                    addItem(container)
                end
            end
        end
    end

    return itemsFound
end

local function countBaseParts(inst)
    if not inst then return 0 end
    if inst:IsA("BasePart") then return 1 end
    local n = 0
    for _, d in ipairs(inst:GetDescendants()) do
        if d:IsA("BasePart") then
            n = n + 1
        end
    end
    return n
end


-- Find a specific newly spawned Royale High item on the avatar or in EquippedStorage (must have >=1 BasePart!)
local function findSpecificItemOnAvatar(itemName, cat, beforeSet)
    local char = lp.Character
    if not char then return nil, nil end
    local cleanTarget = (itemName or ""):lower():gsub("[^%w]", "")

    local function matchesName(inst)
        if not inst then return false end
        local n = inst.Name:lower():gsub("[^%w]", "")
        if cleanTarget ~= "" and (n == cleanTarget or n:find(cleanTarget, 1, true) or cleanTarget:find(n, 1, true)) then
            return true
        end
        for _, vName in ipairs({"HeelName", "ItemName", "AssetName", "SkirtName", "BodiceName"}) do
            local sv = inst:FindFirstChild(vName, true)
            if sv and sv:IsA("StringValue") then
                local vn = sv.Value:lower():gsub("[^%w]", "")
                if cleanTarget ~= "" and (vn == cleanTarget or vn:find(cleanTarget, 1, true) or cleanTarget:find(vn, 1, true)) then
                    return true
                end
            end
        end
        local attr = inst:GetAttribute("RH_ItemName")
        if type(attr) == "string" then
            local an = attr:lower():gsub("[^%w]", "")
            if cleanTarget ~= "" and (an == cleanTarget or an:find(cleanTarget, 1, true) or cleanTarget:find(an, 1, true)) then
                return true
            end
        end
        return false
    end

    -- 1. Check EquippedStorage & WornItems first
    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            if not isBaseAvatarObject(child) and countBaseParts(child) > 0 then
                if matchesName(child) then
                    return child, info.category
                end
                if beforeSet and not beforeSet[child] and (not cat or info.category == cat) then
                    return child, info.category
                end
            end
        end
    end

    -- 2. Check Character direct children (excluding RH_BakedWardrobe)
    for _, d in ipairs(char:GetChildren()) do
        if not isBaseAvatarObject(d) and d.Name ~= "RH_BakedWardrobe" then
            if (d:IsA("Model") or d:IsA("Folder") or d:IsA("Accessory") or d:IsA("MeshPart") or d:IsA("Tool")) and countBaseParts(d) > 0 then
                if matchesName(d) then
                    return d, cat
                end
                if beforeSet and not beforeSet[d] then
                    return d, cat
                end
            end
        end
    end

    -- 3. Universal Weld-to-Limb Scan across char & EquippedStorage (catches ANY folder name Royale High uses!)
    local charLimbs = {}
    for _, limb in ipairs(char:GetChildren()) do
        if limb:IsA("BasePart") and standardBodyParts[limb.Name] then
            charLimbs[limb] = true
        end
    end
    local es = workspace:FindFirstChild("EquippedStorage")
    local searchRoots = { char }
    if es then table.insert(searchRoots, es) end

    for _, root in ipairs(searchRoots) do
        for _, w in ipairs(root:GetDescendants()) do
            if w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D") then
                local otherPart = nil
                if charLimbs[w.Part0] and w.Part1 and not charLimbs[w.Part1] then
                    otherPart = w.Part1
                elseif charLimbs[w.Part1] and w.Part0 and not charLimbs[w.Part0] then
                    otherPart = w.Part0
                end
                if otherPart and not isBaseAvatarObject(otherPart) then
                    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
                    if not (bakedFolder and otherPart:IsDescendantOf(bakedFolder)) then
                        local container = otherPart
                        while container.Parent
                              and container.Parent ~= char
                              and container.Parent ~= workspace
                              and container.Parent ~= es
                              and not container.Parent:IsA("Workspace") do
                            if container.Parent.Name == lp.Name or container.Parent.Name == tostring(lp.UserId) then
                                break
                            end
                            container = container.Parent
                        end
                        if not isBaseAvatarObject(container) and container.Name ~= "RH_BakedWardrobe" and countBaseParts(container) > 0 then
                            if matchesName(container) or (beforeSet and not beforeSet[container]) then
                                return container, cat
                            end
                        end
                    end
                end
            end
        end
    end

    return nil, nil
end

-- 1:1 Single-Clone Layer Baker Engine: Clones exactly 1 instance per slot, matching relative CFrames & welds 1:1
local function bakeItemToAvatar(sourceItem, category, customSlotKey, itemData)
    if not sourceItem then return nil end
    local char = lp.Character
    if not char then return nil end
    if countBaseParts(sourceItem) == 0 then return nil end

    local cat = category or "Accessories"
    local slotKey = nil
    if customSlotKey then
        slotKey = customSlotKey
    elseif cat == "Heels" or cat == "Skirts" or cat == "Bodices" or cat == "Wings" or cat == "Halos" then
        slotKey = cat
    else
        local accName = (itemData and itemData.name) or sourceItem.Name
        if accName == "Accessory" or accName == "Model" or accName == "Folder" or accName == "Part" then
            local sub = sourceItem:FindFirstChildWhichIsA("SpecialMesh", true) or sourceItem:FindFirstChildWhichIsA("MeshPart", true)
            if sub and sub.Name ~= "MeshPart" and sub.Name ~= "Mesh" then
                accName = sub.Name
            end
        end
        slotKey = "Acc_" .. accName:gsub("%s+", "")
    end

    -- Inspect sourceItem's existing joints BEFORE cloning to find the exact character limb it was attached to (e.g. RightHand for a sword)
    local detectedLimb = nil
    local charLimbs = {}
    for _, limb in ipairs(char:GetChildren()) do
        if limb:IsA("BasePart") and standardBodyParts[limb.Name] then
            charLimbs[limb] = true
        end
    end

    local function scanForCharLimb(rootInst)
        for _, j in ipairs(rootInst:GetDescendants()) do
            if j:IsA("Weld") or j:IsA("WeldConstraint") or j:IsA("Motor6D") then
                if j.Part0 and charLimbs[j.Part0] then return j.Part0 end
                if j.Part1 and charLimbs[j.Part1] then return j.Part1 end
            end
        end
        return nil
    end

    detectedLimb = scanForCharLimb(sourceItem)
    if not detectedLimb then
        -- Also check joints stored under char that point into sourceItem
        for _, j in ipairs(char:GetDescendants()) do
            if j:IsA("Weld") or j:IsA("WeldConstraint") or j:IsA("Motor6D") then
                if j.Part0 and j.Part1 then
                    if charLimbs[j.Part0] and (j.Part1 == sourceItem or j.Part1:IsDescendantOf(sourceItem)) then
                        detectedLimb = j.Part0
                        break
                    elseif charLimbs[j.Part1] and (j.Part0 == sourceItem or j.Part0:IsDescendantOf(sourceItem)) then
                        detectedLimb = j.Part1
                        break
                    end
                end
            end
        end
    end

    -- Dedicated container under Character
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
    if not bakedFolder then
        bakedFolder = Instance.new("Folder")
        bakedFolder.Name = "RH_BakedWardrobe"
        bakedFolder.Parent = char
    end

    -- Strict single-clone per slot: destroy old clone for THIS specific slot/accessory
    local existing = bakedFolder:FindFirstChild(slotKey)
    if existing then
        pcall(function() existing:Destroy() end)
    end
    bakedLayersBackup[slotKey] = nil

    -- Deep clone source item
    local ok, clone = pcall(function()
        sourceItem.Archivable = true
        for _, d in ipairs(sourceItem:GetDescendants()) do d.Archivable = true end
        return sourceItem:Clone()
    end)
    if not ok or not clone then return nil end

    -- If sourceItem was a Tool, convert to Model so it wears passively without equipping as a tool
    if clone:IsA("Tool") then
        local m = Instance.new("Model")
        m.Name = clone.Name
        for _, ch in ipairs(clone:GetChildren()) do
            if not ch:IsA("Script") and not ch:IsA("LocalScript") then
                ch.Parent = m
            end
        end
        clone:Destroy()
        clone = m
    end

    local imgToStore = (itemData and itemData.img and itemData.img ~= "") and itemData.img or ""
    if imgToStore == "" and findItemImage then
        imgToStore = findItemImage(itemData and itemData.name or sourceItem.Name, cat)
    end

    clone.Name = slotKey
    clone:SetAttribute("RH_BakedLayer", true)
    clone:SetAttribute("RH_Category", cat)
    clone:SetAttribute("RH_SlotKey", slotKey)
    clone:SetAttribute("RH_ItemName", (itemData and itemData.name) or sourceItem.Name)
    clone:SetAttribute("RH_Image", imgToStore)
    clone:SetAttribute("RH_Price", (itemData and itemData.price) or "")

    -- Preserve internal Motor6Ds, Bones, AnimationControllers, & VFX for 1:1 animations;
    -- only remove external joints pointing outside the clone or BodyMovers
    for _, j in ipairs(clone:GetDescendants()) do
        if j:IsA("BodyMover") or j:IsA("Script") then
            pcall(function() j:Destroy() end)
        elseif j:IsA("JointInstance") or j:IsA("WeldConstraint") then
            local p0ok = j.Part0 and (j.Part0 == clone or j.Part0:IsDescendantOf(clone))
            local p1ok = j.Part1 and (j.Part1 == clone or j.Part1:IsDescendantOf(clone))
            if not (p0ok and p1ok) then
                pcall(function() j:Destroy() end)
            end
        end
    end

    local zeroPhys = PhysicalProperties.new(0.01, 0, 0, 0, 0)
    local function prepPart(p)
        p.CanCollide = false
        p.CanTouch   = false
        p.CanQuery   = false
        p.Anchored   = false
        p.Massless   = true
        pcall(function() p.CustomPhysicalProperties = zeroPhys end)
    end

    -- Parent clone into Workspace (bakedFolder) FIRST so all Welds activate immediately!
    clone.Parent = bakedFolder

    local function weldRigid(part0, part1, weldName)
        local w = Instance.new("Weld")
        w.Name = weldName or "RH_RigidWeld"
        w.Part0 = part0
        w.Part1 = part1
        w.C0 = part0.CFrame:ToObjectSpace(part1.CFrame)
        w.C1 = CFrame.new()
        w.Parent = part1
        return w
    end

    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("LowerTorso")

    if cat == "Heels" and hrp then
        local leftLimb = char:FindFirstChild("LeftLowerLeg") or char:FindFirstChild("LeftUpperLeg") or char:FindFirstChild("Left Leg") or hrp
        local rightLimb = char:FindFirstChild("RightLowerLeg") or char:FindFirstChild("RightUpperLeg") or char:FindFirstChild("Right Leg") or hrp

        local leftParts = {}
        local rightParts = {}

        if clone:IsA("BasePart") then
            prepPart(clone)
            if hrp.CFrame:PointToObjectSpace(clone.Position).X < 0 then table.insert(leftParts, clone) else table.insert(rightParts, clone) end
        else
            for _, p in ipairs(clone:GetDescendants()) do
                if p:IsA("BasePart") then
                    prepPart(p)
                    if hrp.CFrame:PointToObjectSpace(p.Position).X < 0 then table.insert(leftParts, p) else table.insert(rightParts, p) end
                end
            end
        end

        local function rigBoot(partsList, targetLimb)
            if #partsList == 0 or not targetLimb then return end
            local root = partsList[1]
            local maxVol = -1
            for _, p in ipairs(partsList) do
                local vol = p.Size.X * p.Size.Y * p.Size.Z
                if vol > maxVol then maxVol = vol; root = p end
            end

            weldRigid(targetLimb, root, "RH_BootRootWeld")
            for _, p in ipairs(partsList) do
                if p ~= root then
                    local hasInternalJoint = false
                    for _, d in ipairs(clone:GetDescendants()) do
                        if (d:IsA("Motor6D") or d:IsA("Weld") or d:IsA("WeldConstraint"))
                           and (d.Part0 == p or d.Part1 == p) then
                            hasInternalJoint = true
                            break
                        end
                    end
                    if not hasInternalJoint then
                        weldRigid(root, p, "RH_BootSubWeld")
                    end
                end
            end
        end

        rigBoot(leftParts, leftLimb)
        rigBoot(rightParts, rightLimb)
    else
        local targetLimb = detectedLimb
        if not targetLimb then
            if cat == "Skirts" then
                targetLimb = char:FindFirstChild("LowerTorso") or char:FindFirstChild("Torso") or hrp
            elseif cat == "Bodices" or cat == "Wings" then
                targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or hrp
            elseif cat == "Halos" then
                targetLimb = char:FindFirstChild("Head") or hrp
            else
                -- Accessories: check attachments, then nearest character limb
                local att = clone:FindFirstChildWhichIsA("Attachment", true)
                if att then
                    local an = att.Name:lower()
                    if an:find("hat") or an:find("face") or an:find("hair") or an:find("head") then
                        targetLimb = char:FindFirstChild("Head")
                    elseif an:find("waist") then
                        targetLimb = char:FindFirstChild("LowerTorso") or char:FindFirstChild("Torso")
                    elseif an:find("rightgrip") or an:find("righthand") then
                        targetLimb = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm")
                    elseif an:find("leftgrip") or an:find("lefthand") then
                        targetLimb = char:FindFirstChild("LeftHand") or char:FindFirstChild("Left Arm")
                    elseif an:find("rightshoulder") then
                        targetLimb = char:FindFirstChild("RightUpperArm") or char:FindFirstChild("Right Arm")
                    elseif an:find("leftshoulder") then
                        targetLimb = char:FindFirstChild("LeftUpperArm") or char:FindFirstChild("Left Arm")
                    elseif an:find("body") or an:find("back") or an:find("neck") or an:find("collar") then
                        targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
                    end
                end
            end
        end

        local allParts = {}
        if clone:IsA("BasePart") then
            prepPart(clone)
            table.insert(allParts, clone)
        else
            for _, p in ipairs(clone:GetDescendants()) do
                if p:IsA("BasePart") then
                    prepPart(p)
                    table.insert(allParts, p)
                end
            end
        end

        if #allParts > 0 then
            -- Safe PrimaryPart check (never access .PrimaryPart on Folder/Accessory!)
            local root = (clone:IsA("Model") and clone.PrimaryPart)
                      or clone:FindFirstChild("Middle")
                      or clone:FindFirstChild("Center")
                      or clone:FindFirstChild("Handle")
                      or allParts[1]

            -- If targetLimb still unknown, pick nearest character limb to root
            if not targetLimb then
                local bestDist = math.huge
                for limb, _ in pairs(charLimbs) do
                    if limb.Name ~= "HumanoidRootPart" then
                        local dist = (limb.Position - root.Position).Magnitude
                        if dist < bestDist then
                            bestDist = dist
                            targetLimb = limb
                        end
                    end
                end
                targetLimb = targetLimb or char:FindFirstChild("UpperTorso") or hrp
            end

            if targetLimb then
                weldRigid(targetLimb, root, "RH_UnifiedRootWeld")
                for _, p in ipairs(allParts) do
                    if p ~= root then
                        -- Keep animated Motor6D / internal joint chains moving 1:1 instead of overriding them with a static Weld
                        local hasInternalJoint = false
                        for _, d in ipairs(clone:GetDescendants()) do
                            if (d:IsA("Motor6D") or d:IsA("Weld") or d:IsA("WeldConstraint"))
                               and (d.Part0 == p or d.Part1 == p) then
                                hasInternalJoint = true
                                break
                            end
                        end
                        if not hasInternalJoint then
                            weldRigid(root, p, "RH_UnifiedSubWeld")
                        end
                    end
                end
            end
        end
    end

    -- Abort if clone ended up with 0 visible BaseParts so we never leave an invisible shell that hides the torso
    if countBaseParts(clone) == 0 then
        pcall(function() clone:Destroy() end)
        return nil
    end

    -- Snapshot & lock any base avatar body parts hidden by this morph (corset torso, heel legs, sleeve arms, etc.)
    local hiddenMap = {}
    for _, limb in ipairs(char:GetChildren()) do
        if limb:IsA("BasePart") and standardBodyParts[limb.Name] and limb.Name ~= "HumanoidRootPart" and limb.Name ~= "Head" then
            if limb.Transparency >= 0.95 then
                hiddenMap[limb.Name] = true
            end
        end
    end
    -- Detect any custom body-limb replacements inside sourceItem (custom arms, custom torso, custom legs)
    local armSleeveNames = {
        "LeftUpperArm", "LeftLowerArm", "Left Arm",
        "RightUpperArm", "RightLowerArm", "Right Arm"
    }
    local handPartNames = { "LeftHand", "RightHand" }
    local torsoPartNames = { "UpperTorso", "LowerTorso", "Torso" }

    if cat == "Heels" then
        for _, lName in ipairs(legPartNames) do hiddenMap[lName] = true end
    end
    for _, tName in ipairs(torsoPartNames) do
        if sourceItem:FindFirstChild(tName, true) or sourceItem:FindFirstChild("HideTorso", true) then
            for _, x in ipairs(torsoPartNames) do hiddenMap[x] = true end
            break
        end
    end
    -- Only hide individual arm segments (never hide LeftHand / RightHand unless the item actually has custom hands!)
    for _, aName in ipairs(armSleeveNames) do
        if sourceItem:FindFirstChild(aName, true) or sourceItem:FindFirstChild("HideArms", true) then
            hiddenMap[aName] = true
        end
    end
    for _, hName in ipairs(handPartNames) do
        if sourceItem:FindFirstChild("HideHands", true) then
            hiddenMap[hName] = true
        end
    end
    for _, lName in ipairs(legPartNames) do
        if sourceItem:FindFirstChild(lName, true) or sourceItem:FindFirstChild("HideLegs", true) then
            hiddenMap[lName] = true
        end
    end
    if next(hiddenMap) ~= nil then
        hiddenBodyPartsBySlot[slotKey] = hiddenMap
        enforceHiddenBodyParts()
    end

    -- Replay any internal item animations (e.g., animated halos, wings, swaying ribbons, pets) 1:1
    task.spawn(function()
        pcall(function()
            local animCtrl = clone:FindFirstChildWhichIsA("AnimationController", true)
            local animator = animCtrl and (animCtrl:FindFirstChildWhichIsA("Animator") or Instance.new("Animator", animCtrl))
            if animator then
                for _, anim in ipairs(clone:GetDescendants()) do
                    if anim:IsA("Animation") and anim.AnimationId ~= "" then
                        pcall(function()
                            local track = animator:LoadAnimation(anim)
                            track.Looped = true
                            track:Play()
                        end)
                    end
                end
            end
        end)
    end)

    bakedLayersBackup[slotKey] = deepClone(clone)
    return clone
end

-- Lock avatar's current look into single-layer baked slots
local isBakingDebounce = false
local function lockCurrentAvatarLook(catHint, itemData)
    if isBakingDebounce then return 0 end

    local items = findRoyaleHighItemsOnAvatar()
    if #items == 0 then return 0 end

    isBakingDebounce = true
    local count = 0
    for _, item in ipairs(items) do
        local cat = catHint or "Accessories"
        if item.Parent and item.Parent.Parent and item.Parent.Parent.Name == "EquippedStorage" then
            cat = item.Parent.Name
        end
        local b = bakeItemToAvatar(item, cat, nil, itemData)
        if b then count = count + 1 end
    end

    task.delay(0.3, function()
        isBakingDebounce = false
    end)
    return count
end

-- Unequip a single worn slot
local function unequipWornItem(item)
    if not item then return end
    local char = lp.Character
    local itemName = item.name or ""
    local slotKey  = item.slotKey or ""
    local cat      = item.cat or ""

    -- 1. Remove ONLY this specific item from RH_BakedWardrobe
    if item.bakedInst and item.bakedInst.Parent then
        pcall(function() item.bakedInst:Destroy() end)
    end
    if char and slotKey ~= "" then
        local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
        if bakedFolder then
            local b = bakedFolder:FindFirstChild(slotKey)
            if b then pcall(function() b:Destroy() end) end
        end
    end

    -- 2. Clear ONLY this slot's backup table entry
    if slotKey ~= "" and bakedLayersBackup[slotKey] then
        bakedLayersBackup[slotKey] = nil
    elseif cat ~= "" and (cat == "Heels" or cat == "Skirts" or cat == "Bodices" or cat == "Wings" or cat == "Halos") then
        bakedLayersBackup[cat] = nil
    end

    -- 3. Destroy live try-on instance if this item was a live try-on
    local hadLive = (item.liveInst ~= nil)
    if item.liveInst and item.liveInst.Parent then
        pcall(function() item.liveInst:Destroy() end)
    end

    -- 4. Purge ONLY matching item from EquippedStorage / WornItems (guard against empty itemName!)
    if itemName ~= "" and #itemName >= 3 then
        local lowName = itemName:lower()
        for _, info in ipairs(getEquippedFolders()) do
            for _, child in ipairs(info.folder:GetChildren()) do
                local cn = child.Name:lower()
                if cn == lowName or (cn ~= "" and #cn >= 3 and (cn:find(lowName, 1, true) or lowName:find(cn, 1, true))) then
                    pcall(function() child:Destroy() end)
                end
            end
        end
    end

    -- 5. Purge ONLY matching slot/item directly on Character (never touch other slots!)
    if char and (slotKey ~= "" or itemName ~= "") then
        for _, child in ipairs(char:GetChildren()) do
            if child.Name ~= "RH_BakedWardrobe" then
                if (slotKey ~= "" and child.Name == slotKey)
                   or (itemName ~= "" and child.Name == itemName)
                   or (slotKey ~= "" and child:GetAttribute("RH_SlotKey") == slotKey) then
                    pcall(function() child:Destroy() end)
                end
            end
        end
    end

    -- 6. Restore any base body parts hidden ONLY by this slot (if no other active slot still hides them)
    local removedMap = (slotKey ~= "" and hiddenBodyPartsBySlot[slotKey]) or nil
    if slotKey ~= "" then
        hiddenBodyPartsBySlot[slotKey] = nil
    end
    if cat == "Heels" and (slotKey == "Heels" or slotKey == "") then
        removedMap = removedMap or {}
        for _, lName in ipairs(legPartNames) do
            removedMap[lName] = true
        end
        hiddenBodyPartsBySlot["Heels"] = nil
    elseif cat == "Bodices" and (slotKey == "Bodices" or slotKey == "") then
        hiddenBodyPartsBySlot["Bodices"] = nil
    end
    if removedMap then
        restoreUnhiddenBodyParts(removedMap)
    end

    -- 7. Destroy custom wings only if unequipping Wings
    if cat == "Wings" and char then
        for _, n in ipairs({"Wings_TRYON", "CustomWingRight", "CustomWingLeft"}) do
            local w = char:FindFirstChild(n)
            if w then pcall(function() w:Destroy() end) end
        end
    end

    -- 8. Only fire ExitPreview if we were removing a live server try-on item (not when unequipping a single baked clone)
    if hadLive and exitPreviewRemote then
        pcall(function() exitPreviewRemote:FireServer() end)
    end

    -- 9. Release movement and refresh UI
    releasePlayerMovement()
    updateTabCounts()
    if refreshGridRef then
        refreshGridRef("Worn")
    end
end

-- Clear all baked layers and temporary try-ons, resetting character to normal
local function clearAllWornLayers()
    local char = lp.Character

    for _, info in ipairs(getEquippedFolders()) do
        for _, c in ipairs(info.folder:GetChildren()) do
            pcall(function() c:Destroy() end)
        end
    end

    if char then
        local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
        if bakedFolder then
            pcall(function() bakedFolder:Destroy() end)
        end

        for _, c in ipairs(char:GetChildren()) do
            if c:GetAttribute("RH_BakedLayer")
               or c:GetAttribute("CustomSavedLayer")
               or c:GetAttribute("RH_ClientLayer")
               or c:GetAttribute("RH_SlotKey")
               or c.Name:find("RH_Baked")
               or c.Name == "Wings_TRYON"
               or c.Name:find("CustomWing") then
                pcall(function() c:Destroy() end)
            end
        end

        for partName, _ in pairs(standardBodyParts) do
            if partName ~= "HumanoidRootPart" then
                local limb = char:FindFirstChild(partName)
                if limb and limb:IsA("BasePart") then
                    limb.Transparency = 0
                end
            end
        end
    end

    hiddenBodyPartsBySlot = {}
    bakedLayersBackup = {}
    if exitPreviewRemote then
        pcall(function() exitPreviewRemote:FireServer() end)
    end
end

-- Get list of all currently worn/baked items for Worn section
local function getWornItemsList()
    local wornList = {}
    local seen = {}
    local char = lp.Character
    if not char then return wornList end

    -- 1. Scan RH_BakedWardrobe (our primary baked layers)
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
    if bakedFolder then
        for _, clone in ipairs(bakedFolder:GetChildren()) do
            local itemName = clone:GetAttribute("RH_ItemName") or clone.Name:gsub("^RH_Baked_", "")
            local cat = clone:GetAttribute("RH_Category") or "Accessories"
            local slotKey = clone:GetAttribute("RH_SlotKey") or clone.Name
            local img = clone:GetAttribute("RH_Image") or ""
            local price = clone:GetAttribute("RH_Price") or ""

            if (not img or img == "") and findItemImage then
                img = findItemImage(itemName, cat)
            end

            local cleanKey = (cat .. "_" .. itemName):lower():gsub("%s+", "")
            if not seen[cleanKey] and not seen[slotKey:lower()] then
                seen[cleanKey] = true
                seen[slotKey:lower()] = true
                table.insert(wornList, {
                    name = itemName,
                    cat = cat,
                    slotKey = slotKey,
                    img = img,
                    price = price ~= "" and price or "Equipped",
                    isWorn = true,
                    bakedInst = clone
                })
            end
        end
    end

    -- 2. Scan EquippedStorage & WornItems for any live items not yet baked
    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            if not isBaseAvatarObject(child) then
                local cleanKey = (info.category .. "_" .. child.Name):lower():gsub("%s+", "")
                if not seen[cleanKey] and not seen[child.Name:lower()] then
                    seen[cleanKey] = true
                    local img = findItemImage and findItemImage(child.Name, info.category) or ""
                    table.insert(wornList, {
                        name = child.Name,
                        cat = info.category,
                        slotKey = info.category .. "_" .. child.Name,
                        img = img,
                        price = "Live Try-On",
                        isWorn = true,
                        liveInst = child
                    })
                end

local function addOrUpdateCatalogItem(cat, alias, assetName, price, img, isServerData)
    local key = (assetName or alias):lower()
    local existing = catalogItemByKey[key]

    local pStr = "N/A"
    if type(price) == "number" then
        pStr = string.format("%d", price):reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
    elseif type(price) == "string" and price ~= "" then
        pStr = price
    end

    local imgStr = ""
    if type(img) == "string" and img ~= "" then
        if img:find("rbxassetid://") or img:find("http") then
            imgStr = img
        elseif tonumber(img) then
            imgStr = "rbxassetid://" .. img
        end
    elseif type(img) == "number" and img > 0 then
        imgStr = "rbxassetid://" .. tostring(img)
    end
    if imgStr == "" and rhImageLookup then
        local c1 = (alias or ""):lower():gsub("[^%w]", "")
        local c2 = (assetName or ""):lower():gsub("[^%w]", "")
        imgStr = rhImageLookup[c1] or rhImageLookup[c2] or ""
    end

    if existing then
        if isServerData and imgStr ~= "" then
            existing.img = imgStr
            existing.isServerImg = true
        elseif (not existing.img or existing.img == "") and imgStr ~= "" then
            existing.img = imgStr
        end
        if pStr ~= "N/A" and (existing.price == "N/A" or isServerData) then
            existing.price = pStr
        end
        return
    end

    local itemObj = {
        name        = alias or assetName,
        assetName   = assetName or alias,
        price       = pStr,
        img         = imgStr,
        cat         = cat,
        isServerImg = (isServerData and imgStr ~= "")
    }
    catalogItemByKey[key] = itemObj
    if alias then
        catalogItemByKey[alias:lower()] = itemObj
        catalogItemByKey[alias:lower():gsub("%s+", "")] = itemObj
    end
    if assetName then
        catalogItemByKey[assetName:lower()] = itemObj
        catalogItemByKey[assetName:lower():gsub("%s+", "")] = itemObj
    end
    if categories[cat] then
        table.insert(categories[cat], itemObj)
    end
end

    if not name or name == "" then return "" end
    local clean = name:lower():gsub("%s+", ""):gsub("[^%w]", "")

    -- 0. Direct lookup from extracted 1,346-item Royale High thumbnail map
    if rhImageLookup[clean] then
        return rhImageLookup[clean]
    end

    -- 1. Direct key
    local direct = catalogItemByKey[clean] or catalogItemByKey[name:lower()]
    if direct and direct.img and direct.img ~= "" then
        return direct.img
    end

    -- 2. Scan category list if provided
    local targetList = cat and categories[cat]
    if targetList then
        for _, it in ipairs(targetList) do
            local itClean = it.name:lower():gsub("%s+", ""):gsub("[^%w]", "")
            local itAssetClean = (it.assetName or ""):lower():gsub("%s+", ""):gsub("[^%w]", "")
            if itClean == clean or itAssetClean == clean or clean:find(itClean, 1, true) or itClean:find(clean, 1, true) then
                if it.img and it.img ~= "" then return it.img end
            end
        end
    end

    -- 3. Scan all catalog items
    for _, it in pairs(catalogItemByKey) do
        local itClean = it.name:lower():gsub("%s+", ""):gsub("[^%w]", "")
        local itAssetClean = (it.assetName or ""):lower():gsub("%s+", ""):gsub("[^%w]", "")
        if itClean == clean or itAssetClean == clean or clean:find(itClean, 1, true) or itClean:find(clean, 1, true) then
            if it.img and it.img ~= "" then return it.img end
        end
    end

    -- 4. Substring fallback on rhImageLookup
    for k, img in pairs(rhImageLookup) do
        if #k >= 6 and (clean:find(k, 1, true) or k:find(clean, 1, true)) then
            return img
        end
    end

    return ""
end

-- Live Server Synchronizer
local function syncWithLiveServer()
    local net = ReplicatedStorage:FindFirstChild("Network")
    local fn  = net and net:FindFirstChild("Functions") and net.Functions:FindFirstChild("Gui")
    if not fn then return end

    -- 1. Heels
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("HeelMorphs") and fn.HeelMorphs:FindFirstChild("List") then
                local heels = fn.HeelMorphs.List:InvokeServer()
                if type(heels) == "table" then
                    for assetName, data in pairs(heels) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Heels", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Heels") end
                end
            end
        end)
    end)

    -- 2. Skirts
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Skirts") and fn.Skirts:FindFirstChild("List") then
                local skirts = fn.Skirts.List:InvokeServer()
                if type(skirts) == "table" then
                    for assetName, data in pairs(skirts) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Skirts", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Skirts") end
                end
            end
        end)
    end)

    -- 3. Accessories, Bodices, & Halos
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Accessories") and fn.Accessories:FindFirstChild("List") then
                local accs = fn.Accessories.List:InvokeServer()
                if type(accs) == "table" then
                    for assetName, data in pairs(accs) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        local sub = (data.Type or data.Category or ""):lower()
                        local aliasL = alias:lower()
                        local assetL = assetName:lower()

                        if aliasL:find("halo") or assetL:find("halo") then
                            addOrUpdateCatalogItem("Halos", alias, assetName, data.Price or "Special", img, true)
                        elseif aliasL:find("bodice") or aliasL:find("corset") or aliasL:find("jacket") or sub:find("bodice") or sub:find("torso") then
                            addOrUpdateCatalogItem("Bodices", alias, assetName, data.Price, img, true)
                        else
                            addOrUpdateCatalogItem("Accessories", alias, assetName, data.Price, img, true)
                        end
                    end
                    updateTabCounts()
                    if refreshGridRef then
                        refreshGridRef("Accessories")
                        refreshGridRef("Bodices")
                        refreshGridRef("Halos")
                    end
                end
            end
        end)
    end)

    -- 4. Wings
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Wings") and fn.Wings:FindFirstChild("List") then
                local wings = fn.Wings.List:InvokeServer()
                if type(wings) == "table" then
                    for assetName, data in pairs(wings) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Wings", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Wings") end
                end
            end
        end)
    end)
end

local function findShopTryonButton(itemName)
    local shop = pg:FindFirstChild("Shop")
    if not shop then return nil end
    local dir = shop:FindFirstChild("Directory")
    if not dir then return nil end

    local targetLower = itemName:lower()
    for _, folder in ipairs(dir:GetChildren()) do
        local mid = folder:FindFirstChild("MiddlePanel")
        local sf = mid and mid:FindFirstChild("ScrollingFrame")
        if sf then
            for _, itemFrame in ipairs(sf:GetChildren()) do
                if itemFrame:IsA("Frame") then
                    local tryon = itemFrame:FindFirstChild("Tryon")
                    if tryon and tryon.Visible then
                        local fName = itemFrame.Name:lower()
                        local nLabel = itemFrame:FindFirstChild("name")
                        local lText = (nLabel and nLabel:IsA("TextLabel")) and nLabel.Text:lower() or ""
                        if fName == targetLower or lText == targetLower
                           or fName:find(targetLower, 1, true) or targetLower:find(fName, 1, true)
                           or (lText ~= "" and lText:find(targetLower, 1, true)) then
                            return tryon
                        end
                    end
                end
            end
        end
    end
    return nil
end

local function clickButtonVIM(btn)
    if not btn then return false end
    local ok = pcall(function()
        local pos = btn.AbsolutePosition
        local sz  = btn.AbsoluteSize
        local cx  = pos.X + sz.X / 2
        local cy  = pos.Y + sz.Y / 2
        if VIM then
            VIM:SendMouseButtonEvent(cx, cy, 0, true, game, 1)
            task.wait(0.04)
            VIM:SendMouseButtonEvent(cx, cy, 0, false, game, 1)
        elseif VU then
            VU:Button1Down(Vector2.new(cx, cy), workspace.CurrentCamera)
            task.wait(0.04)
            VU:Button1Up(Vector2.new(cx, cy), workspace.CurrentCamera)
        end
    end)
    return ok
end

local function applyDirectColorToItem(item, c3, group)
    if not item or not c3 then return false end
    local char = lp.Character
    if not char then return false end
    group = tonumber(group) or 1
    local assetName = item.assetName or item.name

    local slotKey = item.slotKey
    if not slotKey or slotKey == "" then
        slotKey = (item.cat == "Heels" or item.cat == "Skirts" or item.cat == "Bodices" or item.cat == "Wings" or item.cat == "Halos")
                  and item.cat
                  or ("Acc_" .. (assetName or ""):gsub("%s+", ""))
    end

    local function isSkinPart(p)
        local pn = p.Name:lower()
        if standardBodyParts[p.Name] or p.Name == "HumanoidRootPart" or p.Name == "Head" then
            return true
        end
        if pn == "coloredleg" or pn:find("skin") or pn:find("skintone")
           or pn == "leg" or pn == "arm" or pn == "hand" or pn == "foot" or pn == "thigh" or pn == "body" then
            return true
        end
        return false
    end

    local function modelHasNumberedGroups(m)
        for _, d in ipairs(m:GetDescendants()) do
            if d:GetAttribute("MultiColor") ~= nil or tonumber(d.Name) ~= nil then
                return true
            end
        end
        return tonumber(m.Name) ~= nil or m:GetAttribute("MultiColor") ~= nil
    end

    local function partMatchesGroup(p, hasNumbers)
        local mc = p:GetAttribute("MultiColor")
        if mc ~= nil then
            return tonumber(mc) == group
        end
        if hasNumbers then
            if tonumber(p.Name) == group then return true end
            local par = p.Parent
            while par and par ~= char do
                if tonumber(par.Name) == group then return true end
                par = par.Parent
            end
            return false
        end
        local n = (p.Name .. " " .. (p.Parent and p.Parent.Name or "")):lower()
        local isGroup2 = n:find("2") or n:find("trim") or n:find("lace") or n:find("accent")
                      or n:find("ribbon") or n:find("bow") or n:find("secondary") or n:find("buckle")
                      or n:find("detail") or n:find("gem") or n:find("jewel") or n:find("metal")
        if group == 2 then
            return isGroup2
        else
            return not isGroup2 or n:find("1") or n:find("main") or n:find("base")
        end
    end

    local function colorSinglePart(p)
        if p:IsA("MeshPart") then
            pcall(function() p.TextureID = "" end)
        end
        local sm = p:FindFirstChildOfClass("SpecialMesh")
        if sm then
            pcall(function()
                sm.TextureId = ""
                sm.VertexColor = Vector3.new(c3.R, c3.G, c3.B)
            end)
        end
        local sa = p:FindFirstChildOfClass("SurfaceAppearance")
        if sa then
            pcall(function() sa.Color = c3 end)
        end
        p.Color = c3
    end

    local function applyColorToModelParts(m)
        if not m then return false end
        local hasNumbers = modelHasNumberedGroups(m)
        local allTargets = m:GetDescendants()
        if m:IsA("BasePart") then table.insert(allTargets, m) end

        local coloredCount = 0
        for _, p in ipairs(allTargets) do
            if p:IsA("BasePart") and not isSkinPart(p) then
                if partMatchesGroup(p, hasNumbers) then
                    colorSinglePart(p)
                    coloredCount = coloredCount + 1
                end
            elseif p:IsA("SpecialMesh") and p.Parent and not isSkinPart(p.Parent) then
                if partMatchesGroup(p.Parent, hasNumbers) or partMatchesGroup(p, hasNumbers) then
                    pcall(function()
                        p.TextureId = ""
                        p.VertexColor = Vector3.new(c3.R, c3.G, c3.B)
                    end)
                end
            end
        end

        if coloredCount == 0 and group == 2 then
            local parts = {}
            for _, p in ipairs(allTargets) do
                if p:IsA("BasePart") and not isSkinPart(p) then
                    table.insert(parts, p)
                end
            end
            table.sort(parts, function(a, b)
                return (a.Size.X * a.Size.Y * a.Size.Z) < (b.Size.X * b.Size.Y * b.Size.Z)
            end)
            local half = math.max(1, math.floor(#parts / 2))
            for i = 1, half do
                if parts[i] then
                    colorSinglePart(parts[i])
                    coloredCount = coloredCount + 1
                end
            end
        end
        return coloredCount > 0
    end

    local didColor = false
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
    if bakedFolder then
        local bakedModel = bakedFolder:FindFirstChild(slotKey)
        if not bakedModel then
            local cleanA = (item.name or ""):lower():gsub("[^%w]", "")
            local cleanB = (assetName or ""):lower():gsub("[^%w]", "")
            for _, ch in ipairs(bakedFolder:GetChildren()) do
                local cn = (ch:GetAttribute("RH_ItemName") or ch.Name):lower():gsub("[^%w]", "")
                if (cleanA ~= "" and (cn == cleanA or cn:find(cleanA, 1, true)))
                   or (cleanB ~= "" and (cn == cleanB or cn:find(cleanB, 1, true))) then
                    bakedModel = ch
                    break
                end
            end
        end
        if bakedModel then
            if applyColorToModelParts(bakedModel) then didColor = true end
            bakedLayersBackup[bakedModel.Name] = deepClone(bakedModel)
        end
    end

    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            if child.Name == item.name or child.Name == assetName then
                if applyColorToModelParts(child) then didColor = true end
            end
        end
    end

    return didColor
end

local function equipWings(item)
    local char = lp.Character
    if not char then return "No character" end
    local itemName = item.assetName or item.name

    task.spawn(function()
        disableFreezeOnlyConnections()
        local wingInst = nil

        -- 1. Official Royale High Wings TryOn Remote
        pcall(function()
            local net = ReplicatedStorage:FindFirstChild("Network")
            local fn = net and net:FindFirstChild("Functions") and net.Functions:FindFirstChild("Gui")
            local wFolder = fn and fn:FindFirstChild("Wings")
            local tryOnFn = wFolder and wFolder:FindFirstChild("TryOn")
            if tryOnFn and tryOnFn:IsA("RemoteFunction") then
                local wModel = tryOnFn:InvokeServer(itemName)
                if typeof(wModel) == "Instance" then
                    wingInst = wModel:FindFirstChild("Chest") or wModel
                end
            end
        end)

        -- 2. Check character/EquippedStorage or fallback to GetWingModel directly
        if not wingInst or countBaseParts(wingInst) == 0 then
            task.wait(0.25)
            wingInst = char:FindFirstChild("Wings_TRYON")
            if not wingInst then
                for _, info in ipairs(getEquippedFolders()) do
                    if info.category == "Wings" then
                        local fCh = info.folder:GetChildren()
                        if #fCh > 0 then wingInst = fCh[#fCh] break end
                    end
                end
            end
        end

        if not wingInst or countBaseParts(wingInst) == 0 then
            pcall(function()
                local getWingFn = ReplicatedStorage:FindFirstChild("GetWingModel")
                if getWingFn and getWingFn:IsA("RemoteFunction") then
                    local wm = getWingFn:InvokeServer(itemName)
                    if typeof(wm) == "Instance" then
                        wingInst = wm:FindFirstChild("Chest") or wm
                    end
                end
            end)
        end

        local baked = nil
        if wingInst and countBaseParts(wingInst) > 0 then
            baked = bakeItemToAvatar(wingInst, "Wings", "Wings", item)
        end

        hideTryOnOverlays()
        releasePlayerMovement()
        updateTabCounts()
        if refreshGridRef then refreshGridRef() end
        if setStatusRef then
            if baked then
                setStatusRef("Wings Equipped!", C.text)
            else
                setStatusRef("Wings Failed", C.muted)
            end
        end
    end)

    return "OK"
end

local function equipItem(item)
    local char = lp.Character
    if not char then return "No character" end

    local itemName = item.assetName or item.name
    local itemCat  = item.cat

    if itemCat == "Wings" then
        return equipWings(item)
    end

    -- Snapshot existing items before triggering preview so we only wait for the NEW item
    local beforeSet = {}
    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            beforeSet[child] = true
        end
    end
    for _, d in ipairs(char:GetChildren()) do
        beforeSet[d] = true
    end

    -- 1. Direct Server Remote
    if previewItemRemote then
        pcall(function()
            previewItemRemote:FireServer(itemName)
        end)
    end

    -- 2. Fallback Shop TryOn Button Click only if PreviewItem remote wasn't found
    if not previewItemRemote then
        task.spawn(function()
            local btn = findShopTryonButton(item.name) or findShopTryonButton(itemName)
            if btn then clickButtonVIM(btn) end
        end)
    end

    -- 3. Targeted Auto-Bake Polling: Waits specifically for THIS new item to replicate, then bakes it 1:1
    task.spawn(function()
        disableFreezeOnlyConnections()
        hideTryOnOverlays()
        task.delay(0.35, hideTryOnOverlays)
        task.delay(1.0, hideTryOnOverlays)

        local startTick = tick()
        local baked = nil
        while tick() - startTick < 4.5 do
            task.wait(0.10)
            local targetInst, foundCat = findSpecificItemOnAvatar(itemName, itemCat, beforeSet)
            if targetInst then
                local lastCount = countBaseParts(targetInst)
                local stableChecks = 0
                local settleStart = tick()
                while tick() - settleStart < 1.4 do
                    task.wait(0.10)
                    local curCount = countBaseParts(targetInst)
                    if curCount > 0 and curCount == lastCount and (tick() - settleStart) >= 0.45 then
                        stableChecks = stableChecks + 1
                        if stableChecks >= 3 then break end
                    else
                        lastCount = curCount
                        stableChecks = 0
                    end
                end
                baked = bakeItemToAvatar(targetInst, itemCat or foundCat, nil, item)
                if baked then
                    -- Background catchup: if more meshes stream in on mobile 1.0s-2.0s later, silently update the clone 1:1
                    local initialParts = countBaseParts(baked)
                    local trackedSource = targetInst
                    task.spawn(function()
                        for _, delaySec in ipairs({1.0, 1.2}) do
                            task.wait(delaySec)
                            if trackedSource and trackedSource.Parent and countBaseParts(trackedSource) > initialParts then
                                local updated = bakeItemToAvatar(trackedSource, itemCat or foundCat, nil, item)
                                if updated then initialParts = countBaseParts(updated) end
                            end
                        end
                    end)
                    break
                end
            end
        end

        -- Fallback: if named match missed, bake any non-base item currently on avatar not already in beforeSet
        if not baked then
            local fallbackItems = findRoyaleHighItemsOnAvatar()
            for _, fi in ipairs(fallbackItems) do
                if not beforeSet[fi] and countBaseParts(fi) > 0 then
                    baked = bakeItemToAvatar(fi, itemCat, nil, item)
                    if baked then break end
                end
            end
        end

        -- Release client movement freezes WITHOUT cancelling server preview prematurely
        releasePlayerMovement()
        updateTabCounts()
        if refreshGridRef then refreshGridRef() end
        if setStatusRef then
            if baked then
                setStatusRef("Cloned & Locked!", C.text)
            else
                setStatusRef("Previewing (tap Bake/Lock)", C.sub)
            end
        end
    end)

    return "OK"
end

local function formatCommas(str)
	local numStr = str:gsub("[^%d]", "")
	if numStr == "" then return "" end
	local k
	while true do
		numStr, k = string.gsub(numStr, "^(-?%d+)(%d%d%d)", '%1,%2')
		if k == 0 then break end
	end
	return numStr
end

local cachedStatLabels = { Diamonds = {}, Level = {}, Candy = {}, Cash = {} }
local lastStatScanTick = 0

local function rebuildStatLabelCache(force)
	local now = tick()
	if not force and (now - lastStatScanTick) < 10 then return end
	lastStatScanTick = now

	cachedStatLabels = { Diamonds = {}, Level = {}, Candy = {}, Cash = {} }
	local roots = {}
	for _, gui in ipairs(pg:GetChildren()) do
		if gui.Name == "Shop" then
			if gui:FindFirstChild("Header") then table.insert(roots, gui.Header) end
		elseif gui.Name ~= "RH_Wardrobe_v49" then
			table.insert(roots, gui)
		end
	end
	for _, root in ipairs(roots) do
	for _, d in ipairs(root:GetDescendants()) do
		if d:IsA("TextLabel") then
			local nl = d.Name:lower()
			local fn = d:GetFullName():lower()
			if not fn:find("rh_wardrobe") then
				if nl:find("diamondamount") or nl:find("diamondcount") then
					table.insert(cachedStatLabels.Diamonds, d)
				elseif nl:find("candyamount") or nl:find("candycount") then
					table.insert(cachedStatLabels.Candy, d)
				elseif nl:find("cashamount") or nl:find("cloudcash") or nl:find("dailyloginrewardscurrency") then
					table.insert(cachedStatLabels.Cash, d)
				elseif (nl:find("level") or fn:find("level")) and not fn:find("quality") and not fn:find("volume") and not fn:find("unlock") then
					table.insert(cachedStatLabels.Level, d)
				end
			end
		end
	end
	end
end
task.spawn(function()
	task.wait(0.5)
	pcall(rebuildStatLabelCache, true)
end)

local function updateDiamonds(val)
	if val == "" then return end
	rebuildStatLabelCache(false)
	local num = tonumber((val:gsub("[^%d]", "")))
	pcall(function() pg.Shop.Header.diamondamount.Text = val end)
	pcall(function() pg.HUD.Frame.Middle.DiamondsFrame.DiamondAmount.Text = val end)
	pcall(function() pg.MermaidShopDiamondCount.Frame.DiamondAmount.Text = val end)
	for _, d in ipairs(cachedStatLabels.Diamonds) do
		if d and d.Parent then pcall(function() d.Text = val end) end
	end
	if num then
		pcall(function()
			local ls = lp:FindFirstChild("leaderstats")
			if ls and ls:FindFirstChild("Diamonds") then ls.Diamonds.Value = num end
		end)
	end
end

local function updateLevel(val)
	if val == "" then return end
	rebuildStatLabelCache(false)
	local rawNumStr = val:gsub("[^%d]", "")
	local num = tonumber(rawNumStr)
	if num then
		pcall(function()
			local ls = lp:FindFirstChild("leaderstats")
			if ls and ls:FindFirstChild("Level") then ls.Level.Value = num end
		end)
	end
	for _, d in ipairs(cachedStatLabels.Level) do
		if d and d.Parent then
			local txt = d.Text or ""
			if txt:match("^Grade Level:%s*%d+") then
				pcall(function() d.Text = "Grade Level: " .. rawNumStr end)
			elseif txt:match("^Level:%s*%d+") then
				pcall(function() d.Text = "Level: " .. rawNumStr end)
			elseif txt:match("^Level%s+[%d,]+") then
				pcall(function() d.Text = "Level " .. val end)
			elseif txt:match("^[Ll][Vv][Ll]?%.?%s*[%d,]+") then
				pcall(function() d.Text = "Lv. " .. val end)
			elseif tonumber(txt:gsub("[^%d]", "")) then
				pcall(function() d.Text = val end)
			end
		end
	end
	if lp.Character then
		for _, d in ipairs(lp.Character:GetDescendants()) do
			if d:IsA("TextLabel") then
				local nl = d.Name:lower()
				local txt = d.Text or ""
				if (nl:find("level") or (d.Parent and d.Parent.Name:lower():find("level"))) and tonumber(txt:gsub("[^%d]", "")) then
					if txt:match("^Level%s+") then
						pcall(function() d.Text = "Level " .. val end)
					else
						pcall(function() d.Text = val end)
					end
				end
			end
		end
	end
end

local function updateCandy(val)
	if val == "" then return end
	rebuildStatLabelCache(false)
	pcall(function() pg.Shop.Header.candyamount.Text = val end)
	pcall(function() pg.HUD.Frame.HalloweenCandy.CandyAmount.Text = val end)
	for _, d in ipairs(cachedStatLabels.Candy) do
		if d and d.Parent then pcall(function() d.Text = val end) end
	end
end

local function updateCash(val)
	if val == "" then return end
	rebuildStatLabelCache(false)
	for _, d in ipairs(cachedStatLabels.Cash) do
		if d and d.Parent then pcall(function() d.Text = val end) end
	end
end

-- Client-side visual "Card Accepted!" + Receipt animation for Diamonds, Candy, and Cloud Cash
task.spawn(function()
	local hookedPurchaseGui = false
	while not hookedPurchaseGui do
		local pGuiRoot = pg:FindFirstChild("PurchaseGui")
		local pFailedDiag = pGuiRoot and pGuiRoot:FindFirstChild("PurchaseFailed", true)
		local pGui = pFailedDiag and pFailedDiag.Parent
		if pGui then
			hookedPurchaseGui = true
			local function hookCurrencyFail(failFrame, statKey, updateFn)
				if not failFrame then return end
				local animGen = 0
				local lastTriggerTick = 0
				failFrame:GetPropertyChangedSignal("Visible"):Connect(function()
					if not failFrame.Visible then return end
					local now = tick()
					if (now - lastTriggerTick) < 0.25 then return end
					lastTriggerTick = now

					local spoofStr = activeSpoofs[statKey]
					if not spoofStr or spoofStr == "" then return end
					local spoofBal = tonumber((spoofStr:gsub("[^%d]", ""))) or 0

					local priceLbl = pGui:FindFirstChild("Receipt")
					             and pGui.Receipt:FindFirstChild("PriceSplit")
					             and pGui.Receipt.PriceSplit:FindFirstChild("Price")
					local priceNum = priceLbl and tonumber((priceLbl.Text:gsub("[^%d]", ""))) or 0
					if spoofBal < priceNum then return end

					animGen = animGen + 1
					local myGen = animGen
					failFrame.Visible = false

					task.spawn(function()
						local newBal = math.max(0, spoofBal - priceNum)
						local formattedNew = formatCommas(tostring(newBal))
						activeSpoofs[statKey] = formattedNew
						updateFn(formattedNew)

						-- Update client-side "Owned: X" visual badge in Shop if visible
						pcall(function()
							local itemNameLbl = pGui:FindFirstChild("ItemName", true) or (pGui:FindFirstChild("Receipt") and pGui.Receipt:FindFirstChild("ItemName", true))
							local boughtName = itemNameLbl and itemNameLbl:IsA("TextLabel") and itemNameLbl.Text or ""
							local shop = pg:FindFirstChild("Shop")
							if shop and boughtName ~= "" then
								for _, lbl in ipairs(shop:GetDescendants()) do
									if lbl:IsA("TextLabel") and lbl.Visible then
										local tl = (lbl.Text or ""):lower()
										if tl:match("^owned:?%s*%d+") or tl == "not owned" then
											local card = lbl.Parent
											local hasMatch = false
											for _ = 1, 4 do
												if not card then break end
												for _, ch in ipairs(card:GetDescendants()) do
													if ch:IsA("TextLabel") and ch.Text == boughtName then
														hasMatch = true
														break
													end
												end
												if hasMatch then break end
												card = card.Parent
											end
											if hasMatch then
												local curOwned = tonumber((tl:gsub("[^%d]", ""))) or 0
												lbl.Text = "Owned: " .. tostring(curOwned + 1)
											end
										end
									end
								end
							end
						end)

						pcall(function()
							local s = pGui:FindFirstChild("Success") or pGuiRoot:FindFirstChild("Success", true)
							if s and s:IsA("Sound") then s:Play() end
						end)

						local pAcc = pGui:FindFirstChild("PurchaseAccepted")
						if pAcc then
							pAcc.Visible = true
							if pAcc:FindFirstChild("MainText") then
								pAcc.MainText.Text = "Card Accepted!"
							end
							for i = 1, 5 do
								local f = pAcc:FindFirstChild(tostring(i))
								if f and f:IsA("GuiObject") then f.Visible = true end
								if i > 1 then
									local prev = pAcc:FindFirstChild(tostring(i - 1))
									if prev and prev:IsA("GuiObject") then prev.Visible = false end
								end
								task.wait(0.15)
							end
							if pAcc:FindFirstChild("MainText") then
								pAcc.MainText.Text = "Purchase Complete!"
							end
						end

						pcall(function()
							local cSnd = pGui:FindFirstChild("Complete") or pGuiRoot:FindFirstChild("Complete", true)
							if cSnd and cSnd:IsA("Sound") then cSnd:Play() end
						end)

						local receipt = pGui:FindFirstChild("Receipt")
						if receipt then
							receipt.Position = UDim2.new(0, 0, 0.275, 0)
							receipt.Visible = true
							pcall(function()
								receipt:TweenPosition(UDim2.new(0, 0, 0.72, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.45, true)
							end)
						end

						task.wait(2.2)
						if animGen ~= myGen then return end
						if pAcc then pAcc.Visible = false end
						if receipt then
							receipt.Visible = false
							receipt.Position = UDim2.new(0, 0, 0.275, 0)
						end
						if pGui:FindFirstChild("PurchaseScreen") then
							pGui.PurchaseScreen.Visible = true
						end
						if pGuiRoot and pGuiRoot:IsA("ScreenGui") then
							pGuiRoot.Enabled = false
						end
					end)
				end)
			end

			hookCurrencyFail(pGui:FindFirstChild("PurchaseFailed"), "Diamonds", updateDiamonds)
			hookCurrencyFail(pGui:FindFirstChild("PurchaseFailedCandy"), "Candy", updateCandy)
			hookCurrencyFail(pGui:FindFirstChild("PurchaseFailedDailyLoginRewards"), "Cloud Cash", updateCash)
			-- Also catch generic currency fail frames if named differently
			for _, ch in ipairs(pGui:GetChildren()) do
				local cnl = ch.Name:lower()
				if cnl:find("purchasefailed") and ch.Name ~= "PurchaseFailed" and ch.Name ~= "PurchaseFailedCandy" and ch.Name ~= "PurchaseFailedDailyLoginRewards" then
					if cnl:find("candy") then
						hookCurrencyFail(ch, "Candy", updateCandy)
					elseif cnl:find("cash") or cnl:find("cloud") or cnl:find("daily") then
						hookCurrencyFail(ch, "Cloud Cash", updateCash)
					end
				end
			end
		end
		task.wait(1)
	end
end)

local function openInsp(item)
	selectedItem = item
	inspName.Text = item.name:upper()

		inspPrice.Text = "Equipped: " .. (item.cat or "Custom")
		equipBtn.Text = "UNEQUIP / REMOVE"
		equipBtn.BackgroundColor3 = C.card
		equipBtn.TextColor3 = C.text
	else
		inspPrice.Text = (item.price ~= "" and ("Price: " .. item.price) or "Price: Special")
		equipBtn.Text = "WEAR / STACK"
		equipBtn.BackgroundColor3 = C.btnHi
		equipBtn.TextColor3 = C.btnTextHi
	end
	if item.img and item.img ~= "" then
		inspImg.Image = item.img
		inspPlaceholder.Visible = false
	else
		inspImg.Image = ""
		inspPlaceholder.Visible = true
	end
	Inspector.Visible = true
	Inspector.Position = UDim2.new(0, 0, 1, -InspHeight)
end

local function handleWearOrRemove()
	if not selectedItem then return end
	if selectedItem.isWorn then
		setStatus("Unequipping...", C.sub)
		unequipWornItem(selectedItem)
		setStatus("Item Removed!", C.text)
		closeInsp()
		updateTabCounts()
		if refreshGridRef then refreshGridRef("Worn") end
	else
		setStatus("Stacking...", C.sub)
		local res = equipItem(selectedItem)
		if res == "OK" then
			setStatus("Cloning & Stacking...", C.sub)
		else
			setStatus("Failed", C.muted)
		end
		updateTabCounts()
	end
end
equipBtn.Activated:Connect(handleWearOrRemove)

local function makeCard(item, order)
	local card = mk("Frame", {
		Name             = "Card_" .. item.name:sub(1,10),
		BackgroundColor3 = C.card,
		BorderSizePixel  = 0,
		LayoutOrder      = order,
		ZIndex           = 13,
	}, GridScroll)
	corner(4, card)
	stroke(1, C.border, card)

	local img = mk("ImageLabel", {
		Size             = UDim2.new(1, -6, 0, 38),
		Position         = UDim2.new(0, 3, 0, 3),
		BackgroundColor3 = C.btn,
		BorderSizePixel  = 0,
		ScaleType        = Enum.ScaleType.Fit,
		Image            = item.img or "",
		ZIndex           = 14,
	}, card)
	corner(3, img)

	if not item.img or item.img == "" then
		if item.cat == "Wings" then
			_G.RH_WingModelCache = _G.RH_WingModelCache or {}
			local phLbl = mk("TextLabel", {
				Size               = UDim2.new(1, 0, 1, 0),
				BackgroundTransparency = 1,
				Text               = "...",
				TextColor3         = C.muted,
				Font               = Enum.Font.GothamBold,
				TextSize           = 7,
				ZIndex             = 15,
			}, img)
			task.spawn(function()
				local wingKey = item.assetName or item.name
				local tmpl = _G.RH_WingModelCache[wingKey]
				if tmpl == nil then
					local getWingFn = ReplicatedStorage:FindFirstChild("GetWingModel")
					if getWingFn and getWingFn:IsA("RemoteFunction") then
						local ok, res = pcall(function() return getWingFn:InvokeServer(wingKey) end)
						tmpl = (ok and typeof(res) == "Instance") and res or false
						_G.RH_WingModelCache[wingKey] = tmpl
					else
						_G.RH_WingModelCache[wingKey] = false
					end
				end
				if tmpl and typeof(tmpl) == "Instance" and img and img.Parent then
					local clone = tmpl:Clone()
					local vpf = mk("ViewportFrame", {
						Size               = UDim2.new(1, 0, 1, 0),
						BackgroundTransparency = 1,
						ZIndex             = 15,
					}, img)
					local cam = Instance.new("Camera")
					cam.Parent = vpf
					vpf.CurrentCamera = cam
					clone.Parent = vpf
					local cf, sz
					if clone:IsA("Model") then
						cf, sz = clone:GetBoundingBox()
					else
						local bp = clone:FindFirstChildWhichIsA("BasePart", true)
						cf = bp and bp.CFrame or CFrame.new()
						sz = bp and bp.Size or Vector3.new(2, 2, 1)
					end
					local maxDim = math.max(sz.X, sz.Y, sz.Z, 1)
					cam.CFrame = CFrame.new(cf.Position + Vector3.new(0, 0, maxDim * 1.15), cf.Position)
					if phLbl then phLbl.Visible = false end
				elseif phLbl and phLbl.Parent then
					phLbl.Text = "WING"
				end
			end)
		else
			mk("TextLabel", {
				Size               = UDim2.new(1, 0, 1, 0),
				BackgroundTransparency = 1,
				Text               = item.isWorn and "WORN" or "NO IMG",
				TextColor3         = item.isWorn and C.text or C.muted,
				Font               = Enum.Font.GothamBold,
				TextSize           = 7,
				ZIndex             = 15,
			}, img)
		end
	end

	mk("TextLabel", {
		Size               = UDim2.new(1, -4, 0, 20),
		Position           = UDim2.new(0, 2, 0, 44),
		BackgroundTransparency = 1,
		Text               = item.name,
		TextColor3         = C.text,
		Font               = Enum.Font.Gotham,
		TextSize           = 6.5,
		TextWrapped        = true,
		TextYAlignment     = Enum.TextYAlignment.Top,
		ZIndex             = 14,
	}, card)

	local btn = mk("TextButton", {
		Size               = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text               = "",
		ZIndex             = 16,
	}, card)
	btn.Activated:Connect(function() openInsp(item) end)

	cardInstances[item] = card
	return card
end

local function refreshGrid()
	for _, card in pairs(cardInstances) do card:Destroy() end
	cardInstances = {}

	if currentCat == "Stats" then return end

	local list = {}
	if currentCat == "Worn" then
		list = getWornItemsList()
	else
		list = categories[currentCat] or {}
	end

	local search = currentSearch:lower()
	local filtered = {}
	for _, item in ipairs(list) do
		if search == "" or item.name:lower():find(search, 1, true) then
			table.insert(filtered, item)
		end
	end

	if currentCat == "Worn" then
		if #filtered == 0 then
			local emptyCard = mk("Frame", {
				Name             = "EmptyNotice",
				Size             = UDim2.new(1, -8, 0, 64),
				BackgroundColor3 = C.card,
				BorderSizePixel  = 0,
				LayoutOrder      = 1,
				ZIndex           = 13,
			}, GridScroll)
			corner(4, emptyCard)
			stroke(1, C.border, emptyCard)

			mk("TextLabel", {
				Size               = UDim2.new(1, -12, 1, 0),
				Position           = UDim2.new(0, 6, 0, 0),
				BackgroundTransparency = 1,
				Text               = "NO ITEMS WORN\n\nEquip items from Heels, Skirts, or Bodices to wear and stack them!",
				TextColor3         = C.sub,
				Font               = Enum.Font.Gotham,
				TextSize           = 7.5,
				TextWrapped        = true,
				ZIndex             = 14,
			}, emptyCard)
			cardInstances[emptyCard] = emptyCard
			return
		end
	else
		-- Sort: Items with valid images FIRST, items with NO image LAST
		table.sort(filtered, function(a, b)
			local aHas = (a.img and a.img ~= "") and 1 or 0
			local bHas = (b.img and b.img ~= "") and 1 or 0
			if aHas ~= bHas then
				return aHas > bHas
			end
			return a.name:lower() < b.name:lower()
		end)
	end

	for order, item in ipairs(filtered) do
		makeCard(item, order)
	end
end

refreshGridRef = function(targetCat)
    if not targetCat or targetCat == currentCat then
        refreshGrid()
    end
end

		StatsPanel.Visible     = true
	else
		StatsPanel.Visible     = false
		SearchBoxFrame.Visible = true
		GridScroll.Visible     = true
		refreshGrid()
	end
end
