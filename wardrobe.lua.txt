-- ==========================================================
-- ROYALE HIGH DRESS-UP & INVENTORY DEEP PROBE
-- ==========================================================
local lp = game:GetService("Players").LocalPlayer
local pg = lp:WaitForChild("PlayerGui")
local rep = game:GetService("ReplicatedStorage")

task.spawn(function()
    local out = {}
    local function log(str)
        table.insert(out, str)
        print(str)
    end

    log("=== ROYALE HIGH DRESS-UP & INVENTORY PROBE ===")

    -- 1. Scan StarterDressUpHUD and CharacterCreation
    local dressUpGuis = {"StarterDressUpHUD", "CharacterCreation", "ToolInventory", "Shop"}
    for _, gName in ipairs(dressUpGuis) do
        local g = pg:FindFirstChild(gName)
        if g then
            log("[GUI FOUND] " .. gName)
            for _, d in ipairs(g:GetDescendants()) do
                if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") or d:IsA("BindableFunction") or d:IsA("BindableEvent") then
                    log("  -> " .. d.ClassName .. ": " .. d:GetFullName())
                end
            end
        else
            log("[GUI NOT FOUND] " .. gName)
        end
    end

    -- 2. Scan ReplicatedStorage for DressUp / Customization Remotes & Modules
    log("\n--- REPLICATED STORAGE DRESS-UP ASSETS ---")
    for _, d in ipairs(rep:GetDescendants()) do
        local nameLower = d.Name:lower()
        if nameLower:find("dressup") or nameLower:find("wear") or nameLower:find("equip") 
            or nameLower:find("custo") or nameLower:find("outfit") or nameLower:find("itemdata")
            or nameLower:find("inventory") or nameLower:find("clothing") then
            log("  -> " .. d.ClassName .. ": " .. d:GetFullName())
        end
    end

    -- 3. Probe Shop ButtonCopy (How the game generates real item thumbnails & buttons)
    log("\n--- SHOP BUTTONCOPY TEST ---")
    local dir = pg:FindFirstChild("Shop") and pg.Shop:FindFirstChild("Directory")
    if dir and dir:FindFirstChild("Shoes") and dir.Shoes:FindFirstChild("GetButtonCopy") then
        pcall(function()
            local copyFunc = dir.Shoes.GetButtonCopy
            log("GetButtonCopy exists in Shoes! Class: " .. copyFunc.ClassName)
        end)
    end

    -- 4. Check Character Anchors / Movement Freeze Root Cause
    log("\n--- CHARACTER STATUS ---")
    local char = lp.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChild("Humanoid")
        log("HRP Anchored: " .. tostring(hrp and hrp.Anchored))
        log("Humanoid WalkSpeed: " .. tostring(hum and hum.WalkSpeed))
        log("Humanoid PlatformStand: " .. tostring(hum and hum.PlatformStand))
    end

    log("==============================================")
    local report = table.concat(out, "\n")
    if setclipboard then
        setclipboard(report)
        log("[!] Report successfully copied to your clipboard!")
    end
end)
