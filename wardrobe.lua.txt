-- 1. ABSOLUTE MOVEMENT UNFREEZER
-- ==========================================================
local function unfreezeControls()
	pcall(function()
		local PlayerModule = lp:FindFirstChild("PlayerScripts") and lp.PlayerScripts:FindFirstChild("PlayerModule")
		if PlayerModule then
			local Controls = require(PlayerModule)
			if Controls and Controls.GetControls then
				Controls:GetControls():Enable()
			end
		end
	end)
	pcall(function()
		ContextActionService:UnbindAction("FreezeMovement")
		ContextActionService:UnbindAction("ShopFreeze")
		ContextActionService:UnbindAction("TryOnFreeze")
		ContextActionService:UnbindAction("DisableMovement")
	end)
end

local function bindMovementGuards(char)
	if not char then return end
	local hum = char:WaitForChild("Humanoid", 4)
	local hrp = char:WaitForChild("HumanoidRootPart", 4)

	if hum then
		hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
			if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
		end)
		hum:GetPropertyChangedSignal("JumpPower"):Connect(function()
			if hum.JumpPower < 50 then hum.JumpPower = 50 end
		end)
		hum:GetPropertyChangedSignal("PlatformStand"):Connect(function()
			if hum.PlatformStand then hum.PlatformStand = false end
		end)
		if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
		if hum.JumpPower < 50 then hum.JumpPower = 50 end
		if hum.PlatformStand then hum.PlatformStand = false end
	end

	if hrp then
		hrp:GetPropertyChangedSignal("Anchored"):Connect(function()
			if hrp.Anchored then hrp.Anchored = false end
		end)
		if hrp.Anchored then hrp.Anchored = false end
	end

	unfreezeControls()
end

if lp.Character then bindMovementGuards(lp.Character) end
lp.CharacterAdded:Connect(bindMovementGuards)

-- 2. METATABLE DELETION INTERCEPTOR & RECURSIVE CLONING
-- ==========================================================
local function deepClone(inst)
	if not inst then return nil end
	pcall(function()
		inst.Archivable = true
		for _, d in ipairs(inst:GetDescendants()) do
			d.Archivable = true
		end
	end)
	return inst:Clone()
end

local function reattachModelJoints(mod, char, joints)
	for _, wd in ipairs(joints) do
		local p0 = mod:FindFirstChild(wd.part0Name, true)
		local p1 = char:FindFirstChild(wd.part1Name)
		if p0 and p1 then
			if wd.className == "Motor6D" then
				local m = Instance.new("Motor6D")
				m.Part0 = p0
				m.Part1 = p1
				if wd.c0 then m.C0 = wd.c0 end
				if wd.c1 then m.C1 = wd.c1 end
				m.Parent = p0
			elseif wd.className == "WeldConstraint" then
				local wc = Instance.new("WeldConstraint")
				wc.Part0 = p0
				wc.Part1 = p1
				wc.Parent = p0
			else
				local w = Instance.new("Weld")
				w.Part0 = p0
				w.Part1 = p1
				if wd.c0 then w.C0 = wd.c0 end
				if wd.c1 then w.C1 = wd.c1 end
				w.Parent = p0
			end
		end
	end
end

local function restoreAllSavedLayers()
	local char = lp.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")

	for rawName, data in pairs(savedWornLayers) do
		local layerName = rawName .. "_Layer"
		local existing = char:FindFirstChild(layerName) or char:FindFirstChild(rawName)
		if not existing and data.backup then
			pcall(function()
				local copy = deepClone(data.backup)
				copy.Name = layerName
				copy:SetAttribute("CustomSavedLayer", true)

				if data.isAccessory and hum then
					hum:AddAccessory(copy)
				elseif data.isModel then
					reattachModelJoints(copy, char, data.joints or {})
					copy.Parent = char
				else
					copy.Parent = char
				end
			end)
		end
	end
end

local function clearAllWornLayers()
	local char = lp.Character
	for rawName, data in pairs(savedWornLayers) do
		if char then
			local layerName = rawName .. "_Layer"
			local i1 = char:FindFirstChild(layerName)
			local i2 = char:FindFirstChild(rawName)
			if i1 then
				i1:SetAttribute("CustomSavedLayer", nil)
				pcall(function() i1:Destroy() end)
			end
			if i2 then
				i2:SetAttribute("CustomSavedLayer", nil)
				pcall(function() i2:Destroy() end)
			end
		end
	end
	savedWornLayers = {}

	if char then
		for _, child in ipairs(char:GetChildren()) do
			if child:GetAttribute("CustomSavedLayer") or child.Name:find("_Layer") then
				child:SetAttribute("CustomSavedLayer", nil)
				pcall(function() child:Destroy() end)
			end
		end
	end
end

task.spawn(function()
	while true do
		task.wait(0.5)
		pcall(function()
			local char = lp.Character
			if char and next(savedWornLayers) ~= nil then
				restoreAllSavedLayers()
			end
		end)
	end
end)

local function findItemImage(card)
	local cur = card:FindFirstChild("CurrentImage", true)
	if cur and cur:IsA("ImageLabel") and cur.Image ~= "" and not ignoredImages[cur.Image] then
		return cur.Image, cur.ImageRectOffset, cur.ImageRectSize
	end

	local img = card:FindFirstChild("Image")
	if img and img:IsA("ImageLabel") and img.Image ~= "" and not ignoredImages[img.Image] then
		return img.Image, img.ImageRectOffset, img.ImageRectSize
	end

	local prev = card:FindFirstChild("Preview")
	if prev and prev:IsA("ImageLabel") and prev.Image ~= "" and not ignoredImages[prev.Image] then
		return prev.Image, prev.ImageRectOffset, prev.ImageRectSize
	end

	for _, child in ipairs(card:GetDescendants()) do
		if (child:IsA("ImageLabel") or child:IsA("ImageButton")) and child.Visible then
			local id = tostring(child.Image)
			if id ~= "" and not ignoredImages[id] 
				and child.Name ~= "ButtonBackground" 
				and child.Name ~= "Imagenon" 
				and child.Name ~= "BackArrow" 
				and child.Name ~= "NextArrow" 
				and child.Name ~= "Left" 
				and child.Name ~= "Right" then
				return id, child.ImageRectOffset, child.ImageRectSize
			end
		end
	end

	return "", Vector2.new(0, 0), Vector2.new(0, 0)
end

local function classifyCategory(itemName, defaultCat)
	local lower = itemName:lower()
	if lower:find("halo") then
		return "Halos"
	elseif lower:find("heel") or lower:find("boot") or lower:find("shoe") or lower:find("loafer") 
		or lower:find("platform") or lower:find("footwear") or lower:find("sandal") or lower:find("sneaker") then
		return "Heels"
	elseif lower:find("skirt") or lower:find("dress") or lower:find("tut") then
		return "Skirts"
	elseif lower:find("bodice") or lower:find("corset") or lower:find("bustier") or lower:find("torso") then
		return "Bodices"
	elseif lower:find("wing") then
		return "Wings"
	else
		return defaultCat or "Accessories"
	end
end

local function processLiveCard(card, defaultCat)
	if not card:IsA("Frame") or card.Name == "UIListLayout" or card.Name == "UIGridLayout" or card.Name == "UIPadding" then return end

	local nameObj = card:FindFirstChild("name")
	local itemName = (nameObj and nameObj:IsA("TextLabel") and nameObj.Text ~= "") and nameObj.Text or card.Name
	if itemName == "" or itemName == " " then return end

	local priceObj = card:FindFirstChild("price")
	local itemPrice = (priceObj and priceObj:IsA("TextLabel")) and priceObj.Text or ""
	itemPrice = itemPrice:gsub("ðŸ’Ž", "Diamonds: "):gsub("💎", "Diamonds: ")
	itemPrice = itemPrice:gsub("ðŸ ¬", "Candy: "):gsub("🍬", "Candy: ")
	itemPrice = itemPrice:gsub("ðŸ’µ", "Cash: "):gsub("💵", "Cash: ")

	local descObj = card:FindFirstChild("smalldescription")
	local itemDesc = ""
	if descObj and descObj.Text ~= "" and descObj.Text ~= " " then
		itemDesc = descObj.Text
	else
		local noteObj = card:FindFirstChild("note")
		if noteObj and noteObj.Text ~= "" and noteObj.Text ~= "Sparkles!" then
			itemDesc = noteObj.Text
		end
	end

	local hasToggles = card:FindFirstChild("sliderwrapper") ~= nil or card:FindFirstChild("CurrentImage") ~= nil
	local imgId, rOff, rSz = findItemImage(card)
	local targetCat = classifyCategory(itemName, defaultCat)

	addItem(targetCat, {
		name = itemName,
		price = itemPrice,
		desc = itemDesc,
		image = imgId,
		rectOff = rOff,
		rectSize = rSz,
		hasToggles = hasToggles,
		cardRef = card,
	})
end

local function forcePopulateShop()
	local shop = pg:FindFirstChild("Shop")
	if not shop then return end
	local dir = shop:FindFirstChild("Directory")
	if not dir then return end

	for _, child in ipairs(dir:GetDescendants()) do
		if (child:IsA("TextButton") or child:IsA("ImageButton")) and child.Visible then
			local n = child.Name:lower()
			if n:find("shoe") or n:find("skirt") or n:find("access") or n:find("wing") or n:find("monthly") then
				pcall(function()
					if firesignal then firesignal(child.MouseButton1Click) end
				end)
			end
		end
	end
end

local function scrapeReplicatedStorage()
	pcall(function()
		for _, mod in ipairs(ReplicatedStorage:GetDescendants()) do
			if mod:IsA("ModuleScript") then
				local n = mod.Name:lower()
				if n:find("searchmeta") or n:find("itemmeta") or n:find("shopdata") or n:find("catalog") then
					local s, data = pcall(require, mod)
					if s and type(data) == "table" then
						for k, v in pairs(data) do
							if type(v) == "table" then
								local name = v.Name or v.ItemName or (type(k) == "string" and k)
								if name and type(name) == "string" and name ~= "" then
									local cat = classifyCategory(name, v.Category)
									local price = v.Price and tostring(v.Price) or ""
									local desc = v.Description or v.Desc or ""
									local img = v.Image or v.Thumbnail or ""
									if type(img) == "number" then img = "rbxassetid://" .. tostring(img) end
									addItem(cat, {
										name = name,
										price = price,
										desc = desc,
										image = img,
										rectOff = Vector2.new(0, 0),
										rectSize = Vector2.new(0, 0),
										hasToggles = (v.Toggles ~= nil),
										cardRef = nil,
									})
								end
							end
						end
					end
				end
			end
		end
	end)
end

local function searchShopForItem(itemName)
	local shop = pg:FindFirstChild("Shop")
	if not shop then return end

	-- Find search box inside Shop and trigger search
	for _, d in ipairs(shop:GetDescendants()) do
		if d:IsA("TextBox") and (d.Name:lower():find("search") or (d.PlaceholderText and d.PlaceholderText:lower():find("search"))) then
			pcall(function()
				d.Text = itemName
				if getconnections then
					for _, c in pairs(getconnections(d.FocusLost)) do c:Fire(true) end
				end
			end)
		end
	end

	-- Also enable off-sale / all items filter if present
	for _, d in ipairs(shop:GetDescendants()) do
		if (d:IsA("TextButton") or d:IsA("ImageButton")) and (d.Name:lower():find("offsale") or (d:IsA("TextButton") and d.Text:lower():find("offsale"))) then
			pcall(function()
				if firesignal then firesignal(d.MouseButton1Click) end
				if getconnections then
					for _, c in pairs(getconnections(d.MouseButton1Click)) do c:Fire() end
				end
			end)
		end
	end
end

local function findLiveShopCard(itemName)
	local shop = pg:FindFirstChild("Shop")
	if not shop then return nil end

	local lower = itemName:lower()
	for _, d in ipairs(shop:GetDescendants()) do
		if d:IsA("Frame") and (d:FindFirstChild("Tryon") or d:FindFirstChild("Button")) then
			local n = d:FindFirstChild("name")
			if (n and n:IsA("TextLabel") and n.Text:lower() == lower) or d.Name:lower() == lower then
				return d
			end
		end
	end

	-- Auto-search off-sale item in live shop
	searchShopForItem(itemName)
	task.wait(0.12)

	for _, d in ipairs(shop:GetDescendants()) do
		if d:IsA("Frame") and (d:FindFirstChild("Tryon") or d:FindFirstChild("Button")) then
			local n = d:FindFirstChild("name")
			if (n and n:IsA("TextLabel") and n.Text:lower() == lower) or d.Name:lower() == lower then
				return d
			end
		end
	end

	return nil
end

local function triggerShopTryon(card)
	if not card then return false end
	local btn = card:FindFirstChild("Tryon") or card:FindFirstChild("Button") or card:FindFirstChildWhichIsA("TextButton")
	if not btn then return false end

	pcall(function()
		if firesignal then firesignal(btn.MouseButton1Click) end
	end)
	pcall(function()
		if getconnections then
			for _, con in pairs(getconnections(btn.MouseButton1Click)) do
				con:Fire()
			end
		end
	end)
	return true
end

local function equipItemWithLayering(itemName, itemCat, cardRef)
	local char = lp.Character
	if not char then return end

	if itemCat == "Skirts" or itemCat == "Heels" then
		for name, data in pairs(savedWornLayers) do
			if data.category == itemCat then
				local lyr = char:FindFirstChild(name .. "_Layer") or char:FindFirstChild(name)
				if lyr then
					lyr:SetAttribute("CustomSavedLayer", nil)
					pcall(function() lyr:Destroy() end)
				end
				savedWornLayers[name] = nil
			end
		end
	end

	local preExisting = {}
	for _, child in ipairs(char:GetChildren()) do
		if child:IsA("Accessory") or child:IsA("Model") then
			preExisting[child] = true
		end
	end

	local card = cardRef or findLiveShopCard(itemName)
	triggerShopTryon(card)

	pcall(function()
		local pe = pg.CharacterCreation.Inventory:FindFirstChild("PromptEquip")
		if pe and pe:IsA("BindableEvent") then pe:Fire(itemName) end
	end)

	task.spawn(function()
		local newItem = nil
		for step = 1, 16 do
			task.wait(0.06)
			for _, child in ipairs(char:GetChildren()) do
				if (child:IsA("Accessory") or child:IsA("Model")) 
				   and not standardBodyParts[child.Name] 
				   and not preExisting[child] 
				   and not child:GetAttribute("CustomSavedLayer") then
					newItem = child
					break
				end
			end
			if newItem then break end
		end

		if newItem then
			pcall(function()
				newItem:SetAttribute("CustomSavedLayer", true)
				local jointsList = {}
				for _, w in ipairs(newItem:GetDescendants()) do
					if (w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D")) and w.Part1 and w.Part1:IsDescendantOf(char) then
						table.insert(jointsList, {
							part0Name = w.Part0 and w.Part0.Name or "",
							part1Name = w.Part1.Name,
							c0 = (w:IsA("Motor6D") or w:IsA("Weld")) and w.C0 or nil,
							c1 = (w:IsA("Motor6D") or w:IsA("Weld")) and w.C1 or nil,
							className = w.ClassName,
						})
					end
				end

				local backupCopy = deepClone(newItem)
				for _, d in ipairs(backupCopy:GetDescendants()) do
					if d.Name:lower():find("tryon") then d:Destroy() end
				end
				backupCopy:SetAttribute("IsTryOn", nil)
				backupCopy:SetAttribute("TryOn", nil)

				savedWornLayers[itemName] = {
					backup = backupCopy,
					joints = jointsList,
					isAccessory = newItem:IsA("Accessory"),
					isModel = newItem:IsA("Model"),
					category = itemCat,
				}
			end)
		end

		restoreAllSavedLayers()
		unfreezeControls()
	end)
end-- ==========================================================

local function formatCommas(str)
	local numStr = str:gsub("[^%d]", "")
	if numStr == "" then return "" end
	local k
	while true do
		numStr, k = string.gsub(numStr, "^(-?%d+)(%d%d%d)", '%1,%2')
		if k == 0 then break end
	end
	return numStr
end

local function updateDiamonds(val)
	if val == "" then return end
	pcall(function() pg.Shop.Header.diamondamount.Text = val end)
	pcall(function() pg.HUD.Frame.Middle.DiamondsFrame.DiamondAmount.Text = val end)
	pcall(function() pg.MermaidShopDiamondCount.Frame.DiamondAmount.Text = val end)
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("TextLabel") and (d.Name:lower():find("diamondamount") or d.Name:lower():find("diamondcount")) then
			pcall(function() d.Text = val end)
		end
	end
end

local function updateLevel(val)
	if val == "" then return end
	pcall(function() pg.TopBar.Topbar.ButtonList["NOT IN USE"].LevelNumber.Text = val end)
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("TextLabel") and (d.Name:lower():find("levelnumber") or (d.Name:lower() == "level" and tonumber(d.Text:gsub("[^%d]", "")))) then
			pcall(function() d.Text = val end)
		end
	end
end

local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = Main.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UIS.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)
