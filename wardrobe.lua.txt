-- ==========================================================
-- ROYALE HIGH ENVIRONMENT DIAGNOSTIC (AUTO-CLIPBOARD)
-- Pure Monochrome (Black & White, zero emojis)
-- ==========================================================

local Players = game:GetService("Players")
local StarterGui = game:GetService("StarterGui")
local lp = Players.LocalPlayer
local pg = lp:WaitForChild("PlayerGui")

local function notify(title, msg)
	pcall(function()
		StarterGui:SetCore("SendNotification", {
			Title = title,
			Text = msg,
			Duration = 6,
		})
	end)
end

local function copyToClipboard(text)
	if setclipboard then
		setclipboard(text)
		return true
	elseif toclipboard then
		toclipboard(text)
		return true
	end
	return false
end

notify("SCANNER STARTED", "Collecting shop paths and button data...")

task.spawn(function()
	task.wait(0.6)

	local foundShops = {}
	local tryonButtons = {}
	local itemFrames = 0
	local eventType = "Unknown"

	-- 1. Identify shop and wardrobe screen GUIs
	for _, gui in ipairs(pg:GetChildren()) do
		if gui:IsA("ScreenGui") then
			local gName = gui.Name:lower()
			if gName:find("shop") or gName:find("dress") or gName:find("wardrobe") or gName:find("creation") or gName:find("inventory") then
				table.insert(foundShops, gui.Name)
			end
		end
	end

	-- 2. Scan all try-on buttons and item cards
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("GuiButton") then
			local n = d.Name:lower()
			local t = (d:IsA("TextButton") and d.Text:lower()) or ""
			if n:find("try") or n:find("preview") or t:find("try") or t:find("preview") then
				table.insert(tryonButtons, d:GetFullName())
			end
		elseif d:IsA("Frame") and (d.Name:lower() == "item" or d:FindFirstChild("name") or d:FindFirstChild("Tryon")) then
			itemFrames = itemFrames + 1
		end
	end

	-- 3. Check event connection types
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("GuiButton") and d.Name:lower():find("try") then
			pcall(function()
				if getconnections then
					local act = #getconnections(d.Activated) > 0
					local m1c = #getconnections(d.MouseButton1Click) > 0
					if act and m1c then
						eventType = "Both (Activated & MouseButton1Click)"
					elseif act then
						eventType = "Activated"
					elseif m1c then
						eventType = "MouseButton1Click"
					else
						eventType = "No active script connections found"
					end
				end
			end)
			break
		end
	end

	-- 4. Build clipboard text report
	local lines = {}
	table.insert(lines, "=== ROYALE HIGH DIAGNOSTIC REPORT ===")
	table.insert(lines, "PlaceId: " .. tostring(game.PlaceId))
	table.insert(lines, "Shop GUIs Found: " .. (#foundShops > 0 and table.concat(foundShops, ", ") or "None detected"))
	table.insert(lines, "Total Item Frames: " .. tostring(itemFrames))
	table.insert(lines, "Total Try-On Buttons: " .. tostring(#tryonButtons))
	table.insert(lines, "Button Event Type: " .. eventType)
	table.insert(lines, "\nSample Try-On Button Paths (Top 5):")
	for i = 1, math.min(5, #tryonButtons) do
		table.insert(lines, tostring(i) .. ". " .. tryonButtons[i])
	end
	table.insert(lines, "======================================")

	local fullReport = table.concat(lines, "\n")
	local copied = copyToClipboard(fullReport)

	-- 5. Notify user
	if copied then
		notify("DONE!", "Diagnostic report copied to your clipboard!")
	else
		notify("DONE!", "Scan finished. Check developer console (F9).")
	end

	print(fullReport)
end)
