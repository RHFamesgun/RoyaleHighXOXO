local logs = {}
local harvested = {}
local isRunning = false

local function appendLog(txt)
	table.insert(logs, txt)
	local disp = table.concat(logs, "\n")
	LogText.Text = disp
	LogText.Size = UDim2.new(1, -12, 0, #logs * 14 + 20)
	LogBox.CanvasSize = UDim2.new(0, 0, 0, #logs * 14 + 24)
	LogBox.CanvasPosition = Vector2.new(0, LogBox.CanvasSize.Y.Offset)
end

local function clean(s)
	if not s then return "" end
	s = tostring(s):gsub("[\128-\255]", "")
	return s:match("^%s*(.-)%s*$")
end

local function record(name, category, assetId, extra)
	name = clean(name)
	if not name or #name < 2 or name == "Frame" or name == "Template" or name == "UIListLayout" then return end
	local key = name:lower()
	if harvested[key] then return end

	harvested[key] = {
		name = name,
		category = category or "General",
		assetId = clean(assetId),
		extra = clean(extra),
	}
end

local function runMegaHarvest()
	if isRunning then return end
	isRunning = true
	HarvestBtn.Text = "HARVESTING..."
	table.clear(logs)
	table.clear(harvested)

	appendLog("=== STARTING FULL MEGA HARVEST ===")

	-- 1. Sweep ReplicatedStorage.ReplicatedHair
	pcall(function()
		local hair = ReplicatedStorage:FindFirstChild("ReplicatedHair")
		if hair then
			local count = 0
			for _, h in ipairs(hair:GetChildren()) do
				count = count + 1
				local meshId = ""
				for _, d in ipairs(h:GetDescendants()) do
					if d:IsA("MeshPart") and d.MeshId ~= "" then
						meshId = d.MeshId
						break
					elseif d:IsA("SpecialMesh") and d.MeshId ~= "" then
						meshId = d.MeshId
						break
					end
				end
				record(h.Name, "Hair", meshId, "ReplicatedHair")
			end
			appendLog("[+] ReplicatedHair: " .. count .. " items")
		end
	end)

	-- 2. Sweep Shop Directory (Wings, Shoes, Skirts, Accessories, Monthly)
	pcall(function()
		local shop = pg:FindFirstChild("Shop")
		local dir = shop and shop:FindFirstChild("Directory")
		if dir then
			for _, catFolder in ipairs(dir:GetChildren()) do
				local catName = catFolder.Name
				local getAsset = catFolder:FindFirstChild("GetAssetList")
				if getAsset and getAsset:IsA("BindableFunction") then
					pcall(function()
						local items = getAsset:Invoke()
						if type(items) == "table" then
							local ct = 0
							for k, v in pairs(items) do
								ct = ct + 1
								if type(v) == "table" then
									local n = v.Name or v.Item or v.Title or (type(k) == "string" and k)
									local img = v.Image or v.Icon or v.Thumbnail or ""
									local p = v.Price or v.Cost or ""
									record(n, catName, img, "Price: " .. tostring(p))
								elseif type(v) == "string" then
									record(v, catName, "", "")
								end
							end
							appendLog("[+] GetAssetList (" .. catName .. "): " .. ct .. " items")
						end
					end)
				end

				for _, d in ipairs(catFolder:GetDescendants()) do
					if d:IsA("Frame") and d:FindFirstChild("Tryon") then
						local img = ""
						for _, ch in ipairs(d:GetDescendants()) do
							if ch:IsA("ImageLabel") and ch.Visible and ch.Image ~= "" and not ch.Image:find("3570695787") then
								img = ch.Image
								break
							end
						end
						record(d.Name, catName, img, "ShopTryOn")
					end
				end
			end
		end
	end)

	-- 3. Sweep CharacterCreation (Makeup, Hair, Faces, Outfits, Wings)
	pcall(function()
		local cc = pg:FindFirstChild("CharacterCreation")
		if cc then
			local count = 0
			for _, d in ipairs(cc:GetDescendants()) do
				if d:IsA("ImageButton") or d:IsA("ImageLabel") then
					if d.Image ~= "" and not d.Image:find("3570695787") and #d.Name > 2 then
						count = count + 1
						record(d.Name, "DressUpAsset", d.Image, d.Parent and d.Parent.Name or "")
					end
				end
			end
			appendLog("[+] CharacterCreation Visual Assets: " .. count)
		end
	end)

	-- 4. Sweep All ReplicatedStorage ModuleScripts (Item Definitions & Tables)
	pcall(function()
		local moduleCount = 0
		for _, d in ipairs(ReplicatedStorage:GetDescendants()) do
			if d:IsA("ModuleScript") then
				pcall(function()
					local data = require(d)
					if type(data) == "table" then
						for k, v in pairs(data) do
							if type(v) == "table" then
								local n = v.Name or v.Title or v.Item or (type(k) == "string" and k)
								local im = v.Image or v.Icon or v.Texture or v.Thumbnail or ""
								if n and type(n) == "string" and #n > 2 then
									moduleCount = moduleCount + 1
									record(n, "GameDatabase", im, d.Name)
								end
							elseif type(v) == "string" and #v > 2 then
								moduleCount = moduleCount + 1
								record(v, "GameDatabase", "", d.Name)
							end
						end
					end
				end)
			end
		end
		appendLog("[+] Ingested from Game Data Modules: " .. moduleCount)
	end)

	-- 5. Sweep Meshes & Models in ReplicatedStorage
	pcall(function()
		local count = 0
		for _, d in ipairs(ReplicatedStorage:GetDescendants()) do
			if d:IsA("MeshPart") or d:IsA("SpecialMesh") then
				local asset = d:IsA("MeshPart") and d.MeshId or d.MeshId
				if asset ~= "" then
					count = count + 1
					record(d.Name, "MeshModel", asset, d.Parent and d.Parent.Name or "")
				end
			end
		end
		appendLog("[+] 3D Mesh Assets: " .. count)
	end)

	local totalCount = 0
	for _ in pairs(harvested) do totalCount = totalCount + 1 end
	appendLog("=== HARVEST COMPLETE: " .. totalCount .. " TOTAL ITEMS ===")

	HarvestBtn.Text = "HARVESTED (" .. totalCount .. ")"
	isRunning = false
end

	local count = 0

		table.insert(lines, string.format(

	end
