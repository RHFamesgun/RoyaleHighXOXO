local function getGuiContainer()
	if gethui then
		local s, h = pcall(gethui)
		if s and h then return h end
	end
	local s, cg = pcall(function() return game:GetService("CoreGui") end)
	if s and cg then return cg end
	return pg
end

-- ==========================================================
-- 1. ABSOLUTE MOVEMENT UNFREEZER
-- ==========================================================
local function unfreezeControls()
	pcall(function()
		local PlayerModule = lp:FindFirstChild("PlayerScripts") and lp.PlayerScripts:FindFirstChild("PlayerModule")
		if PlayerModule then
			local Controls = require(PlayerModule)
			if Controls and Controls.GetControls then
				Controls:GetControls():Enable()
			end
		end
	end)
	pcall(function()
		ContextActionService:UnbindAction("FreezeMovement")
		ContextActionService:UnbindAction("ShopFreeze")
		ContextActionService:UnbindAction("TryOnFreeze")
		ContextActionService:UnbindAction("DisableMovement")
		ContextActionService:UnbindAction("CharacterCreationFreeze")
	end)
end

local function bindMovementGuards(char)
	if not char then return end
	local hum = char:WaitForChild("Humanoid", 4)
	local hrp = char:WaitForChild("HumanoidRootPart", 4)

	if hum then
		hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
			if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
		end)
		hum:GetPropertyChangedSignal("PlatformStand"):Connect(function()
			if hum.PlatformStand then hum.PlatformStand = false end
		end)
		if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
		if hum.PlatformStand then hum.PlatformStand = false end
	end

	if hrp then
		hrp:GetPropertyChangedSignal("Anchored"):Connect(function()
			if hrp.Anchored then hrp.Anchored = false end
		end)
		if hrp.Anchored then hrp.Anchored = false end
	end

	unfreezeControls()
end

if lp.Character then bindMovementGuards(lp.Character) end
lp.CharacterAdded:Connect(bindMovementGuards)

task.spawn(function()
	while true do
		task.wait(1)
		pcall(function()
			unfreezeControls()
			local char = lp.Character
			if char then
				local hrp = char:FindFirstChild("HumanoidRootPart")
				if hrp and hrp.Anchored then hrp.Anchored = false end
				local hum = char:FindFirstChild("Humanoid")
				if hum and hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
			end
		end)
	end
end)

-- ==========================================================
-- 2. METATABLE DELETION INTERCEPTOR & RECURSIVE CLONING
-- ==========================================================
local function deepClone(inst)
	if not inst then return nil end
	pcall(function()
		inst.Archivable = true
		for _, d in ipairs(inst:GetDescendants()) do
			d.Archivable = true
		end
	end)
	return inst:Clone()
end

pcall(function()
	if hookmetamethod and newcclosure and checkcaller then
		local rawNamecall
		rawNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
			local method = getnamecallmethod()
			if not checkcaller() and (method == "Destroy" or method == "destroy") then
				if typeof(self) == "Instance" and self:GetAttribute("CustomSavedLayer") then
					return nil
				end
			end
			return rawNamecall(self, ...)
		end))

		local rawNewindex
		rawNewindex = hookmetamethod(game, "__newindex", newcclosure(function(self, idx, val)
			if not checkcaller() and (idx == "Parent" or idx == "parent") and val == nil then
				if typeof(self) == "Instance" and self:GetAttribute("CustomSavedLayer") then
					return nil
				end
			end
			return rawNewindex(self, idx, val)
		end))
	end
end)

local savedWornLayers = {}

local standardBodyParts = {

local function reattachModelJoints(mod, char, joints)
	for _, wd in ipairs(joints) do
		local p0 = mod:FindFirstChild(wd.part0Name, true)
		local p1 = char:FindFirstChild(wd.part1Name)
		if p0 and p1 then
			if wd.className == "Motor6D" then
				local m = Instance.new("Motor6D")
				m.Part0 = p0
				m.Part1 = p1
				if wd.c0 then m.C0 = wd.c0 end
				if wd.c1 then m.C1 = wd.c1 end
				m.Parent = p0
			elseif wd.className == "WeldConstraint" then
				local wc = Instance.new("WeldConstraint")
				wc.Part0 = p0
				wc.Part1 = p1
				wc.Parent = p0
			else
				local w = Instance.new("Weld")
				w.Part0 = p0
				w.Part1 = p1
				if wd.c0 then w.C0 = wd.c0 end
				if wd.c1 then w.C1 = wd.c1 end
				w.Parent = p0
			end
		end
	end
end

local function restoreAllSavedLayers()
	local char = lp.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")

	for rawName, data in pairs(savedWornLayers) do
		local layerName = rawName .. "_Layer"
		local existing = char:FindFirstChild(layerName) or char:FindFirstChild(rawName)
		if not existing and data.backup then
			pcall(function()
				local copy = deepClone(data.backup)
				copy.Name = layerName
				copy:SetAttribute("CustomSavedLayer", true)

				if data.isAccessory and hum then
					hum:AddAccessory(copy)
				elseif data.isModel then
					reattachModelJoints(copy, char, data.joints or {})
					copy.Parent = char
				else
					copy.Parent = char
				end
			end)
		end
	end
end

local function clearAllWornLayers()
	local char = lp.Character
	for rawName, data in pairs(savedWornLayers) do
		if char then
			local layerName = rawName .. "_Layer"
			local i1 = char:FindFirstChild(layerName)
			local i2 = char:FindFirstChild(rawName)
			if i1 then
				i1:SetAttribute("CustomSavedLayer", nil)
				pcall(function() i1:Destroy() end)
			end
			if i2 then
				i2:SetAttribute("CustomSavedLayer", nil)
				pcall(function() i2:Destroy() end)
			end
		end
	end
	savedWornLayers = {}

	if char then
		for _, child in ipairs(char:GetChildren()) do
			if child:GetAttribute("CustomSavedLayer") or child.Name:find("_Layer") then
				child:SetAttribute("CustomSavedLayer", nil)
				pcall(function() child:Destroy() end)
			end
		end
	end

	-- Also reset native game accessories if remote available
	pcall(function()
		local resetRemote = ReplicatedStorage:FindFirstChild("Network") and ReplicatedStorage.Network.Events.Player.ResetAllEquippedItems
		if resetRemote then resetRemote:FireServer() end
	end)
end

task.spawn(function()
	while true do
		task.wait(0.5)
		pcall(function()
			local char = lp.Character
			if char and next(savedWornLayers) ~= nil then
				restoreAllSavedLayers()
			end
		end)
	end
end)

local seenItems = {}

local function addItem(cat, item)
	if not item.name or item.name == "" or item.name == " " then return end
	local key = item.name:lower()
	if seenItems[key] then return end
	seenItems[key] = true

	item.hasRealImage = (item.image ~= nil and item.image ~= "" and item.image:find("rbxassetid://") ~= nil)

	if not categories[cat] then categories[cat] = {} end
	table.insert(categories[cat], item)
end

local function classifyCategory(itemName, defaultCat)
	local lower = itemName:lower()
	if lower:find("halo") then
		return "Halos"
	elseif lower:find("heel") or lower:find("boot") or lower:find("shoe") or lower:find("loafer") or lower:find("platform") or lower:find("footwear") or lower:find("sandal") or lower:find("sneaker") then
		return "Heels"
	elseif lower:find("skirt") or lower:find("dress") or lower:find("tutu") then
		return "Skirts"
	elseif lower:find("bodice") or lower:find("corset") or lower:find("bustier") then
		return "Bodices"
	elseif lower:find("wing") then
		return "Wings"
	else
		return defaultCat or "Accessories"
	end
end

-- ==========================================================
-- 5. NATIVE DRESS-UP & INVENTORY EQUIPPING ENGINE
-- ==========================================================
local function equipNative(itemName, itemCat)
	local char = lp.Character
	if not char then return end

	-- Manage single-slot swap for skirts/heels
	if itemCat == "Skirts" or itemCat == "Heels" then
		for name, data in pairs(savedWornLayers) do
			if data.category == itemCat then
				local lyr = char:FindFirstChild(name .. "_Layer") or char:FindFirstChild(name)
				if lyr then
					lyr:SetAttribute("CustomSavedLayer", nil)
					pcall(function() lyr:Destroy() end)
				end
				savedWornLayers[name] = nil
			end
		end
	end

	-- 1. Try Native ReplicatedStorage Inventory.EquipItem RemoteFunction
	local invFolder = ReplicatedStorage:FindFirstChild("Inventory")
	if invFolder and invFolder:FindFirstChild("EquipItem") then
		pcall(function()
			invFolder.EquipItem:InvokeServer(itemName)
		end)
	end

	-- 2. Try Native Accessory Equip RemoteEvent
	pcall(function()
		local netAccessory = ReplicatedStorage:FindFirstChild("Network") and ReplicatedStorage.Network.Events.Player.Accessory.Equip
		if netAccessory then netAccessory:FireServer(itemName) end
	end)

	-- 3. Try CharacterCreation PromptEquip BindableEvent
	pcall(function()
		local cc = pg:FindFirstChild("CharacterCreation")
		if cc and cc:FindFirstChild("Inventory") and cc.Inventory:FindFirstChild("PromptEquip") then
			cc.Inventory.PromptEquip:Fire(itemName)
		end
	end)

	-- 4. Try TogglesNetwork Equip
	pcall(function()
		local tog = ReplicatedStorage:FindFirstChild("TogglesNetwork") and ReplicatedStorage.TogglesNetwork:FindFirstChild("Equip")
		if tog then tog:FireServer(itemName) end
	end)

	-- 5. Capture new character instance & protect as custom layer
	task.spawn(function()
		local newItem = nil
		for step = 1, 20 do
			task.wait(0.08)
			for _, child in ipairs(char:GetChildren()) do
				if (child:IsA("Accessory") or child:IsA("Model"))
				   and not standardBodyParts[child.Name]
				   and not child:GetAttribute("CustomSavedLayer") then
					local lower = child.Name:lower()
					if lower:find(itemName:lower(), 1, true) or itemName:lower():find(lower, 1, true) then
						newItem = child
						break
					end
				end
			end
			if newItem then break end
		end

		if newItem then
			pcall(function()
				newItem:SetAttribute("CustomSavedLayer", true)
				local jointsList = {}
				for _, w in ipairs(newItem:GetDescendants()) do
					if (w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D")) and w.Part1 and w.Part1:IsDescendantOf(char) then
						table.insert(jointsList, {
							part0Name = w.Part0 and w.Part0.Name or "",
							part1Name = w.Part1.Name,
							c0        = (w:IsA("Motor6D") or w:IsA("Weld")) and w.C0 or nil,
							c1        = (w:IsA("Motor6D") or w:IsA("Weld")) and w.C1 or nil,
							className = w.ClassName,
						})
					end
				end

				local backupCopy = deepClone(newItem)
				backupCopy:SetAttribute("CustomSavedLayer", true)

				savedWornLayers[itemName] = {
					backup     = backupCopy,
					joints     = jointsList,
					isAccessory = newItem:IsA("Accessory"),
					isModel    = newItem:IsA("Model"),
					category   = itemCat,
				}
			end)
		end

		restoreAllSavedLayers()
		unfreezeControls()
	end)
end

local function formatCommas(str)
	local numStr = str:gsub("[^%d]", "")
	if numStr == "" then return "" end
	local k
	while true do
		numStr, k = string.gsub(numStr, "^(-?%d+)(%d%d%d)", '%1,%2')
		if k == 0 then break end
	end
	return numStr
end

local function updateDiamonds(val)
	if val == "" then return end
	pcall(function() pg.Shop.Header.diamondamount.Text = val end)
	pcall(function() pg.HUD.Frame.Middle.DiamondsFrame.DiamondAmount.Text = val end)
	pcall(function() pg.MermaidShopDiamondCount.Frame.DiamondAmount.Text = val end)
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("TextLabel") and (d.Name:lower():find("diamondamount") or d.Name:lower():find("diamondcount")) then
			pcall(function() d.Text = val end)
		end
	end
end

local function updateLevel(val)
	if val == "" then return end
	for _, d in ipairs(pg:GetDescendants()) do
		if d:IsA("TextLabel") and (d.Name:lower():find("levelnumber") or (d.Name:lower() == "level" and tonumber(d.Text:gsub("[^%d]", "")))) then
			pcall(function() d.Text = val end)
		end
	end
end

local function updateCandy(val)
	if val == "" then return end
	pcall(function() pg.Shop.Header.candyamount.Text = val end)
	pcall(function() pg.HUD.Frame.HalloweenCandy.CandyAmount.Text = val end)
end

local function updateCash(val)
	if val == "" then return end
	pcall(function() pg.HUD.Frame.Middle.CloudCashStuff.CloudCash.Text = val end)
end

		if activeSpoofs["Candy"]      then updateCandy(activeSpoofs["Candy"])          end
		if activeSpoofs["Cloud Cash"] then updateCash(activeSpoofs["Cloud Cash"])      end
	end
end)

local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos  = Main.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UIS.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)
