-- ROYALE HIGH EXACT EQUIP ACTION SNIFFER
-- ==========================================================
local Players = game:GetService("Players")
local rep = game:GetService("ReplicatedStorage")
local lp = Players.LocalPlayer

local captured = {}
local function record(text)
    table.insert(captured, text)
    print("[SNIFFER] " .. text)
end

record("=== SNIFFER ACTIVE: GO PUT ON ANY ITEM IN-GAME NOW ===")

-- Hook RemoteEvents & RemoteFunctions
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if method == "FireServer" or method == "InvokeServer" then
        local fullName = tostring(self:GetFullName())
        local nameLower = fullName:lower()
        if nameLower:find("equip") or nameLower:find("wear") or nameLower:find("outfit") 
            or nameLower:find("accessory") or nameLower:find("custom") or nameLower:find("item")
            or nameLower:find("closet") or nameLower:find("dressup") then
            
            local argDump = {}
            for i, arg in ipairs(args) do
                argDump[i] = typeof(arg) .. ": " .. tostring(arg)
            end
            local line = string.format("[%s] %s -> Args: (%s)", method, fullName, table.concat(argDump, ", "))
            record(line)
            
            if setclipboard then
                setclipboard(table.concat(captured, "\n"))
            end
        end
    end
    return oldNamecall(self, ...)
end))

-- Also watch Character for newly attached objects
if lp.Character then
    lp.Character.ChildAdded:Connect(function(child)
        if child:IsA("Accessory") or child:IsA("Model") then
            record("[CHAR NEW ITEM] Class: " .. child.ClassName .. " | Name: " .. child.Name)
            if setclipboard then
                setclipboard(table.concat(captured, "\n"))
            end
        end
    end)
end

task.delay(30, function()
    record("=== SNIFFER FINISHED (30s) ===")
    local full = table.concat(captured, "\n")
    if setclipboard then setclipboard(full) end
end)
