local function disableFreezeOnlyConnections()
    if getconnections and previewItemRemote then
        pcall(function()
            for _, conn in ipairs(getconnections(previewItemRemote.OnClientEvent)) do
                pcall(function()
                    local fn = conn.Function
                    local isFreeze = false
                    if getconstants and fn then
                        for _, c in ipairs(getconstants(fn)) do
                            if c == "ExitTryOnNonWings" or c == "PrimaryPart" or c == "Anchored" then
                                isFreeze = true
                                break
                            end
                        end
                    end
                    if isFreeze then
                        conn:Disable()
                    end
                end)
            end
        end)
    end
end

local function hideTryOnOverlays()
    pcall(function()
        local hud = pg:FindFirstChild("HUD")
        if hud then
            for _, desc in ipairs(hud:GetDescendants()) do
                if desc:IsA("GuiObject") and (desc.Name:find("ExitTryOn") or desc.Name == "TryOn") then
                    desc.Visible = false
                end
            end
        end
        local shop = pg:FindFirstChild("Shop")
        if shop then
            for _, desc in ipairs(shop:GetDescendants()) do
                if desc:IsA("GuiObject") and desc.Name == "ExitTryOnNonWings" then
                    desc.Visible = false
                end
            end
        end
    end)
end

local function releasePlayerMovement()
    local char = lp.Character
    if not char then return end

    -- 1. Unanchor all character parts and primary part
    local pp = char.PrimaryPart or char:FindFirstChild("HumanoidRootPart")
    if pp and pp.Anchored then pp.Anchored = false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if hrp and hrp.Anchored then hrp.Anchored = false end

    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Anchored then
            part.Anchored = false
        end
    end

    -- 2. Humanoid walkspeed, jumping, sitting, platformstand
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        if hum.PlatformStand then hum.PlatformStand = false end
        if hum.Sit then hum.Sit = false end
        if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
        if hum.JumpPower and hum.JumpPower < 50 then hum.JumpPower = 50 end
        if hum.JumpHeight and hum.JumpHeight < 7.2 then hum.JumpHeight = 7.2 end

        -- Stop forced idle or freeze animations
        for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
            local tn = track.Animation and track.Animation.Name:lower() or ""
            if tn:find("idle") or tn:find("stand") or tn:find("freeze") then
                pcall(function() track:Stop(0.05) end)
            end
        end
    end

    -- 3. Unbind all 10 known ContextActionService freeze actions in Royale High
    local freezeActions = {
        "ControlsDisabled", "DisableControlsCutscene", "IntroScriptDisableControls",
        "RH3CutsceneControlsDisabled", "FreezeMovement", "ShopFreeze", "TryOnFreeze",
        "DisableMovement", "CharacterCreationFreeze", "TryOnControl"
    }
    for _, act in ipairs(freezeActions) do
        pcall(function() ContextActionService:UnbindAction(act) end)
    end

    -- 4. Enable Roblox PlayerModule controls
    pcall(function()
        local pm = lp.PlayerScripts:FindFirstChild("PlayerModule")
        if pm then
            local mod = require(pm)
            if mod and mod.GetControls then
                local ctrl = mod:GetControls()
                if ctrl and ctrl.Enable then
                    ctrl:Enable()
                end
            end
        end
    end)

    -- 5. Terminate Royale High's internal try-on anchoring while-loop by firing ExitPreview
    if exitPreviewRemote then
        pcall(function() exitPreviewRemote:FireServer() end)
    end

    -- 6. Hide all TryOn billboard HUD elements
    pcall(hideTryOnOverlays)
end

    local pp = char.PrimaryPart or char:FindFirstChild("HumanoidRootPart")
    if pp and pp.Anchored then pp.Anchored = false end

    if hum then
        if hum.PlatformStand then hum.PlatformStand = false end
        if hum.Sit then hum.Sit = false end
        if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
    end
end

local function deepClone(inst)
    if not inst then return nil end
    local ok, result = pcall(function()
        inst.Archivable = true
        for _, d in ipairs(inst:GetDescendants()) do
            d.Archivable = true
        end
        return inst:Clone()
    end)
    return ok and result or nil
end

local function getEquippedFolders()
    local list = {}
    local es = workspace:FindFirstChild("EquippedStorage")
    local char = lp.Character
    local categoriesList = {"Heels", "Skirts", "Accessories", "Wings", "Bodices", "Halos"}

    if es then
        for _, cat in ipairs(categoriesList) do
            local cf = es:FindFirstChild(cat)
            local pf = cf and (cf:FindFirstChild(lp.Name) or cf:FindFirstChild(tostring(lp.UserId)))
            if pf then
                table.insert(list, {folder = pf, category = cat})
            end
        end
    end

    if char then
        local wi = char:FindFirstChild("WornItems")
        if wi then
            for _, cat in ipairs(categoriesList) do
                local cf = wi:FindFirstChild(cat)
                local pf = cf and (cf:FindFirstChild(lp.Name) or cf:FindFirstChild(tostring(lp.UserId)))
                if pf then
                    table.insert(list, {folder = pf, category = cat})
                end
            end
        end
    end

    return list
end

-- Strictly identify base avatar body parts and hair to prevent hair duplication
local function isBaseAvatarObject(inst)
    if not inst then return true end
    if defaultAvatarInstances[inst] then return true end
    if inst:GetAttribute("RH_BakedLayer") then return true end
    if inst.Name == "Humanoid" or inst.Name == "Animate" or inst.Name == "Head" then return true end
    if standardBodyParts[inst.Name] then return true end

    -- Check if it's avatar hair
    if inst:IsA("Accessory") then
        local handle = inst:FindFirstChild("Handle")
        if handle then
            if handle:FindFirstChild("HairAttachment") then return true end
        end
        local ln = inst.Name:lower()
        if ln:find("hair") or ln:find("wig") or ln:find("bangs") or ln:find("curl")
           or ln:find("ponytail") or ln:find("pigtail") or ln:find("afro") or ln:find("braid")
           or ln:find("locs") or ln:find("bun") or ln:find("fade") or ln:find("extension") then
            return true
        end
    end

    if inst.Name == "Handle" and inst.Parent and inst.Parent:IsA("Accessory") then
        if inst:FindFirstChild("HairAttachment") then return true end
    end

    return false
end

-- Find all actual Royale High clothing pieces attached to avatar
local function findRoyaleHighItemsOnAvatar()
    local char = lp.Character
    if not char then return {} end
    local itemsFound = {}
    local seen = {}

    local function addItem(inst)
        if not inst or seen[inst] then return end
        if isBaseAvatarObject(inst) then return end
        seen[inst] = true
        table.insert(itemsFound, inst)
    end

    -- 1. Scan EquippedStorage & WornItems folders
    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            addItem(child)
        end
    end

    -- 2. Scan Character hierarchy for custom models/folders/items
    for _, d in ipairs(char:GetChildren()) do
        if not isBaseAvatarObject(d) and d.Name ~= "RH_BakedWardrobe" then
            if d:IsA("Model") or d:IsA("Folder") or d:IsA("Accessory") or d:IsA("MeshPart") then
                addItem(d)
            end
        end
    end

    -- 3. Scan welds connected to character rig limbs
    local charLimbs = {}
    for _, limb in ipairs(char:GetChildren()) do
        if limb:IsA("BasePart") and standardBodyParts[limb.Name] then
            charLimbs[limb] = true
        end
    end

    local searchRoots = { char }
    local es = workspace:FindFirstChild("EquippedStorage")
    if es then table.insert(searchRoots, es) end

    for _, root in ipairs(searchRoots) do
        for _, w in ipairs(root:GetDescendants()) do
            if (w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D")) then
                local otherPart = nil
                if charLimbs[w.Part0] and w.Part1 and not charLimbs[w.Part1] then
                    otherPart = w.Part1
                elseif charLimbs[w.Part1] and w.Part0 and not charLimbs[w.Part0] then
                    otherPart = w.Part0
                end

                if otherPart and not isBaseAvatarObject(otherPart) then
                    local container = otherPart
                    while container.Parent
                          and container.Parent ~= char
                          and container.Parent ~= workspace
                          and container.Parent ~= es
                          and not container.Parent:IsA("Workspace") do
                        if container.Parent.Name == lp.Name or container.Parent.Name == tostring(lp.UserId) then
                            break
                        end
                        container = container.Parent
                    end
                    addItem(container)
                end
            end
        end
    end

    return itemsFound
end

-- 1:1 Single-Clone Layer Baker Engine: Clones exactly 1 instance per slot, matching relative CFrames & welds 1:1
local function bakeItemToAvatar(sourceItem, category, customSlotKey, itemData)
    if not sourceItem then return nil end
    local char = lp.Character
    if not char then return nil end

    local cat = category or "Accessories"
    local slotKey = nil
    if customSlotKey then
        slotKey = customSlotKey
    elseif cat == "Heels" or cat == "Skirts" or cat == "Bodices" or cat == "Wings" or cat == "Halos" then
        slotKey = cat
    else
        -- Multi-Accessory Engine: Every accessory gets its own distinct slot key so you can wear unlimited accessories!
        local accName = (itemData and itemData.name) or sourceItem.Name
        if accName == "Accessory" or accName == "Model" or accName == "Folder" or accName == "Part" then
            local sub = sourceItem:FindFirstChildWhichIsA("SpecialMesh", true) or sourceItem:FindFirstChildWhichIsA("MeshPart", true)
            if sub and sub.Name ~= "MeshPart" and sub.Name ~= "Mesh" then
                accName = sub.Name
            end
        end
        slotKey = "Acc_" .. accName:gsub("%s+", "")
    end

    -- Dedicated container under Character
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
    if not bakedFolder then
        bakedFolder = Instance.new("Folder")
        bakedFolder.Name = "RH_BakedWardrobe"
        bakedFolder.Parent = char
    end

    -- Strict single-clone per slot: destroy old clone for THIS specific slot/accessory
    local existing = bakedFolder:FindFirstChild(slotKey)
    if existing then
        pcall(function() existing:Destroy() end)
    end
    bakedLayersBackup[slotKey] = nil

    -- Deep clone source item
    local ok, clone = pcall(function()
        sourceItem.Archivable = true
        for _, d in ipairs(sourceItem:GetDescendants()) do d.Archivable = true end
        return sourceItem:Clone()
    end)
    if not ok or not clone then return nil end

    local imgToStore = (itemData and itemData.img and itemData.img ~= "") and itemData.img or ""
    if imgToStore == "" and findItemImage then
        imgToStore = findItemImage(itemData and itemData.name or sourceItem.Name, cat)
    end

    clone.Name = slotKey
    clone:SetAttribute("RH_BakedLayer", true)
    clone:SetAttribute("RH_Category", cat)
    clone:SetAttribute("RH_SlotKey", slotKey)
    clone:SetAttribute("RH_ItemName", (itemData and itemData.name) or sourceItem.Name)
    clone:SetAttribute("RH_Image", imgToStore)
    clone:SetAttribute("RH_Price", (itemData and itemData.price) or "")

    -- Clean old broken joints in clone
    for _, j in ipairs(clone:GetDescendants()) do
        if j:IsA("JointInstance") or j:IsA("WeldConstraint") then
            j:Destroy()
        end
    end

    local zeroPhys = PhysicalProperties.new(0.01, 0, 0, 0, 0)
    local function prepPart(p)
        p.CanCollide = false
        p.CanTouch   = false
        p.CanQuery   = false
        p.Anchored   = false
        p.Massless   = true
        pcall(function() p.CustomPhysicalProperties = zeroPhys end)
    end

    -- 1:1 Unified Rigid Movement Engine:
    -- Binds all sub-parts of an item to ONE central root part so the entire dress/shoe moves as ONE unified piece without separating!
    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("LowerTorso")

    if cat == "Heels" and hrp then
        -- Heels: Unified Left Boot and Unified Right Boot
        local leftLimb = char:FindFirstChild("LeftLowerLeg") or char:FindFirstChild("LeftUpperLeg") or char:FindFirstChild("Left Leg") or hrp
        local rightLimb = char:FindFirstChild("RightLowerLeg") or char:FindFirstChild("RightUpperLeg") or char:FindFirstChild("Right Leg") or hrp

        local leftParts = {}
        local rightParts = {}

        if clone:IsA("BasePart") then
            if clone.Position.X < hrp.Position.X then table.insert(leftParts, clone) else table.insert(rightParts, clone) end
        else
            for _, p in ipairs(clone:GetDescendants()) do
                if p:IsA("BasePart") then
                    if p.Position.X < hrp.Position.X then table.insert(leftParts, p) else table.insert(rightParts, p) end
                end
            end
        end

        local function rigBoot(partsList, targetLimb)
            if #partsList == 0 then return end
            -- Find largest part as root
            local root = partsList[1]
            local maxVol = 0
            for _, p in ipairs(partsList) do
                prepPart(p)
                local vol = p.Size.X * p.Size.Y * p.Size.Z
                if vol > maxVol then maxVol = vol; root = p end
            end

            -- Match root to limb 1:1
            local relOffset = targetLimb.CFrame:ToObjectSpace(root.CFrame)
            root.CFrame = targetLimb.CFrame * relOffset

            local wc = Instance.new("WeldConstraint")
            wc.Name = "RH_BootRootWeld"
            wc.Part0 = targetLimb
            wc.Part1 = root
            wc.Parent = root

            -- Weld all other parts rigidly to boot root so the boot never splits apart
            for _, p in ipairs(partsList) do
                if p ~= root then
                    local pRel = root.CFrame:ToObjectSpace(p.CFrame)
                    p.CFrame = root.CFrame * pRel

                    local subWc = Instance.new("WeldConstraint")
                    subWc.Name = "RH_BootSubWeld"
                    subWc.Part0 = root
                    subWc.Part1 = p
                    subWc.Parent = p
                end
            end
        end

        rigBoot(leftParts, leftLimb)
        rigBoot(rightParts, rightLimb)
    else
        -- Skirts, Bodices, Wings, Halos, and Accessories: Unified Root Weld
        local targetLimb = nil
        if cat == "Skirts" then
            targetLimb = char:FindFirstChild("LowerTorso") or char:FindFirstChild("Torso") or hrp
        elseif cat == "Bodices" then
            targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or hrp
        elseif cat == "Wings" then
            targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or hrp
        elseif cat == "Halos" then
            targetLimb = char:FindFirstChild("Head") or hrp
        else
            -- Accessories: find nearest or attached limb
            targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Head") or char:FindFirstChild("LowerTorso") or hrp
        end

        local allParts = {}
        if clone:IsA("BasePart") then
            table.insert(allParts, clone)
        else
            for _, p in ipairs(clone:GetDescendants()) do
                if p:IsA("BasePart") then table.insert(allParts, p) end
            end
        end

        if #allParts > 0 and targetLimb then
            -- Find primary root part
            local root = clone.PrimaryPart or clone:FindFirstChild("Middle") or clone:FindFirstChild("Center") or clone:FindFirstChild("Handle") or allParts[1]
            prepPart(root)

            local relOffset = targetLimb.CFrame:ToObjectSpace(root.CFrame)
            root.CFrame = targetLimb.CFrame * relOffset

            local wc = Instance.new("WeldConstraint")
            wc.Name = "RH_UnifiedRootWeld"
            wc.Part0 = targetLimb
            wc.Part1 = root
            wc.Parent = root

            -- Weld all sub-parts to root so the entire item moves together 1:1!
            for _, p in ipairs(allParts) do
                if p ~= root then
                    prepPart(p)
                    local pRel = root.CFrame:ToObjectSpace(p.CFrame)
                    p.CFrame = root.CFrame * pRel

                    local subWc = Instance.new("WeldConstraint")
                    subWc.Name = "RH_UnifiedSubWeld"
                    subWc.Part0 = root
                    subWc.Part1 = p
                    subWc.Parent = p
                end
            end
        end
    end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if clone:IsA("Accessory") and hum then
        hum:AddAccessory(clone)
    else
        clone.Parent = bakedFolder
    end

    if cat == "Heels" then
        for _, lName in ipairs(legPartNames) do
            local limb = char:FindFirstChild(lName)
            if limb and limb:IsA("BasePart") then
                limb.Transparency = 1
            end
        end
    end

    bakedLayersBackup[slotKey] = deepClone(clone)
    return clone
end

-- Lock avatar's current look into single-layer baked slots
local isBakingDebounce = false
local function lockCurrentAvatarLook(catHint, itemData)
    if isBakingDebounce then return 0 end

    local items = findRoyaleHighItemsOnAvatar()
    if #items == 0 then return 0 end

    isBakingDebounce = true
    local count = 0
    for _, item in ipairs(items) do
        local cat = catHint or "Accessories"
        if item.Parent and item.Parent.Parent and item.Parent.Parent.Name == "EquippedStorage" then
            cat = item.Parent.Name
        end
        local b = bakeItemToAvatar(item, cat, nil, itemData)
        if b then count = count + 1 end
    end

    task.delay(0.3, function()
        isBakingDebounce = false
    end)
    return count
end

-- Unequip a single worn slot
local function unequipWornItem(item)
    if not item then return end
    local char = lp.Character
    local itemName = item.name or ""
    local slotKey  = item.slotKey or ""
    local cat      = item.cat or ""

    -- 1. Remove from RH_BakedWardrobe
    if item.bakedInst and item.bakedInst.Parent then
        pcall(function() item.bakedInst:Destroy() end)
    end
    if char then
        local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
        if bakedFolder then
            if slotKey ~= "" then
                local b = bakedFolder:FindFirstChild(slotKey)
                if b then pcall(function() b:Destroy() end) end
            end
            for _, c in ipairs(bakedFolder:GetChildren()) do
                if c.Name == slotKey or c.Name == itemName or c:GetAttribute("RH_ItemName") == itemName or c:GetAttribute("RH_SlotKey") == slotKey then
                    pcall(function() c:Destroy() end)
                end
            end
        end
    end

    -- 2. Clear backup table
    if slotKey ~= "" and bakedLayersBackup[slotKey] then
        bakedLayersBackup[slotKey] = nil
    end
    if cat ~= "" and (cat == "Heels" or cat == "Skirts" or cat == "Bodices" or cat == "Wings" or cat == "Halos") then
        bakedLayersBackup[cat] = nil
    end

    -- 3. Destroy live try-on instance if present
    if item.liveInst and item.liveInst.Parent then
        pcall(function() item.liveInst:Destroy() end)
    end

    -- 4. Purge from EquippedStorage and WornItems folders
    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            if child.Name == itemName or child.Name:lower():find(itemName:lower(), 1, true) or itemName:lower():find(child.Name:lower(), 1, true) then
                pcall(function() child:Destroy() end)
            end
        end
    end

    -- 5. Purge any matching models/folders on Character
    if char then
        for _, child in ipairs(char:GetChildren()) do
            if (child:GetAttribute("RH_BakedLayer") or child.Name:find("RH_Baked") or child.Name == slotKey or child.Name == itemName)
               and child.Name ~= "RH_BakedWardrobe" then
                pcall(function() child:Destroy() end)
            end
        end
    end

    -- 6. Category specifics: restore legs transparency for heels
    if cat == "Heels" and char then
        for _, lName in ipairs(legPartNames) do
            local limb = char:FindFirstChild(lName)
            if limb and limb:IsA("BasePart") then
                limb.Transparency = 0
            end
        end
    end

    -- 7. Destroy custom wings if Wings
    if cat == "Wings" and char then
        for _, n in ipairs({"Wings_TRYON", "CustomWingRight", "CustomWingLeft"}) do
            local w = char:FindFirstChild(n)
            if w then pcall(function() w:Destroy() end) end
        end
    end

    -- 8. Fire ExitPreview to terminate server try-on
    if exitPreviewRemote then
        pcall(function() exitPreviewRemote:FireServer() end)
    end

    -- 9. Release movement and refresh UI
    releasePlayerMovement()
    updateTabCounts()
    if refreshGridRef then
        refreshGridRef("Worn")
    end
end

-- Get list of all currently worn/baked items for Worn section
local function getWornItemsList()
    local wornList = {}
    local seen = {}
    local char = lp.Character
    if not char then return wornList end

    -- 1. Scan RH_BakedWardrobe (our primary baked layers)
    local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
    if bakedFolder then
        for _, clone in ipairs(bakedFolder:GetChildren()) do
            local itemName = clone:GetAttribute("RH_ItemName") or clone.Name:gsub("^RH_Baked_", "")
            local cat = clone:GetAttribute("RH_Category") or "Accessories"
            local slotKey = clone:GetAttribute("RH_SlotKey") or clone.Name
            local img = clone:GetAttribute("RH_Image") or ""
            local price = clone:GetAttribute("RH_Price") or ""

            if (not img or img == "") and findItemImage then
                img = findItemImage(itemName, cat)
            end

            local cleanKey = (cat .. "_" .. itemName):lower():gsub("%s+", "")
            if not seen[cleanKey] and not seen[slotKey:lower()] then
                seen[cleanKey] = true
                seen[slotKey:lower()] = true
                table.insert(wornList, {
                    name = itemName,
                    cat = cat,
                    slotKey = slotKey,
                    img = img,
                    price = price ~= "" and price or "Equipped",
                    isWorn = true,
                    bakedInst = clone
                })
            end
        end
    end

    -- 2. Scan EquippedStorage & WornItems for any live items not yet baked
    for _, info in ipairs(getEquippedFolders()) do
        for _, child in ipairs(info.folder:GetChildren()) do
            if not isBaseAvatarObject(child) then
                local cleanKey = (info.category .. "_" .. child.Name):lower():gsub("%s+", "")
                if not seen[cleanKey] and not seen[child.Name:lower()] then
                    seen[cleanKey] = true
                    local img = findItemImage and findItemImage(child.Name, info.category) or ""
                    table.insert(wornList, {
                        name = child.Name,
                        cat = info.category,
                        slotKey = info.category .. "_" .. child.Name,
                        img = img,
                        price = "Live Try-On",
                        isWorn = true,
                        liveInst = child
                    })
                end
            end
        end
    end

    -- 3. Custom wings
    local wingClone = char:FindFirstChild("Wings_TRYON") or char:FindFirstChild("CustomWingRight") or char:FindFirstChild("CustomWingLeft")
    if wingClone and not seen["wings"] then
        seen["wings"] = true
        local wName = wingClone:GetAttribute("RH_ItemName") or "Custom Wings"
        local img = findItemImage and findItemImage(wName, "Wings") or ""
        table.insert(wornList, {
            name = wName,
            cat = "Wings",
            slotKey = "Wings",
            img = img,
            price = "Equipped Wings",
            isWorn = true,
            bakedInst = wingClone
        })
    end

    return wornList
end

local function addOrUpdateCatalogItem(cat, alias, assetName, price, img, isServerData)
    local key = (assetName or alias):lower()
    local existing = catalogItemByKey[key]

    local pStr = "N/A"
    if type(price) == "number" then
        pStr = string.format("%d", price):reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
    elseif type(price) == "string" and price ~= "" then
        pStr = price
    end

    local imgStr = ""
    if type(img) == "string" and img ~= "" then
        if img:find("rbxassetid://") or img:find("http") then
            imgStr = img
        elseif tonumber(img) then
            imgStr = "rbxassetid://" .. img
        end
    elseif type(img) == "number" and img > 0 then
        imgStr = "rbxassetid://" .. tostring(img)
    end

    if existing then
        if isServerData and imgStr ~= "" then
            existing.img = imgStr
            existing.isServerImg = true
        elseif (not existing.img or existing.img == "") and imgStr ~= "" then
            existing.img = imgStr
        end
        if pStr ~= "N/A" and (existing.price == "N/A" or isServerData) then
            existing.price = pStr
        end
        return
    end

    local itemObj = {
        name        = alias or assetName,
        assetName   = assetName or alias,
        price       = pStr,
        img         = imgStr,
        cat         = cat,
        isServerImg = (isServerData and imgStr ~= "")
    }
    catalogItemByKey[key] = itemObj
    if alias then
        catalogItemByKey[alias:lower()] = itemObj
        catalogItemByKey[alias:lower():gsub("%s+", "")] = itemObj
    end
    if assetName then
        catalogItemByKey[assetName:lower()] = itemObj
        catalogItemByKey[assetName:lower():gsub("%s+", "")] = itemObj
    end
    if categories[cat] then
        table.insert(categories[cat], itemObj)
    end
end

-- Fast smart catalog thumbnail lookup
findItemImage = function(name, cat)
    if not name or name == "" then return "" end
    local clean = name:lower():gsub("%s+", ""):gsub("[^%w]", "")

    -- 1. Direct key
    local direct = catalogItemByKey[clean] or catalogItemByKey[name:lower()]
    if direct and direct.img and direct.img ~= "" then
        return direct.img
    end

    -- 2. Scan category list if provided
    local targetList = cat and categories[cat]
    if targetList then
        for _, it in ipairs(targetList) do
            local itClean = it.name:lower():gsub("%s+", ""):gsub("[^%w]", "")
            local itAssetClean = (it.assetName or ""):lower():gsub("%s+", ""):gsub("[^%w]", "")
            if itClean == clean or itAssetClean == clean or clean:find(itClean, 1, true) or itClean:find(clean, 1, true) then
                if it.img and it.img ~= "" then return it.img end
            end
        end
    end

    -- 3. Scan all catalog items
    for _, it in pairs(catalogItemByKey) do
        local itClean = it.name:lower():gsub("%s+", ""):gsub("[^%w]", "")
        local itAssetClean = (it.assetName or ""):lower():gsub("%s+", ""):gsub("[^%w]", "")
        if itClean == clean or itAssetClean == clean or clean:find(itClean, 1, true) or itClean:find(clean, 1, true) then
            if it.img and it.img ~= "" then return it.img end
        end
    end

    return ""
end

-- Live Server Synchronizer
local function syncWithLiveServer()
    local net = ReplicatedStorage:FindFirstChild("Network")
    local fn  = net and net:FindFirstChild("Functions") and net.Functions:FindFirstChild("Gui")
    if not fn then return end

    -- 1. Heels
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("HeelMorphs") and fn.HeelMorphs:FindFirstChild("List") then
                local heels = fn.HeelMorphs.List:InvokeServer()
                if type(heels) == "table" then
                    for assetName, data in pairs(heels) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Heels", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Heels") end
                end
            end
        end)
    end)

    -- 2. Skirts
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Skirts") and fn.Skirts:FindFirstChild("List") then
                local skirts = fn.Skirts.List:InvokeServer()
                if type(skirts) == "table" then
                    for assetName, data in pairs(skirts) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Skirts", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Skirts") end
                end
            end
        end)
    end)

    -- 3. Accessories, Bodices, & Halos
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Accessories") and fn.Accessories:FindFirstChild("List") then
                local accs = fn.Accessories.List:InvokeServer()
                if type(accs) == "table" then
                    for assetName, data in pairs(accs) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        local sub = (data.Type or data.Category or ""):lower()
                        local aliasL = alias:lower()
                        local assetL = assetName:lower()

                        if aliasL:find("halo") or assetL:find("halo") then
                            addOrUpdateCatalogItem("Halos", alias, assetName, data.Price or "Special", img, true)
                        elseif aliasL:find("bodice") or aliasL:find("corset") or aliasL:find("jacket") or sub:find("bodice") or sub:find("torso") then
                            addOrUpdateCatalogItem("Bodices", alias, assetName, data.Price, img, true)
                        else
                            addOrUpdateCatalogItem("Accessories", alias, assetName, data.Price, img, true)
                        end
                    end
                    updateTabCounts()
                    if refreshGridRef then
                        refreshGridRef("Accessories")
                        refreshGridRef("Bodices")
                        refreshGridRef("Halos")
                    end
                end
            end
        end)
    end)

    -- 4. Wings
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Wings") and fn.Wings:FindFirstChild("List") then
                local wings = fn.Wings.List:InvokeServer()
                if type(wings) == "table" then
                    for assetName, data in pairs(wings) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Wings", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Wings") end
                end
            end
        end)
    end)
end

local function findShopTryonButton(itemName)
    local shop = pg:FindFirstChild("Shop")
    if not shop then return nil end
    local dir = shop:FindFirstChild("Directory")
    if not dir then return nil end

    local targetLower = itemName:lower()
    for _, folder in ipairs(dir:GetChildren()) do
        local mid = folder:FindFirstChild("MiddlePanel")
        local sf = mid and mid:FindFirstChild("ScrollingFrame")
        if sf then
            for _, itemFrame in ipairs(sf:GetChildren()) do
                if itemFrame:IsA("Frame") then
                    local tryon = itemFrame:FindFirstChild("Tryon")
                    if tryon and tryon.Visible then
                        local fName = itemFrame.Name:lower()
                        local nLabel = itemFrame:FindFirstChild("name")
                        local lText = (nLabel and nLabel:IsA("TextLabel")) and nLabel.Text:lower() or ""
                        if fName == targetLower or lText == targetLower
                           or fName:find(targetLower, 1, true) or targetLower:find(fName, 1, true)
                           or (lText ~= "" and lText:find(targetLower, 1, true)) then
                            return tryon
                        end
                    end
                end
            end
        end
    end
    return nil
end

local function clickButtonVIM(btn)
    if not btn then return false end
    local ok = pcall(function()
        local pos = btn.AbsolutePosition
        local sz  = btn.AbsoluteSize
        local cx  = pos.X + sz.X / 2
        local cy  = pos.Y + sz.Y / 2
        if VIM then
            VIM:SendMouseButtonEvent(cx, cy, 0, true, game, 1)
            task.wait(0.04)
            VIM:SendMouseButtonEvent(cx, cy, 0, false, game, 1)
        elseif VU then
            VU:Button1Down(Vector2.new(cx, cy), workspace.CurrentCamera)
            task.wait(0.04)
            VU:Button1Up(Vector2.new(cx, cy), workspace.CurrentCamera)
        end
    end)
    return ok
end

local function openItemColorAndToggles(item)
    if not item then return false end
    local assetName = item.assetName or item.name
    local opened = false

    -- 1. Try official Royale High TogglesLocalModule (for items with toggles)
    pcall(function()
        local cc = pg:FindFirstChild("CharacterCreation")
        local togglesMod = cc and cc:FindFirstChild("TogglesLocalModule", true)
        local ptf = (cc and cc:FindFirstChild("PreviewTogglesFrame", true))
                 or (pg:FindFirstChild("Shop") and pg.Shop:FindFirstChild("PreviewTogglesFrame", true))
                 or (pg:FindFirstChild("HUD") and pg.HUD:FindFirstChild("PreviewTogglesFrame", true))

        if togglesMod and ptf then
            local mod = require(togglesMod)
            if mod and mod.HasToggles and mod:HasToggles(assetName) then
                ptf.Visible = true
                mod:OpenToggleOptions(assetName, ptf)
                opened = true
            end
        end
    end)

    -- 2. Open Royale High official ColorsGui with 2-color palette & skin protection
    pcall(function()
        local cg = pg:FindFirstChild("ColorsGui")
        if not cg then return end
        cg.Enabled = true

        local co = cg:FindFirstChild("ColorOptions")
        if not co then return end
        co.Visible = true

        local colorMod = require(co:WaitForChild("ColorOptionsModule"))
        if colorMod and colorMod.Appear then
            local function onColorPicked(chosenColor, colorGroupIndex, multiOptions)
                local c3 = nil
                if typeof(chosenColor) == "Color3" then
                    c3 = chosenColor
                elseif typeof(chosenColor) == "table" then
                    if chosenColor.r and chosenColor.g and chosenColor.b then
                        c3 = Color3.new(chosenColor.r, chosenColor.g, chosenColor.b)
                    elseif chosenColor[1] and chosenColor[2] and chosenColor[3] then
                        c3 = Color3.new(chosenColor[1], chosenColor[2], chosenColor[3])
                    end
                end

                if not c3 then return end
                local group = (type(colorGroupIndex) == "number") and colorGroupIndex or 1

                local char = lp.Character
                if not char then return end

                local slotKey = (item.cat == "Heels" or item.cat == "Skirts" or item.cat == "Bodices" or item.cat == "Wings" or item.cat == "Halos")
                                and item.cat
                                or ("Acc_" .. (item.assetName or item.name):gsub("%s+", ""))

                -- Separate Color 1 (main body/fabric) and Color 2 (trim/lace/ribbon/secondary)
                local function partMatchesColorGroup(pName, pParentName)
                    local n = (pName .. " " .. pParentName):lower()
                    local isGroup2 = n:find("2") or n:find("trim") or n:find("lace") or n:find("accent")
                                  or n:find("ribbon") or n:find("bow") or n:find("secondary") or n:find("buckle") or n:find("detail")
                    if group == 2 then
                        return isGroup2
                    else
                        return not isGroup2 or n:find("1") or n:find("main") or n:find("base")
                    end
                end

                -- STRICT SKIN PROTECTION: ONLY color parts inside the target item model! NEVER avatar limbs!
                local function applyColorToModelParts(m)
                    if not m then return end
                    for _, p in ipairs(m:GetDescendants()) do
                        if p:IsA("BasePart") and not standardBodyParts[p.Name] and p.Name ~= "HumanoidRootPart" and p.Name ~= "Head" then
                            if partMatchesColorGroup(p.Name, p.Parent.Name) then
                                p.Color = c3
                            end
                        elseif p:IsA("SpecialMesh") then
                            if partMatchesColorGroup(p.Name, p.Parent.Name) then
                                pcall(function() p.VertexColor = Vector3.new(c3.R, c3.G, c3.B) end)
                            end
                        end
                    end
                end

                -- Recolor baked item in RH_BakedWardrobe
                local bakedFolder = char:FindFirstChild("RH_BakedWardrobe")
                if bakedFolder then
                    local bakedModel = bakedFolder:FindFirstChild(slotKey)
                    if bakedModel then
                        applyColorToModelParts(bakedModel)
                    end
                end

                -- Recolor in EquippedStorage
                for _, info in ipairs(getEquippedFolders()) do
                    for _, child in ipairs(info.folder:GetChildren()) do
                        if child.Name == item.name or child.Name == (item.assetName or item.name) then
                            applyColorToModelParts(child)
                        end
                    end
                end

                -- Sync with Royale High server recolor remote
                pcall(function()
                    local inv = ReplicatedStorage:FindFirstChild("Inventory")
                    local eq = inv and inv:FindFirstChild("EquipItem")
                    if eq and eq:IsA("RemoteFunction") then
                        eq:InvokeServer(item.assetName or item.name, item.cat, chosenColor, group, true)
                    end
                end)
            end

            -- Open ColorOptionsModule with 2 Color Groups (Color 1 & Color 2 tabs)
            colorMod:SetItemToUnequip(item.assetName or item.name, item.cat)
            pcall(function() colorMod:RememberSelectionKey(item.assetName or item.name) end)
            colorMod:Appear(onColorPicked, true, false, "Generic", 2, false)
            pcall(function() colorMod:SetNames("Color 1", "Color 2") end)
            opened = true
        end
    end)

    return opened
end

local function equipItem(item)
    local char = lp.Character
    if not char then return "No character" end

    local itemName = item.assetName or item.name
    local itemCat  = item.cat

    if itemCat == "Wings" then
        return equipWings(item)
    end

    -- 1. Direct Server Remote
    if previewItemRemote then
        pcall(function()
            previewItemRemote:FireServer(itemName)
        end)
    end

    -- 2. Parallel Shop TryOn Button Click
    task.spawn(function()
        local btn = findShopTryonButton(item.name) or findShopTryonButton(itemName)
        if btn then
            clickButtonVIM(btn)
        end
    end)

    -- 3. Dynamic Smart Polling: Auto-bakes as soon as meshes replicate, then immediately unfreezes
    task.spawn(function()
        disableFreezeOnlyConnections()
        hideTryOnOverlays()

        local startTick = tick()
        local bakedCount = 0
        while tick() - startTick < 3.2 do
            task.wait(0.12)
            local items = findRoyaleHighItemsOnAvatar()
            if #items > 0 then
                bakedCount = lockCurrentAvatarLook(itemCat, item)
                if bakedCount and bakedCount > 0 then
                    break
                end
            end
        end

        -- Immediately release all movement freezes & terminate try-on anchoring
        releasePlayerMovement()
        updateTabCounts()
        if refreshGridRef then refreshGridRef() end
    end)

    return "OK"
end

local function formatCommas(str)
	local numStr = str:gsub("[^%d]", "")
	if numStr == "" then return "" end
	local k
	while true do
		numStr, k = string.gsub(numStr, "^(-?%d+)(%d%d%d)", '%1,%2')
		if k == 0 then break end
	end
	return numStr
end

local function refreshGrid()
	for _, card in pairs(cardInstances) do card:Destroy() end
	cardInstances = {}

	if currentCat == "Stats" then return end

	local list = {}
	if currentCat == "Worn" then
		list = getWornItemsList()
	else
		list = categories[currentCat] or {}
	end

	local search = currentSearch:lower()
	local filtered = {}
	for _, item in ipairs(list) do
		if search == "" or item.name:lower():find(search, 1, true) then
			table.insert(filtered, item)
		end
	end

	if currentCat == "Worn" then
		if #filtered == 0 then
			local emptyCard = mk("Frame", {
				Name             = "EmptyNotice",
				Size             = UDim2.new(1, -8, 0, 80),
				BackgroundColor3 = C.card,
				BorderSizePixel  = 0,
				LayoutOrder      = 1,
				ZIndex           = 13,
			}, GridScroll)
			corner(4, emptyCard)
			stroke(1, C.border, emptyCard)

			mk("TextLabel", {
				Size               = UDim2.new(1, -12, 1, 0),
				Position           = UDim2.new(0, 6, 0, 0),
				BackgroundTransparency = 1,
				Text               = "NO ITEMS WORN\n\nEquip items from Heels, Skirts, or Bodices to wear and stack them!",
				TextColor3         = C.sub,
				Font               = Enum.Font.Gotham,
				TextSize           = 7.5,
				TextWrapped        = true,
				ZIndex             = 14,
			}, emptyCard)
			cardInstances[emptyCard] = emptyCard
			return
		end
	else
		-- Sort: Items with valid images FIRST, items with NO image LAST
		table.sort(filtered, function(a, b)
			local aHas = (a.img and a.img ~= "") and 1 or 0
			local bHas = (b.img and b.img ~= "") and 1 or 0
			if aHas ~= bHas then
				return aHas > bHas
			end
			return a.name:lower() < b.name:lower()
		end)
	end

	for order, item in ipairs(filtered) do
		makeCard(item, order)
	end
end

refreshGridRef = function(targetCat)
    if not targetCat or targetCat == currentCat then
        refreshGrid()
    end
end
