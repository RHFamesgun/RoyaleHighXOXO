-- 1. SELECTIVE FREEZE NEUTRALIZER & OVERLAY SUPPRESSOR

local function disableFreezeOnlyConnections()
    if getconnections and previewItemRemote then
        pcall(function()
            for _, conn in ipairs(getconnections(previewItemRemote.OnClientEvent)) do
                pcall(function()
                    local fn = conn.Function
                    local isFreeze = false
                    if getconstants and fn then
                        for _, c in ipairs(getconstants(fn)) do
                            if c == "ExitTryOnNonWings" or c == "PrimaryPart" or c == "Anchored" then
                                isFreeze = true
                                break
                            end
                        end
                    end
                    if isFreeze then
                        conn:Disable()
                    end
                end)
            end
        end)
    end
end

local function hideTryOnOverlays()
    pcall(function()
        local hud = pg:FindFirstChild("HUD")
        if hud then
            for _, desc in ipairs(hud:GetDescendants()) do
                if desc:IsA("GuiObject") and (desc.Name:find("ExitTryOn") or desc.Name == "TryOn") then
                    desc.Visible = false
                end
            end
        end
        local shop = pg:FindFirstChild("Shop")
        if shop then
            for _, desc in ipairs(shop:GetDescendants()) do
                if desc:IsA("GuiObject") and desc.Name == "ExitTryOnNonWings" then
                    desc.Visible = false
                end
            end
        end
    end)
end

local function unanchorCharacter()
    local char = lp.Character
    if char then
        local pp = char.PrimaryPart or char:FindFirstChild("HumanoidRootPart")
        if pp and pp.Anchored then pp.Anchored = false end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp and hrp.Anchored then hrp.Anchored = false end

        for _, part in ipairs(char:GetChildren()) do
            if part:IsA("BasePart") and part.Anchored then
                part.Anchored = false
            end
        end

        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            if hum.PlatformStand then hum.PlatformStand = false end
            if hum.Sit then hum.Sit = false end
            if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end

            -- Auto-stop forced idle animation when walking
            if hum.MoveDirection.Magnitude > 0 then
                for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
                    if track.Animation and (track.Animation.Name:lower():find("idle") or track.Animation.Name:lower():find("stand")) then
                        pcall(function() track:Stop(0.1) end)
                    end
                end
            end
        end
    end
end

-- 2. AUTO-STACKING MULTI-WEAR ENGINE

local savedWornLayers = {}

local function deepClone(inst)
    if not inst then return nil end
    local ok, result = pcall(function()
        inst.Archivable = true
        return inst:Clone()
    end)
    return ok and result or nil
end

local function attachModelToCharacter(char, model, cat)
    if not char or not model then return end
    model:SetAttribute("CustomSavedLayer", true)

    local targetLimb = nil
    if cat == "Skirts" then
        targetLimb = char:FindFirstChild("LowerTorso") or char:FindFirstChild("Torso")
    elseif cat == "Bodices" then
        targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    elseif cat == "Heels" then
        targetLimb = char:FindFirstChild("LowerTorso") or char:FindFirstChild("HumanoidRootPart")
    elseif cat == "Wings" then
        targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    else
        targetLimb = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Head") or char.PrimaryPart
    end

    if not targetLimb then return end

    local rootPart = model.PrimaryPart
                  or model:FindFirstChild("Center")
                  or model:FindFirstChild("Middle")
                  or model:FindFirstChild("Handle")
                  or model:FindFirstChildWhichIsA("BasePart")

    if rootPart then
        for _, p in ipairs(model:GetDescendants()) do
            if p:IsA("BasePart") then
                p.Anchored = false
                p.CanCollide = false
            end
        end

        local hasWeld = false
        for _, w in ipairs(model:GetDescendants()) do
            if (w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D")) then
                if w.Part0 == targetLimb or w.Part1 == targetLimb then
                    hasWeld = true
                    break
                end
            end
        end

        if not hasWeld then
            local wc = Instance.new("WeldConstraint")
            wc.Name = "CustomAutoStackWeld"
            wc.Part0 = targetLimb
            wc.Part1 = rootPart
            wc.Parent = rootPart
        end
    end

    model.Parent = char
end

local function restoreAllSavedLayers()
    local char = lp.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")

    for slotKey, data in pairs(savedWornLayers) do
        local existing = char:FindFirstChild(data.name)
        if not existing or not existing:GetAttribute("CustomSavedLayer") then
            if data.backup then
                pcall(function()
                    local clone = deepClone(data.backup)
                    if clone then
                        clone.Name = data.name
                        if clone:IsA("Accessory") and hum then
                            hum:AddAccessory(clone)
                            clone:SetAttribute("CustomSavedLayer", true)
                        else
                            attachModelToCharacter(char, clone, data.category)
                        end
                    end
                end)
            end
        end
    end
end

task.spawn(function()
    while task.wait(0.5) do
        pcall(restoreAllSavedLayers)
    end
end)

local function clearAllWornLayers()
    local char = lp.Character
    if char then
        for _, c in ipairs(char:GetChildren()) do
            if c:GetAttribute("CustomSavedLayer") or c.Name == "Wings_TRYON" or c.Name:find("CustomWing") then
                pcall(function() c:Destroy() end)
            end
        end
    end
    savedWornLayers = {}
    if exitPreviewRemote then
        pcall(function() exitPreviewRemote:FireServer() end)
    end
end

local function addOrUpdateCatalogItem(cat, alias, assetName, price, img, isServerData)
    local key = (assetName or alias):lower()
    local existing = catalogItemByKey[key]

    local pStr = "N/A"
    if type(price) == "number" then
        pStr = string.format("%d", price):reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
    elseif type(price) == "string" and price ~= "" then
        pStr = price
    end

    local imgStr = ""
    if type(img) == "string" and img ~= "" then
        if img:find("rbxassetid://") or img:find("http") then
            imgStr = img
        elseif tonumber(img) then
            imgStr = "rbxassetid://" .. img
        end
    elseif type(img) == "number" and img > 0 then
        imgStr = "rbxassetid://" .. tostring(img)
    end

    if existing then
        if isServerData and imgStr ~= "" then
            existing.img = imgStr
            existing.isServerImg = true
        elseif (not existing.img or existing.img == "") and imgStr ~= "" then
            existing.img = imgStr
        end
        if pStr ~= "N/A" and (existing.price == "N/A" or isServerData) then
            existing.price = pStr
        end
        return
    end

    local itemObj = {
        name        = alias or assetName,
        assetName   = assetName or alias,
        price       = pStr,
        img         = imgStr,
        cat         = cat,
        isServerImg = (isServerData and imgStr ~= "")
    }
    catalogItemByKey[key] = itemObj
    if categories[cat] then
        table.insert(categories[cat], itemObj)
    end
end

-- Live Server Synchronizer
local function syncWithLiveServer()
    local net = ReplicatedStorage:FindFirstChild("Network")
    local fn  = net and net:FindFirstChild("Functions") and net.Functions:FindFirstChild("Gui")
    if not fn then return end

    -- 1. Heels
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("HeelMorphs") and fn.HeelMorphs:FindFirstChild("List") then
                local heels = fn.HeelMorphs.List:InvokeServer()
                if type(heels) == "table" then
                    for assetName, data in pairs(heels) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Heels", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Heels") end
                end
            end
        end)
    end)

    -- 2. Skirts
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Skirts") and fn.Skirts:FindFirstChild("List") then
                local skirts = fn.Skirts.List:InvokeServer()
                if type(skirts) == "table" then
                    for assetName, data in pairs(skirts) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Skirts", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Skirts") end
                end
            end
        end)
    end)

    -- 3. Accessories, Bodices, & Halos
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Accessories") and fn.Accessories:FindFirstChild("List") then
                local accs = fn.Accessories.List:InvokeServer()
                if type(accs) == "table" then
                    for assetName, data in pairs(accs) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        local sub = (data.Type or data.Category or ""):lower()
                        local aliasL = alias:lower()
                        local assetL = assetName:lower()

                        if aliasL:find("halo") or assetL:find("halo") then
                            addOrUpdateCatalogItem("Halos", alias, assetName, data.Price or "Special", img, true)
                        elseif aliasL:find("bodice") or aliasL:find("corset") or aliasL:find("jacket") or sub:find("bodice") or sub:find("torso") then
                            addOrUpdateCatalogItem("Bodices", alias, assetName, data.Price, img, true)
                        else
                            addOrUpdateCatalogItem("Accessories", alias, assetName, data.Price, img, true)
                        end
                    end
                    updateTabCounts()
                    if refreshGridRef then
                        refreshGridRef("Accessories")
                        refreshGridRef("Bodices")
                        refreshGridRef("Halos")
                    end
                end
            end
        end)
    end)

    -- 4. Wings
    task.spawn(function()
        pcall(function()
            if fn:FindFirstChild("Wings") and fn.Wings:FindFirstChild("List") then
                local wings = fn.Wings.List:InvokeServer()
                if type(wings) == "table" then
                    for assetName, data in pairs(wings) do
                        local alias = data.Alias or assetName
                        local img = data.Icon or (data.Icons and data.Icons[1]) or ""
                        addOrUpdateCatalogItem("Wings", alias, assetName, data.Price, img, true)
                    end
                    updateTabCounts()
                    if refreshGridRef then refreshGridRef("Wings") end
                end
            end
        end)
    end)
end

local function findShopTryonButton(itemName)
    local shop = pg:FindFirstChild("Shop")
    if not shop then return nil end
    local dir = shop:FindFirstChild("Directory")
    if not dir then return nil end

    local targetLower = itemName:lower()
    for _, folder in ipairs(dir:GetChildren()) do
        local mid = folder:FindFirstChild("MiddlePanel")
        local sf = mid and mid:FindFirstChild("ScrollingFrame")
        if sf then
            for _, itemFrame in ipairs(sf:GetChildren()) do
                if itemFrame:IsA("Frame") then
                    local tryon = itemFrame:FindFirstChild("Tryon")
                    if tryon and tryon.Visible then
                        local fName = itemFrame.Name:lower()
                        local nLabel = itemFrame:FindFirstChild("name")
                        local lText = (nLabel and nLabel:IsA("TextLabel")) and nLabel.Text:lower() or ""
                        if fName == targetLower or lText == targetLower
                           or fName:find(targetLower, 1, true) or targetLower:find(fName, 1, true)
                           or (lText ~= "" and lText:find(targetLower, 1, true)) then
                            return tryon
                        end
                    end
                end
            end
        end
    end
    return nil
end

local function clickButtonVIM(btn)
    if not btn then return false end
    local ok = pcall(function()
        local pos = btn.AbsolutePosition
        local sz  = btn.AbsoluteSize
        local cx  = pos.X + sz.X / 2
        local cy  = pos.Y + sz.Y / 2
        if VIM then
            VIM:SendMouseButtonEvent(cx, cy, 0, true, game, 1)
            task.wait(0.04)
            VIM:SendMouseButtonEvent(cx, cy, 0, false, game, 1)
        elseif VU then
            VU:Button1Down(Vector2.new(cx, cy), workspace.CurrentCamera)
            task.wait(0.04)
            VU:Button1Up(Vector2.new(cx, cy), workspace.CurrentCamera)
        end
    end)
    return ok
end

    pcall(function()

                 or (pg:FindFirstChild("HUD") and pg.HUD:FindFirstChild("PreviewTogglesFrame", true))

        if togglesMod and ptf then
            local mod = require(togglesMod)
            if mod and mod.OpenToggleOptions then
                ptf.Visible = true
                mod:OpenToggleOptions(assetName, ptf)
                opened = true
            end
        end
    end)

    -- 2. Open Royale High ColorsGui
    pcall(function()
        local cg = pg:FindFirstChild("ColorsGui")
        if cg then
            cg.Enabled = true
            local co = cg:FindFirstChild("ColorOptions")
            if co then co.Visible = true end
            opened = true
        end
    end)

    return opened
end

local function equipWings(item)
    local char = lp.Character
    if not char then return "No character" end
    local assetName = item.assetName or item.name

    for _, c in ipairs(char:GetChildren()) do
        if c.Name == "Wings_TRYON" or c.Name:find("CustomWing") then
            pcall(function() c:Destroy() end)
        end
    end

    local net = ReplicatedStorage:FindFirstChild("Network")
    local fn  = net and net:FindFirstChild("Functions") and net.Functions:FindFirstChild("Gui")
    if fn and fn:FindFirstChild("Wings") and fn.Wings:FindFirstChild("TryOn") then
        local ok, res = pcall(function()
            return fn.Wings.TryOn:InvokeServer(assetName)
        end)
        if ok and res then
            local chest = res.Chest or res:FindFirstChild("Chest") or res
            if chest then
                chest.Name = "Wings_TRYON"
                local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
                local center = chest:FindFirstChild("Center") or chest.PrimaryPart or chest:FindFirstChildWhichIsA("BasePart")
                if torso and center then
                    for _, p in ipairs(chest:GetDescendants()) do
                        if p:IsA("BasePart") then
                            p.Anchored = false
                            p.CanCollide = false
                        end
                    end
                    local weld = Instance.new("WeldConstraint")
                    weld.Name = "CustomWingWeld"
                    weld.Part0 = torso
                    weld.Part1 = center
                    weld.Parent = torso
                    chest.Parent = char
                    chest:SetAttribute("CustomSavedLayer", true)
                    savedWornLayers["Wings"] = {
                        name     = "Wings_TRYON",
                        category = "Wings",
                        backup   = deepClone(chest)
                    }
                    hideTryOnOverlays()
                    return "OK"
                end
            end
        end
    end

    local btn = findShopTryonButton(item.name) or findShopTryonButton(assetName)
    if btn then
        clickButtonVIM(btn)
        hideTryOnOverlays()
        return "OK"
    end
    hideTryOnOverlays()
    return "OK"
end

local function equipItem(item)
    local char = lp.Character
    if not char then return "No character" end

    local itemName = item.assetName or item.name
    local itemCat  = item.cat

    if itemCat == "Wings" then
        return equipWings(item)
    end

    -- Slot replacement: single-slot items replace their own slot, accessories stack
    local slotKey = (itemCat == "Heels" or itemCat == "Skirts" or itemCat == "Bodices") and itemCat or ("Acc_" .. itemName)

    if savedWornLayers[slotKey] then
        local oldName = savedWornLayers[slotKey].name
        local oldInst = char:FindFirstChild(oldName)
        if oldInst then
            oldInst:SetAttribute("CustomSavedLayer", nil)
            pcall(function() oldInst:Destroy() end)
        end
        savedWornLayers[slotKey] = nil
    end

    -- Snapshot all other active layers to keep them preserved
    local otherLayers = {}
    for sk, data in pairs(savedWornLayers) do
        if sk ~= slotKey and data.backup then
            otherLayers[sk] = data
        end
    end

    -- 1. Direct Server Remote
    if previewItemRemote then
        pcall(function()
            previewItemRemote:FireServer(itemName)
        end)
    end

    -- 2. Parallel Shop TryOn Button Click
    task.spawn(function()
        local btn = findShopTryonButton(item.name) or findShopTryonButton(itemName)
        if btn then
            clickButtonVIM(btn)
        end
    end)

    -- Capture replication & protect layer
    task.spawn(function()
        task.wait(0.25)
        disableFreezeOnlyConnections()
        hideTryOnOverlays()

        local newItem = nil
        local targetLower = item.name:lower()
        local assetLower  = itemName:lower()

        for _ = 1, 30 do
            task.wait(0.1)
            char = lp.Character
            if not char then break end
            for _, child in ipairs(char:GetChildren()) do
                if not standardBodyParts[child.Name]
                   and not child:GetAttribute("CustomSavedLayer") then
                    local cln = child.Name:lower()
                    if cln:find(targetLower, 1, true) or targetLower:find(cln, 1, true)
                       or cln:find(assetLower, 1, true) or assetLower:find(cln, 1, true)
                       or child:IsA("Accessory") or child:IsA("Model") or child:IsA("Part") or child:IsA("MeshPart") then
                        newItem = child
                        break
                    end
                end
            end
            if newItem then break end
        end

        if newItem then
            pcall(function()
                newItem:SetAttribute("CustomSavedLayer", true)
                savedWornLayers[slotKey] = {
                    name     = newItem.Name,
                    category = itemCat,
                    backup   = deepClone(newItem)
                }
            end)
        end

        -- Auto-stack all other preserved layers onto the character!
        char = lp.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            for sk, data in pairs(otherLayers) do
                local existing = char:FindFirstChild(data.name)
                if not existing or not existing:GetAttribute("CustomSavedLayer") then
                    if data.backup then
                        pcall(function()
                            local fresh = deepClone(data.backup)
                            if fresh then
                                fresh.Name = data.name
                                if fresh:IsA("Accessory") and hum then
                                    hum:AddAccessory(fresh)
                                    fresh:SetAttribute("CustomSavedLayer", true)
                                else
                                    attachModelToCharacter(char, fresh, data.category)
                                end
                            end
                        end)
                    end
                end
            end
        end

        disableFreezeOnlyConnections()
        hideTryOnOverlays()
    end)

    return "OK"
end

local function openInsp(item)
	selectedItem = item
	inspName.Text = item.name:upper()
	inspPrice.Text = (item.price ~= "" and ("Price: " .. item.price) or "Price: Special")
	if item.img and item.img ~= "" then
		inspImg.Image = item.img
		inspPlaceholder.Visible = false
	else
		inspImg.Image = ""
		inspPlaceholder.Visible = true
	end
	Inspector.Visible = true
	Inspector.Position = UDim2.new(0, 0, 1, -InspHeight)
end

local function handleWear()
	if not selectedItem then return end
	setStatus("Stacking...", C.sub)
	local res = equipItem(selectedItem)
	if res == "OK" then
		setStatus("Layer Stacked!", C.text)
	else
		setStatus("Failed", C.muted)
	end
end
equipBtn.Activated:Connect(handleWear)

local function handleColor()
	if not selectedItem then return end
	local assetName = selectedItem.assetName or selectedItem.name
	local ok = openItemColorAndToggles(assetName)
	if ok then
		setStatus("Colors opened!", C.text)
	else
		setStatus("Open Colors in Dress Up", C.sub)
	end
end
colorBtn.Activated:Connect(handleColor)

	clearAllWornLayers()
	setStatus("Cleared all layers.", C.sub)

local function makeCard(item, order)
	local card = mk("Frame", {
		Name             = "Card_" .. item.name:sub(1,10),
		BackgroundColor3 = C.card,
		BorderSizePixel  = 0,
		LayoutOrder      = order,
		ZIndex           = 13,
	}, GridScroll)
	corner(4, card)
	stroke(1, C.border, card)

	local img = mk("ImageLabel", {
		Size             = UDim2.new(1, -6, 0, 38),
		Position         = UDim2.new(0, 3, 0, 3),
		BackgroundColor3 = C.btn,
		BorderSizePixel  = 0,
		ScaleType        = Enum.ScaleType.Fit,
		Image            = item.img or "",
		ZIndex           = 14,
	}, card)
	corner(3, img)

	if not item.img or item.img == "" then
		mk("TextLabel", {
			Size               = UDim2.new(1, 0, 1, 0),
			BackgroundTransparency = 1,
			Text               = "NO IMG",
			TextColor3         = C.muted,
			Font               = Enum.Font.Gotham,
			TextSize           = 7,
			ZIndex             = 15,
		}, img)
	end

	mk("TextLabel", {
		Size               = UDim2.new(1, -4, 0, 20),
		Position           = UDim2.new(0, 2, 0, 44),
		BackgroundTransparency = 1,
		Text               = item.name,
		TextColor3         = C.text,
		Font               = Enum.Font.Gotham,
		TextSize           = 6.5,
		TextWrapped        = true,
		TextYAlignment     = Enum.TextYAlignment.Top,
		ZIndex             = 14,
	}, card)

	local btn = mk("TextButton", {
		Size               = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text               = "",
		ZIndex             = 16,
	}, card)
	btn.Activated:Connect(function() openInsp(item) end)

	cardInstances[item] = card
	return card
end

local function refreshGrid()
	for _, card in pairs(cardInstances) do card:Destroy() end
	cardInstances = {}

	local list = categories[currentCat] or {}
	local search = currentSearch:lower()
	local order = 0
	for _, item in ipairs(list) do
		if search == "" or item.name:lower():find(search, 1, true) then
			order = order + 1
			makeCard(item, order)
		end
	end
end

refreshGridRef = function(targetCat)
    if not targetCat or targetCat == currentCat then
        refreshGrid()
    end
end

local function setActiveCategory(cat)
	currentCat = cat
	for c, btn in pairs(tabButtons) do
		if c == cat then
			btn.BackgroundColor3 = C.tabActive
			btn.TextColor3       = C.tabActTxt
		else
			btn.BackgroundColor3 = C.tabInact
			btn.TextColor3       = C.tabInaTxt
		end
	end
	closeInsp()
	refreshGrid()
end
