-- GUI host: gethui > CoreGui > PlayerGui
local function getGuiParent()
    if gethui then return gethui() end
    local ok, cg = pcall(function() return game:GetService("CoreGui") end)
    if ok and cg then return cg end
    return pg
end

-- ============================================================
-- SECTION 3: ANTI-FREEZE & ANIMATION GUARDIAN
-- ============================================================
-- RH runs a client script that loops:
-- while v_u_4 do Character.PrimaryPart.Anchored = true ... end
-- We constantly force PrimaryPart and HRP to unanchored and ensure animations play.

local standardBodyParts = {
    HumanoidRootPart=true, Head=true, Torso=true, UpperTorso=true, LowerTorso=true,
    ["Left Arm"]=true, ["Right Arm"]=true, ["Left Leg"]=true, ["Right Leg"]=true,
    LeftUpperArm=true, LeftLowerArm=true, LeftHand=true,
    RightUpperArm=true, RightLowerArm=true, RightHand=true,
    LeftUpperLeg=true, LeftLowerLeg=true, LeftFoot=true,
    RightUpperLeg=true, RightLowerLeg=true, RightFoot=true,
    Animate=true, Humanoid=true, HumanoidDescription=true,
    BodyColors=true, Health=true, Status=true,
}

local function restoreCharacterMobility(char)
    if not char then return end
    pcall(function()
        local pp = char.PrimaryPart or char:FindFirstChild("HumanoidRootPart")
        if pp and pp.Anchored then
            pp.Anchored = false
        end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp and hrp.Anchored then
            hrp.Anchored = false
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            if hum.PlatformStand then hum.PlatformStand = false end
            if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
            if hum.JumpPower < 50 then hum.JumpPower = 50 end
        end
        local animate = char:FindFirstChild("Animate")
        if animate and animate:IsA("LocalScript") and animate.Disabled then
            animate.Disabled = false
        end
    end)
end

-- Hook RenderStepped to ensure the character NEVER gets anchored by RH's Preview loop
RunService.RenderStepped:Connect(function()
    local char = lp.Character
    if char then
        local pp = char.PrimaryPart or char:FindFirstChild("HumanoidRootPart")
        if pp and pp.Anchored then
            pp.Anchored = false
        end
    end
end)

-- Periodically clear any freeze context bindings
task.spawn(function()
    while task.wait(0.5) do
        pcall(function()
            ContextActionService:UnbindAction("FreezeMovement")
            ContextActionService:UnbindAction("ShopFreeze")

            ContextActionService:UnbindAction("DisableMovement")
            ContextActionService:UnbindAction("CharacterCreationFreeze")
            ContextActionService:UnbindAction("TryOnControl")
        end)
        restoreCharacterMobility(lp.Character)
    end
end)

-- ============================================================
-- SECTION 4: MULTI-LAYER ENGINE
-- ============================================================
local savedWornLayers = {} -- [itemName] = { backup = Instance, category = string }

local function deepClone(inst)
    if not inst then return nil end
    local ok, result = pcall(function()
        inst.Archivable = true
        return inst:Clone()
    end)
    return ok and result or nil
end

local function restoreAllSavedLayers()
    local char = lp.Character
    if not char then return end
    for name, data in pairs(savedWornLayers) do
        local existing = char:FindFirstChild(name)
        if not existing or not existing:GetAttribute("CustomSavedLayer") then
            if data.backup then
                pcall(function()
                    local clone = deepClone(data.backup)
                    if clone then
                        clone.Name = name
                        clone:SetAttribute("CustomSavedLayer", true)
                        clone.Parent = char
                    end
                end)
            end
        end
    end
end

-- Layer safeguard loop: re-parent any layer if stripped
task.spawn(function()
    while task.wait(0.5) do
        pcall(restoreAllSavedLayers)
    end
end)

local function clearAllWornLayers()
    local char = lp.Character
    if char then
        for name, _ in pairs(savedWornLayers) do
            local obj = char:FindFirstChild(name)
            if obj then
                obj:SetAttribute("CustomSavedLayer", nil)
                pcall(function() obj:Destroy() end)
            end
        end
    end
    savedWornLayers = {}
end

-- ============================================================
-- SECTION 5: DIRECT REMOTE TRY-ON ENGINE
-- ============================================================
local function equipItem(itemName, itemCat)
    local char = lp.Character
    if not char then return "No character" end

    -- Slot replacement for single-slot items (Heels, Skirts, Bodices)
    if itemCat == "Heels" or itemCat == "Skirts" or itemCat == "Bodices" then
        for savedName, data in pairs(savedWornLayers) do
            if data.category == itemCat then
                local existing = char:FindFirstChild(savedName)
                if existing then
                    existing:SetAttribute("CustomSavedLayer", nil)
                    pcall(function() existing:Destroy() end)
                end
                savedWornLayers[savedName] = nil
            end
        end
    end

    -- Snapshot existing custom layers before preview
    local snapshot = {}
    for _, child in ipairs(char:GetChildren()) do
        if child:GetAttribute("CustomSavedLayer") then
            snapshot[child.Name] = deepClone(child)
        end
    end

    -- Call the exact native remote Royale High uses for Try-On
    if previewItemRemote then
        pcall(function()
            previewItemRemote:FireServer(itemName)
        end)
    end

    -- Process replication & isolate layer
    task.spawn(function()
        task.wait(0.25)
        restoreCharacterMobility(lp.Character)

        -- Find newly replicated asset
        local newItem = nil
        local lowerTarget = itemName:lower()
        for _ = 1, 30 do
            task.wait(0.1)
            char = lp.Character
            if not char then break end
            for _, child in ipairs(char:GetChildren()) do
                if not standardBodyParts[child.Name]
                   and not child:GetAttribute("CustomSavedLayer")
                   and not snapshot[child.Name] then
                    local cln = child.Name:lower()
                    if cln:find(lowerTarget, 1, true)
                       or lowerTarget:find(cln, 1, true)
                       or child:IsA("Accessory") or child:IsA("Model") or child:IsA("Part") or child:IsA("MeshPart") then
                        newItem = child
                        break
                    end
                end
            end
            if newItem then break end
        end

        if newItem then
            pcall(function()
                newItem:SetAttribute("CustomSavedLayer", true)
                savedWornLayers[itemName] = {
                    backup   = deepClone(newItem),
                    category = itemCat
                }
            end)
        end

        -- Restore previously saved layers that may have been cleared
        char = lp.Character
        if char then
            for snapName, snapClone in pairs(snapshot) do
                if not char:FindFirstChild(snapName) or not char:FindFirstChild(snapName):GetAttribute("CustomSavedLayer") then
                    if snapClone then
                        pcall(function()
                            local fresh = deepClone(snapClone)
                            if fresh then
                                fresh.Name = snapName
                                fresh:SetAttribute("CustomSavedLayer", true)
                                fresh.Parent = char
                            end
                        end)
                    end
                end
            end
        end

        restoreCharacterMobility(lp.Character)
    end)

    return "OK"
end

-- Dragging
local dragging, dragStart, startPos
mainFrame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging  = true
		dragStart = input.Position
		startPos  = mainFrame.Position
	end
end)
mainFrame.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)
mainFrame.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)

local selectedItem = nil

	selectedItem   = item
	inspImg.Image  = item.img or ""
	inspName.Text  = item.name or ""
	inspPrice.Text = "Price: " .. (item.price or "N/A")
	inspCat.Text   = "Category: " .. (item.cat or "")
	inspector.Visible  = true
	inspector.Size     = UDim2.new(1, 0, 0, 148)
	inspector.Position = UDim2.new(0, 0, 1, -148)
	mainFrame.Size     = UDim2.new(0, SCREEN_W, 0, SCREEN_H + 148)
end

local function handleWear()
	if not selectedItem then return end
	if selectedItem.isHalo then
		setStatus("Halos cannot be tried on.\nMust be won or traded!", C.subtext)
		return
	end
	setStatus("Equipping...", C.subtext)
	local result = equipItem(selectedItem.name, selectedItem.cat)
	if result == "OK" then
		setStatus("Equipped!", C.text)
	else
		setStatus("Failed: " .. tostring(result), C.subtext)
	end
end

local currentCat    = "Heels"

local cardInstances = {}

local function makeCard(item, order)
	local card = Instance.new("Frame")
	card.Name             = "Card_" .. item.name:sub(1,20)
	card.BackgroundColor3 = C.card
	card.BorderSizePixel  = 0
	card.LayoutOrder      = order
	card.Parent           = gridScroll
	local ucCard = Instance.new("UICorner")
	ucCard.CornerRadius = UDim.new(0, 6)
	ucCard.Parent = card

	local img = Instance.new("ImageLabel")
	img.Size             = UDim2.new(1, -8, 0, 68)
	img.Position         = UDim2.new(0, 4, 0, 4)
	img.BackgroundColor3 = C.imgBg
	img.BorderSizePixel  = 0
	img.ScaleType        = Enum.ScaleType.Fit
	img.Image            = item.img or ""
	img.Parent           = card
	local ucImg = Instance.new("UICorner")
	ucImg.CornerRadius = UDim.new(0, 4)
	ucImg.Parent = img

	if not item.img or item.img == "" then
		local noImg = Instance.new("TextLabel")
		noImg.Text              = "NO IMG"
		noImg.Font              = Enum.Font.Gotham
		noImg.TextSize          = 9
		noImg.TextColor3        = C.subtext
		noImg.BackgroundTransparency = 1
		noImg.Size              = UDim2.new(1, 0, 1, 0)
		noImg.TextXAlignment    = Enum.TextXAlignment.Center
		noImg.Parent            = img
	end

	local nameLbl = Instance.new("TextLabel")
	nameLbl.Text             = item.name
	nameLbl.Font             = Enum.Font.Gotham
	nameLbl.TextSize         = 9
	nameLbl.TextColor3       = C.text
	nameLbl.BackgroundTransparency = 1
	nameLbl.Size             = UDim2.new(1, -4, 0, 26)
	nameLbl.Position         = UDim2.new(0, 2, 0, 76)
	nameLbl.TextWrapped      = true
	nameLbl.TextXAlignment   = Enum.TextXAlignment.Center
	nameLbl.TextYAlignment   = Enum.TextYAlignment.Top
	nameLbl.Parent           = card

	local btn = Instance.new("TextButton")
	btn.Text              = ""
	btn.BackgroundTransparency = 1
	btn.Size              = UDim2.new(1, 0, 1, 0)
	btn.Parent            = card
	btn.ZIndex            = 2
	btn.MouseButton1Click:Connect(function() openInspector(item) end)
	btn.TouchTap:Connect(function() openInspector(item) end)

	cardInstances[item] = card
	return card
end

local function refreshGrid()
	for _, card in pairs(cardInstances) do card:Destroy() end
	cardInstances = {}

	local list = categories[currentCat] or {}
	local search = currentSearch:lower()
	local order = 0
	for _, item in ipairs(list) do
		local match = search == "" or item.name:lower():find(search, 1, true)
		if match then
			order = order + 1
			makeCard(item, order)
		end
	end
end

local function setActiveTab(cat)
	currentCat = cat
	for c, btn in pairs(tabButtons) do
		if c == cat then
			btn.BackgroundColor3 = C.tabActive
			btn.TextColor3       = C.tabActTxt
		else
			btn.BackgroundColor3 = C.tabInact
			btn.TextColor3       = C.tabInaTxt
		end
	end
	refreshGrid()
end
